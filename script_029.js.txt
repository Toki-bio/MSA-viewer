// The changes called from every read (SampleCall), each with its depth, allele fraction, reads per strand and its

// message: no depth, no strand, no codon, no amino-acid change, no way back to the reads.

const _readsRefChar = p => bamState.refSeq[p - (bamState.refBase || 0)];
function _readsRefRowIndex() {

function _readsGenes() {
    const ri = _readsRefRowIndex();
    if (ri < 0 || !state.annot || !state.annot.features) return [];
    return state.annot.features.filter(f => _annotResolveChrom(f.chrom) === ri).map(f => ({
        name: f.name, strand: f.strand || '+', coding: _annotIsCds(f), start: f.start, end: f.end,
        cdsBlocks: f.cdsBlocks && f.cdsBlocks.length ? f.cdsBlocks : [[f.start, f.end]], blocks: f.blocks, codonStart: f.codonStart || 1
    }));
}
// one row per variant and gene: { v, e } (e from VariantEffect.annotate)
function _readsVariantRows() {
    const calls = _readsCalls();
    if (!calls) return [];
    const genes = _readsGenes(), code = _getActiveCode();
    const rows = [];
    for (const v of calls.variants) for (const e of VariantEffect.annotate(v, bamState.refSeq, genes, code)) rows.push({ v, e });
    return rows;
}
function _variantStrandNote(v) {
    const alt = (v.altFwd || 0) + (v.altRev || 0);
    if (alt >= 6 && Math.min(v.altFwd || 0, v.altRev || 0) / alt < 0.1) return 'one strand';
    return '';
}
const _VARIANT_COLUMNS = ['Position', 'Change', 'Zygosity', 'Depth', 'Alt %', 'Alt reads +/-', 'Ref reads +/-', 'Gene', 'Effect', 'c.', 'p.', 'Codon', 'Note'];
function _variantCells(r) {
    const v = r.v, e = r.e;
    const change = v.type === 'snv' ? `${v.ref}>${v.alt}` : v.type === 'del' ? `del ${v.ref}` : `ins ${v.alt}`;
    const notes = [];
    if (_variantStrandNote(v)) notes.push('alt reads on one strand only');
    if (v.depth < 10) notes.push('low depth');
    if (v.type !== 'snv' && v.zyg === 'het') notes.push('heterozygous indel: not in the sample row');
    // an insertion sits between two bases: 'a^b' (as HGVS), not the number of the base after it
    const posTxt = v.type === 'ins' ? `${v.pos - 1}^${v.pos}` : String(v.pos);
    return [posTxt + (bamState.offset ? ` (${bamState.bamRefName}:${v.pos + bamState.offset})` : ''), change, v.zyg, String(v.depth),
        String(Math.round(100 * v.frac)), `${v.altFwd || 0}/${v.altRev || 0}`, v.type === 'snv' ? `${v.refFwd || 0}/${v.refRev || 0}` : '',
        e.gene || '', e.effect, e.c || '', e.p || '', e.codon || '', notes.join('; ')];
}
function _variantsTsv() {

}

// The column of reference position p (0-based) in the alignment (any view), selected and scrolled to
function variantToAlignment(p) {
    const ri = _readsRefRowIndex();
    if (ri < 0) return;
    const seq = state.seqs[ri].seq;
    let col = -1;
    for (let c = 0, k = 0; c < seq.length; c++) { if (seq[c] === '-' || seq[c] === '.') continue; if (k === p) { col = c; break; } k++; }
    if (col < 0) return;
    _leaveReadsForCanvas();
    state.selectedColumns.clear();
    state.selectedColumns.add(col);
    state.lastSelectedColumn = col;
    renderAlignment();
    _scrollToColumn(col);
    showMessage(`Column ${(col + 1).toLocaleString()} (${bamState.refName}:${p + 1}) selected`, 2500);
}

// The sample called from every read (cached until other reads are loaded): the coverage row, the variants
// panel and Sample -> row all read the same counts and calls
// The reference row may have been edited since the reads were loaded: their mismatches and the sample row were
// computed against the old bases. A base edit (same length) is taken over and the calls redone; a change of length
// can shift every position, so it is said (reload the reads)
function _readsSyncRef() {
    if (_genome || !bamState.refSeq || !bamState.refName) return;
    const ri = state.seqs.findIndex(s => String(s.fullHeader || s.header || '').trim().split(/\s+/)[0] === bamState.refName);
    if (ri < 0) return;
    const cur = state.seqs[ri].seq.replace(/[-.]/g, '').toUpperCase();
    if (cur === bamState.refSeq) return;
    if (cur.length !== bamState.refSeq.length && !bamState._lenWarned) {
        bamState._lenWarned = true;
        showMessage(`${bamState.refName} changed length since its reads were loaded (${bamState.refSeq.length} -> ${cur.length} bp): read positions may be shifted. Open the reads file again.`, 8000);
    }
    bamState.refSeq = cur;
    bamState.calls = null;
}

function _readsCalls() {
    if (_genome) return _genomeCalls();
    _readsSyncRef();
    if (!bamState.refSeq || !(bamState.allReads || bamState.reads || []).length || typeof SampleCall === 'undefined') return null;
    if (!bamState.calls) bamState.calls = SampleCall.call(bamState.allReads || bamState.reads, bamState.refSeq);
    return bamState.calls;
}

// Sort the reads at reference position p (0-based) as igv.js does: the most frequent non-reference base first,
// then the other bases, deletions, the reference base, and rows without a read there last. Only the order of
// the packed tracks changes (each track holds non-overlapping reads, so at most one covers p).
function _readBaseAt(read, p) {
    if (read.start != null && (p < read.start || p > read.end)) return null;   // and the read stays packed
    let q = read.pos, rp = 0;
    for (const { op, len } of read.cigar) {
        if (op === 'M' || op === '=' || op === 'X') {
            if (p >= q && p < q + len) { const c = (read.seq[rp + p - q] || 'N').toUpperCase(); return c === '=' ? (_readsRefChar(p) || 'N').toUpperCase() : c; }   // '=': the reference base
            q += len; rp += len;
        }
        else if (op === 'D') { if (p >= q && p < q + len) return '-'; q += len; }
        else if (op === 'N') { if (p >= q && p < q + len) return null; q += len; }
        else if (op === 'I' || op === 'S') rp += len;
        if (q > p) break;
    }
    return null;
}

/**
 * Greedy track-packing for read piles (tview/IGV-style).
 *
 * Sorts reads by start position, then walks the track list and assigns each
 * read to the first track whose last-occupied end is <= this read's start.
 * Non-overlapping reads share a track; overlapping reads get separate tracks.
 *
 * Sorts the input array in place and sets `.track` (0-based) on each read.
 * Returns the total number of tracks needed.
 *
 * Examples (start/end are 0-based, end inclusive):
 *
 *   reads = [{s:0,e:9}, {s:10,e:19}, {s:5,e:14}]
 *   sort by start: [0-9, 5-14, 10-19]
 *   read 0-9   -> track 0 (tracks[0] = 10)
 *   read 5-14  -> 5 < 10, new track 1 (tracks[1] = 15)
 *   read 10-19 -> 10 >= 10, reuse track 0 (tracks[0] = 20)
 *   result: 2 tracks
 *
 *   reads = [{s:0,e:4}, {s:5,e:9}, {s:10,e:14}]
 *   all non-overlapping -> all on track 0
 *   result: 1 track
 *
 *   reads = [{s:0,e:9}, {s:0,e:9}, {s:0,e:9}]
 *   all overlap at same position -> 3 tracks
 *   result: 3 tracks
 *
 *   reads = [{s:0,e:4}, {s:3,e:7}, {s:6,e:10}, {s:9,e:13}]
 *   read 0-4  -> track 0 (tracks[0] = 5)
 *   read 3-7  -> 3 < 5, new track 1 (tracks[1] = 8)
 *   read 6-10 -> 6 >= 5, reuse track 0 (tracks[0] = 11)
 *   read 9-13 -> 9 >= 8, reuse track 1 (tracks[1] = 14)
 *   result: 2 tracks
 *
 * @param {Array} reads - objects with numeric `start` and `end` (0-based, inclusive)
 * @returns {number} total number of tracks assigned
 */
function assignReadTracks(reads) {
    reads.sort((a, b) => a.start - b.start);

    // tracks[t] = next available start position for track t (end+1 of last read)
    const tracks = [];
    for (const read of reads) {
        let assigned = false;
        for (let t = 0; t < tracks.length; t++) {
            if (read.start >= tracks[t]) {
                tracks[t] = read.end + 1;
                read.track = t;
                assigned = true;
                break;
            }
        }
        if (!assigned) {
            read.track = tracks.length;
            tracks.push(read.end + 1);
        }
    }
    return tracks.length;
}

/**
 * Show/hide the BAM button based on whether a reference sequence is loaded.
 * Only show when exactly one reference sequence is loaded (not zero, not multiple).
 */
function updateBamButtonVisibility() {
    const bamBtn = document.getElementById('openBamToolbarButton');
    if (!bamBtn) return;
    // one row (a reference), or reads already loaded, or a sample row from reads: with exactly one row only, the
    // button went away once a sample row was added, so a second sample could not be loaded. Not on every
    // alignment: one more toolbar button made the bar wrap when the Selections menu appears on a selection, and
    // the rows jumped under the mouse between two clicks (1200 px wide)
    const eligible = !!(state.seqs && (state.seqs.length === 1 || (bamState.reads && bamState.reads.length) || state.seqs.some(s => s._fromReads != null)));
    bamBtn.style.display = eligible ? '' : 'none';

    // Also update the "Clear reads" button visibility
    const clearBtn = ensureClearReadsButton();
    if (clearBtn) {
        clearBtn.style.display = (eligible && bamState.reads.length > 0) ? '' : 'none';
    }
}

/**
