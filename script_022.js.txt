// Where a feature label sticks while its box is partly scrolled off: just right of the
// name column. Measured from a rendered name cell, once per name width and font size.
let _annotLabelLeftKey = null;
function _syncAnnotLabelLeft() {
    const key = effectiveNameLength() + ':' + (alignmentContainer.style.fontSize || '');
    if (key === _annotLabelLeftKey) return;
    const nm = alignmentContainer.querySelector('.seq-line[data-seq-index] > .seq-name');
    if (!nm) return;   // first block of a windowed render: measured on the next one
    const left = nm.getBoundingClientRect().right - alignmentContainer.getBoundingClientRect().left + 4;
    alignmentContainer.style.setProperty('--annot-label-left', Math.max(0, left).toFixed(1) + 'px');
    _annotLabelLeftKey = key;
}

function _annotTooltipHtml(d) {
    const f = d.f;
    const fmt = n => n.toLocaleString();
    const where = d.seqIndex === 'cols' ? 'alignment columns' : _escapeHtml(state.seqs[d.seqIndex]?.header || f.chrom);
    let h = `<b>${_escapeHtml(f.name)}</b>`;
    if (f.desc) h += `<br>${_escapeHtml(f.desc)}`;
    h += `<br>${where}: ${fmt(f.start + 1)}&ndash;${fmt(f.end)}${f.strand ? ' (' + f.strand + ')' : ''}, ${fmt(f.end - f.start)} bp`;
    if (f.cdsBlocks && f.cdsBlocks.length > 1) h += `<br>${f.cdsBlocks.length} coding exons, ${fmt(f.cdsBlocks.reduce((t, [a, b]) => t + b - a, 0))} bp coding`;
    if (d.seqIndex !== 'cols') h += `<br>columns ${fmt(d.cs + 1)}&ndash;${fmt(d.ce + 1)}`;
    if (f.score != null) h += `<br>score ${f.score}`;
    return h;
}

// The slots of a feature's columns a..b: first to last shown column (a fold inside stays under the box), null when all
// are hidden. The cells are drawn by slot; by column the box stood over other residues once columns were hidden.
function _annotSlotSpan(a, b) {
    const so = _canvasState.slotOf;
    if (!so) return [a, b];
    let c0 = Math.max(0, a), c1 = Math.min(so.length - 1, b);
    while (c0 <= c1 && so[c0] < 0) c0++;
    while (c1 >= c0 && so[c1] < 0) c1--;
    return c0 <= c1 ? [so[c0], so[c1]] : null;
}

        // differently from the same path uncut: the frame copy then kept pixels a full redraw does not

        // from the text of the other bands, whose order changed their antialiasing after a read-back.

// The feature under a Canvas-mode pointer position, or null.
function _canvasHitTestAnnot(clientX, clientY) {
    const L = state._annotLayout;
    const canvas = _canvasState.canvas;
    const m = _canvasState.metrics;
    if (!L || !canvas || !m || !m.trackH) return null;
    // _canvasLocal: in the Block view every band has its own track at its top (it read band 1's)
    const P = _canvasLocal(clientX, clientY);
    const x = P.x;
    const y = P.y + P.oy;
    if (y < 0 || y >= m.trackH || x < m.nameW) return null;
    const slot = Math.floor((x - m.nameW + P.ox) / m.charW);
    if (slot >= P.s1) return null;   // past the band's last column
    const lane = Math.floor(y / m.charH);
    const d = L.feats.find(q => { if (q.lane !== lane) return false; const sp = _annotSlotSpan(q.cs, q.ce); return !!sp && sp[0] <= slot && sp[1] >= slot; });
    return d ? { d, col: _canvasSlotToCol(slot), lane } : null;
}

// Loading, clearing, the Settings panel
function setAnnotation(text, sourceName) {
    const parsed = parseBedText(text);
    if (!parsed.features.length) {
        showMessage(`No BED features found in ${sourceName}`, 4000);
        return false;
    }
    state.annot = { text: String(text), sourceName, trackName: parsed.trackName, features: parsed.features, skippedLines: parsed.skipped };
    // A mitochondrial gene set (tRNAs, rRNAs, ND/COX/ATP/CYTB) with the Standard code
    // still selected: switch to the vertebrate mitochondrial code, and say so. A whole
    // mitogenome has many such features; a single-gene file (one COX1 feature) counts when
    // every feature in it has a mitochondrial gene name
    const mt = parsed.features.filter(f => /^(trn|tRNA|ND\d|NAD\d|COX\d|CO[123]\b|ATP[68]|CYTB|COB\b|12S|16S|rrn|control region|D-loop|OL$)/i.test(f.name)).length;
    const codeSel = el('codonCode');
    if ((mt >= 10 || mt === parsed.features.length) && codeSel && codeSel.value === '1') {
        codeSel.value = '2';
        showMessageAfterRender('Annotation looks mitochondrial: genetic code set to Vertebrate Mito (Display > Code)', 5000);
    }
    if (state.seqs?.length) renderAlignment(); else updateAnnotPanel();
    showMessage(`Annotation: ${parsed.features.length} feature${parsed.features.length === 1 ? '' : 's'} from ${sourceName}`, 3000);
    return true;
}

// A GenBank location as sorted 0-based half-open blocks: join(), order(), complement(), partial < >, single bases.
// Positions in another record (ACC:12..40) and between-base sites (12^13) are left out.
function _gbLocationBlocks(loc) {
    const s = String(loc || '').replace(/\s+/g, '');
    if (/[A-Za-z0-9_.]+:\d/.test(s.replace(/^(complement|join|order)\(/g, ''))) return null;
    const blocks = [];
    for (const m of s.matchAll(/<?(\d+)(?:\.\.>?(\d+))?/g)) {
        if (s[m.index + m[0].length] === '^' || s[m.index - 1] === '^') continue;
        const a = +m[1], b = m[2] ? +m[2] : a;
        if (a >= 1 && b >= a) blocks.push([a - 1, b]);
    }
    blocks.sort((x, y) => x[0] - y[0]);
    return blocks.length ? { blocks, strand: /complement\(/.test(s) ? '-' : '+' } : null;
}

// The CDS, tRNA and rRNA features of GenBank records as the annotation track (BED12, so snapshots keep it):
// they were parsed and dropped, so codon analysis never knew a GenBank file's genes and "Auto" guessed a frame
// from ATG counts. A BED the user loaded is not replaced; a GenBank file's own features are. /transl_table
// sets the genetic code.
function _annotFromGenBank(seqs) {
    if (state.annot && !state.annot.fromGenBank) return;
    const lines = [];
    let table = null;
    for (const s of seqs || []) {
        const feats = s._genbank && s._genbank.features;
        if (!feats || !feats.length) continue;
        const chrom = String(s.header || '').split(/\s+/)[0];
        for (const f of feats) {
            if (!/^(CDS|tRNA|rRNA)$/.test(f.type)) continue;
            const loc = _gbLocationBlocks(f.location);
            if (!loc) continue;
            const q = f.qualifiers || {};
            const name = String(q.gene || q.locus_tag || q.product || f.type).replace(/\s+/g, '_');
            const start = loc.blocks[0][0], end = loc.blocks[loc.blocks.length - 1][1];
            const cds = f.type === 'CDS';
            if (cds && q.transl_table && !table) table = String(q.transl_table);
            const desc = cds ? `CDS codon_start=${/^[123]$/.test(String(q.codon_start)) ? q.codon_start : 1}${q.product ? ' ' + q.product : ''}`
                : `${f.type}${q.product ? ' ' + q.product : ''}`;
            lines.push([chrom, start, end, name, 0, loc.strand, start, cds ? end : start, '.', loc.blocks.length,
                loc.blocks.map(b => b[1] - b[0]).join(',') + ',', loc.blocks.map(b => b[0] - start).join(',') + ',', desc].join('\t'));
        }
    }
    if (!lines.length) {
        if (state.annot && state.annot.fromGenBank) state.annot = null;     // the previous GenBank's genes are not this file's
        return;
    }
    const text = lines.join('\n') + '\n';
    const parsed = parseBedText(text);
    state.annot = { text, sourceName: 'GenBank features', trackName: 'GenBank features', features: parsed.features, skippedLines: parsed.skipped, fromGenBank: true };
    const codeSel = el('codonCode');
    if (table && codeSel && [...codeSel.options].some(o => o.value === table) && codeSel.value !== table) {
        codeSel.value = table;
        showMessageAfterRender(`GenBank /transl_table=${table}: genetic code set to ${codeSel.options[codeSel.selectedIndex].text} (Display > Code)`, 5000);
    }
}

function loadAnnotationFromUrl(u) {
    const url = String(u || '').trim();
    if (!url) return Promise.resolve(false);
    showMessage('Loading annotation...', 0);
    return fetch(url)
        .then(r => { if (!r.ok) throw new Error('HTTP ' + r.status); return r.text(); })
        .then(t => setAnnotation(t, url.split('/').pop() || url))
        .catch(err => {
            console.warn('[annot] load failed:', err);
            showMessage(`Failed to load annotation: ${err.message}`, 5000);
            return false;
        });
}

function generateScale(maxLength, interval = 10, startPos = 0) {
    const scaleArray = new Array(maxLength).fill(' ');

    // Find the first multiple of interval in this block's range
    const startAbs = startPos + 1;
    const endAbs = startPos + maxLength;
    const firstMultiple = Math.ceil(startAbs / interval) * interval;

    // Show position markers at multiples of interval within this block
    for (let absPos = firstMultiple; absPos <= endAbs; absPos += interval) {   // endAbs is the block's last column
        let label;
        // Alternate: 10 number, 20 *, 30 number, 40 *, etc.
        if ((absPos / interval) % 2 === 1) {
            label = absPos.toString();
        } else {
            label = '*';
        }

        const relIdx = absPos - startPos - 1; // 0-based index for the target position
        const startIdx = relIdx - (label.length - 1); // Right-align: last digit at relIdx

        // Only place the label if it fits completely within the block
        if (startIdx >= 0 && startIdx + label.length <= maxLength) {
            for (let i = 0; i < label.length; i++) {
                scaleArray[startIdx + i] = label[i];
            }
        }
    }

    return scaleArray.join('');
}

/**
 * The ruler's labels when columns are hidden (Variable sites only). generateScale() writes one character per
 * column, so with hidden columns a number such as "230" lost its middle digit and left scraps ("2", "0").
 * Here each number is one whole label, right-aligned on its own column like generateScale's, and a label is
 * left out when its column is hidden, when it would not fit before the first shown column (lo), or when it would
 * touch the previous label. cols[s] is the column in slot s (a negative entry is a marker's slot, -1 minus the column after it; it carries no label).
 * Returns [slot of the label's last character, text] for the slots in [lo, hi), in order.
 */
