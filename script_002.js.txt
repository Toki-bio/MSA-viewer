// ---- Residue overlay: every mark a residue can carry, combined by one rule ----
// Search hits, TSD marks, repeat highlights and SNP-group letters used to be drawn by three
// mechanisms (CSS classes with !important, inline !important styles, painting once), so which
// one showed was decided by selector specificity and by which feature ran last. Now one
// function decides, for DOM and Canvas alike:
//   fill (first wins): search hit > TSD colour > repeat tint > SNP letter
//   text channels, always added: TSD -> bold + a second font (sans-serif), plus underline for
//   the "bold" style; restriction site -> italic; trimmed residue -> line-through.
//   A mark whose fill is taken keeps a text channel: a repeat -> overline in its colour, an
//   SNP letter -> text in the group colour, bold.
// Selections are not fills: they are tints drawn over whatever is there (CSS background-image
// in the DOM, translucent rectangles in Canvas).
const _OVERLAY_SANS = "'ViewAlign Sans', Arial, Helvetica, 'Liberation Sans', sans-serif";   // metrics: see styles.css

function _overlayRowCtx(row) {
    const q = state.seqs[row];
    const ctx = {
        row,
        header: q?.header,
        hits: state.searchHistory?.length ? _getSearchHitsForRow(row) : _EMPTY_HITS,
        tsd: state.tsdMarks.size ? state.tsdMarks.get(row) : null,
        tsdPrev: state.tsdPreview?.size ? state.tsdPreview.get(row) : null,
        dotReg: state.dotRegionMarks?.size ? state.dotRegionMarks.get(row) : null,   // a dot-plot stretch shown here
        rep: state.repeatHighlights?.size ? _repeatColsForRow(row) : _EMPTY_HITS,
        snp: !!state._clusterCharMap,
        trim: state.trimBoundaries || null
    };
    ctx.any = !!(ctx.hits.size || ctx.tsd?.size || ctx.tsdPrev?.size || ctx.dotReg?.size || ctx.rep.size || ctx.snp);
    return ctx;
}

function _tsdColour() {
    return /^#[0-9A-Fa-f]{6}$/.test(state.tsdMarkColor || '') ? state.tsdMarkColor : '#ffd54f';
}

// null, or { bg, fg, bold, sans, italic, deco: [], decoColor, titles: [], hit, tsd, rep, snp, covered: [] }
function residueOverlay(ctx, pos, base) {
    if (!ctx.any) return null;
    const hit = ctx.hits.get(pos);
    const tsd = !!ctx.tsd?.has(pos);
    const tsdPrev = !tsd && !!ctx.tsdPrev?.has(pos);   // found by the last TSD run, not marked yet
    const dotReg = !!ctx.dotReg?.has(pos);
    const rep = ctx.rep.get(pos);
    const snp = ctx.snp ? _clusterCharDiag(ctx.header, pos, base) : null;
    if (!hit && !tsd && !tsdPrev && !dotReg && !rep && !snp) return null;
    const o = { bg: null, fg: null, bold: false, sans: false, italic: false, deco: [], decoColor: null,
        titles: [], hit: hit || null, tsd, tsdPrev, dotReg, rep: rep || null, snp, covered: [] };
    if (hit) {
        o.bg = hit.color || '#ffcc00';
        o.fg = '#000';
        o.bold = true;
        if (/restriction/.test(hit.strand || '')) o.italic = true;
    }
    if (tsd) {
        o.bold = true;
        o.sans = true;
        o.titles.push('TSD');
        if (state.tsdMarkStyle === 'color') {
            if (!o.bg) { o.bg = _tsdColour(); o.fg = '#111'; } else o.covered.push('tsd');
        } else if (state.tsdMarkStyle === 'bold') {
            o.deco.push('underline');
        }
    }
    if (tsdPrev) {
        o.titles.push('TSD found (not marked yet)');
        if (!o.bg) { o.bg = _tsdColour() + '55'; o.fg = '#111'; }   // a third of the mark's strength
    }
    if (rep) {
        o.titles.push('repeat region');
        // the tint is light: dark text, or white text from dark shading becomes unreadable
        if (!o.bg) { o.bg = rep + '66'; o.fg = '#111'; }
        else { o.deco.push('overline'); o.decoColor = rep; o.covered.push('repeat'); }
    }
    if (dotReg) {
        o.titles.push('dot-plot stretch');
        if (!o.bg) { o.bg = DOT_REGION_TINT + '55'; o.fg = '#111'; }
        else {   // a repeat on the same residue keeps its overline colour (the line is drawn once)
            if (!o.deco.includes('overline')) o.deco.push('overline');
            if (!o.decoColor) o.decoColor = DOT_REGION_TINT;
            o.covered.push('dot');
        }
    }
    if (snp) {
        o.titles.push(snp.title);
        if (!o.bg) { o.bg = snp.bg; o.fg = snp.fg; if (snp.bold) o.bold = true; }
        else { o.fg = snp.color; o.bold = true; o.covered.push('snp'); }
    }
    if (o.deco.length && ctx.trim && (pos <= ctx.trim.leftTrimEnd || pos >= ctx.trim.rightTrimStart)) o.deco.push('line-through');
    // Same marks -> same look: the class list is computed once per combination
    o.key = (hit ? hit.className : '') + '|' + (tsd ? state.tsdMarkStyle + _tsdColour() : '') + (tsdPrev ? 'P' + _tsdColour() : '') + (dotReg ? 'D' + DOT_REGION_TINT : '') + '|' + (rep || '')
        + '|' + (snp ? snp.bg + snp.fg + snp.color : '') + '|' + o.deco.join(',');
    return o;
}

// The span's whole inline style: the overlay owns it (it was shared by three painters)
function overlayCss(o) {
    if (!o) return '';
    let s = '';
    if (o.bg) s += `background-color:${o.bg} !important;`;
    if (o.fg) s += `color:${o.fg} !important;`;
    if (o.bold) s += 'font-weight:700 !important;';
    if (o.italic) s += 'font-style:italic !important;';
    // A second font has other glyph widths: lock the box to the grid's residue width
    if (o.sans) s += `font-family:${_OVERLAY_SANS} !important;width:var(--res-w) !important;text-align:center;`
        + 'padding-top:calc(2 * var(--text-dy-sans, var(--text-dy, 0px))) !important;line-height:calc(var(--row-h, 15px) - 2 * var(--text-dy-sans, var(--text-dy, 0px))) !important;';   // its own baseline correction (see _syncResidueWidthVar)
    if (o.deco.length) s += `text-decoration-line:${o.deco.join(' ')} !important;` + (o.decoColor ? `text-decoration-color:${o.decoColor} !important;` : '');
    return s;
}

// Each distinct look becomes one generated CSS class, created the first time it is needed.
// Marked residues share a handful of looks, so this is a few rules instead of a style
// attribute the browser has to parse on every marked residue (that doubled redraw time).
// The selector outranks the colour-scheme rules (#alignmentContainer.color-scheme-x ...),
// the shading classes and the trim tint, which all use !important too.
const _ovClassByDecl = new Map();
let _ovSheet = null, _ovClassSeq = 0;
function _overlayStyleClass(o) {
    const decl = overlayCss(o);
    if (!decl) return '';
    let cls = _ovClassByDecl.get(decl);
    if (!cls) {
        cls = 'ov-' + (_ovClassSeq++);   // a counter, not the table size: looks can be forgotten (below)
        if (!_ovSheet) {
            const st = document.createElement('style');
            st.id = 'residueOverlayStyles';
            document.head.appendChild(st);
            _ovSheet = st.sheet;
        }
        _ovSheet.insertRule(`#alignmentContainer .seq-line .seq-data > span.${cls}.${cls} { ${decl} }`, _ovSheet.cssRules.length);
        _ovClassByDecl.set(decl, cls);
    }
    return cls;
}

// Drops every generated look that uses the colour `hex` (rule and table entry), for a colour that is about to stop being
// used: the dot plot's tint changes continuously while a colour picker is dragged and each intermediate colour used to
// leave a rule and a table entry behind for good. The caller repaints the overlays afterwards (a span that still names a
// dropped class is given the new one then).
function _overlayForgetColour(hex) {
    if (!hex) return;
    const h = String(hex).toLowerCase(), gone = new Set();
    for (const [decl, cls] of _ovClassByDecl) if (decl.toLowerCase().includes(h)) { gone.add(cls); _ovClassByDecl.delete(decl); }
    if (_ovSheet && gone.size) {
        for (let i = _ovSheet.cssRules.length - 1; i >= 0; i--) {
            const m = /\.(ov-\d+)\./.exec(_ovSheet.cssRules[i].selectorText || '');
            if (m && gone.has(m[1])) _ovSheet.deleteRule(i);
        }
    }
    if (gone.size) _ovClassListMemo.clear();
}

function _overlayClassesOn(span) {

function _overlayClassListNow(o) {
    let c = ' ' + _overlayStyleClass(o);
    if (o.hit) c += ' ' + o.hit.className;
    if (o.tsd) c += ' tsd-mark' + (state.tsdMarkStyle === 'bold' ? ' tsd-mark-bold' : state.tsdMarkStyle === 'color' ? ' tsd-mark-color' : '');

// Attributes that go with the overlay (not classes: the caller sets className)
function _applyOverlayAttrs(span, o) {
    const title = o && o.titles.length ? o.titles.join(' | ') : '';
    if (title) { if (span.title !== title) span.title = title; }
    else if (span.hasAttribute('title')) span.removeAttribute('title');
    if (o?.hit) span.dataset.searchHit = o.hit.className; else if (span.dataset.searchHit !== undefined) delete span.dataset.searchHit;
    if (o?.rep) span.dataset.repeatHl = '1'; else if (span.dataset.repeatHl) delete span.dataset.repeatHl;
}

// Class part only, for passes that do not recompute the base class
function _syncOverlayClasses(span, o) {
    const want = _overlayClassList(o);
    const have = _overlayClassesOn(span);
    if (want === have.join(' ')) return;
    if (have.length) span.classList.remove(...have);
    if (want) span.classList.add(...want.split(' '));
}

// Bring every rendered residue span in line with the overlay (after a search, a repeat or
// group change). Rows with nothing to show and nothing shown are skipped.
function syncResidueOverlays() {
    if (!alignmentContainer) return;
    if (isCanvasMode()) { _canvasState.scheduleDraw?.(); return; }
    if (document.getElementById('modeReads')?.checked) return;
    alignmentContainer.querySelectorAll('.seq-line[data-seq-index]:not(.consensus-line)').forEach(line => {
        const row = parseInt(line.dataset.seqIndex, 10);
        if (!(row >= 0) || row >= state.seqs.length) return;
        const data = line.querySelector('.seq-data');
        if (!data) return;
        const ctx = _overlayRowCtx(row);
        if (!ctx.any && !data.querySelector('[class*=" ov-"]')) return;
        const seq = state.seqs[row].seq;
        for (const span of data.children) {
            const p = span.dataset.pos;
            if (p === undefined) continue;
            const o = ctx.any ? residueOverlay(ctx, +p, seq[+p] || '-') : null;
            if (!o && !span.className.includes(' ov-')) continue;
            _syncOverlayClasses(span, o);
            _applyOverlayAttrs(span, o);
        }
    });
}

// --res-w: the pitch of the residue grid, i.e. the width of one residue span (spans have no fixed
// width: a Courier glyph sets it). It is not the glyph's advance: a span is an inline-block, and
// the browser rounds its width up to a layout unit (Courier New 13px: a 7.8013px glyph makes a
// 7.8125px cell). Everything laid out in columns but not built of spans uses it: the ruler (one
// run of text, which advanced 7.8013px per column and drifted ~5 columns left by column 3700),
// the annotation track, rows waiting to be drawn, and the second font's residues.
// Measured with a probe outside the alignment once per font size, before the alignment is laid
// out at that size: measuring a residue would force a layout of the whole alignment, and
// resetting the variable restyles every residue.
// --row-h: the height of a row = the font size + 2 px, as a whole number of device pixels. With rows of
// exactly the font size (line-height 1.0) a white letter's crossbar lay flush against a white cell above
// it at browser zoom 90 % (owner, 2026-10-23); the 2 px keep box colour above and below the ink, and the
// snapping keeps every row edge on a device pixel (no blended seams at 90 %, 110 %, 125 %...).
let _resWidthFor = null;
function _rowHeightPx(fontPx) {
    const dpr = window.devicePixelRatio || 1;
    return Math.round((fontPx + 2) * dpr) / dpr;
}
// The font's metrics at a size: ascent and descent of the font box (CSS px), and the height of a capital
// as drawn (DEVICE px): the hinted glyph is taller than the font's design value at small sizes (13 px
// at browser zoom 90 %: 9 device rows, not 6.7), so it is read from a drawn H.
let _fontMetricsCache = null;
function _fontMetrics(fontPx) {
    const dpr = window.devicePixelRatio || 1;
    if (_fontMetricsCache && _fontMetricsCache.fontPx === fontPx && _fontMetricsCache.dpr === dpr) return _fontMetricsCache;
    const cv = document.createElement('canvas');
    const side = Math.ceil(fontPx * 2.5 * dpr);
    cv.width = side; cv.height = side;
    const c = cv.getContext('2d');
    c.setTransform(dpr, 0, 0, dpr, 0, 0);
    c.font = fontPx + 'px "Courier New", monospace';
    const x = c.measureText('X');
    const baseDev = Math.round(fontPx * 1.6 * dpr);           // an integer device row, as the rows use
    c.fillStyle = '#000'; c.textBaseline = 'alphabetic'; c.fillText('H', 2, baseDev / dpr);
    const d = c.getImageData(0, 0, side, side).data;
    let top = -1;
    for (let y = 0; y < side && top < 0; y++) for (let i = y * side * 4 + 3; i < (y + 1) * side * 4; i += 4) if (d[i] > 127) { top = y; break; }
    // the ascent the DOM lays text out with: the font's ascent at the device size, as a whole device pixel
    c.setTransform(1, 0, 0, 1, 0, 0);
    c.font = (fontPx * dpr) + 'px "Courier New", monospace';
    const xd = c.measureText('X');
    _fontMetricsCache = { fontPx, dpr,
        ascent: x.fontBoundingBoxAscent != null ? x.fontBoundingBoxAscent : Math.ceil(fontPx * 0.8),
        descent: x.fontBoundingBoxDescent != null ? x.fontBoundingBoxDescent : Math.ceil(fontPx * 0.15),
        ascentDev: Math.round(xd.fontBoundingBoxAscent != null ? xd.fontBoundingBoxAscent : fontPx * dpr * 0.8),
        capDev: top >= 0 ? baseDev - top : Math.round(fontPx * 0.57 * dpr) };
    return _fontMetricsCache;
}
// The device row (from the row's top) the letters' baseline goes on in a row of rowH CSS px: the capitals
// centred in the row. The browser's own placement centres the font box, which has more room under the
// baseline (descenders) than over the capitals, so the letters sat high: at browser zoom 90 % no device
// row of box colour was left over a white letter's top (owner, 2026-10-23). The same row serves the DOM
// (--text-dy) and Canvas (_canvasTextBaseline), so a view switch does not move the letters in their cells.
function _textBaselineDev(fontPx, rowH) {
    const dpr = window.devicePixelRatio || 1, m = _fontMetrics(fontPx);
    const rowDev = Math.round(rowH * dpr);
    // rows of box colour above the capitals = B - capDev, below = rowDev - 1 - B: equal, or one more below
    // (measured 2026-10-23, 13 px letters: 2/2 at 90 %, 3/4 at 100 %, 4/4 at 110 %, 4/5 at 125 %, 5/5 at 150 %)
    return Math.round((rowDev - 1 + m.capDev) / 2);
}
function _textBaselinePx(fontPx, rowH) {      // Canvas: on the device row (crisp); a hair past it, as 14/1.25*1.25 = 13.99...
    return (_textBaselineDev(fontPx, rowH) + 0.02) / (window.devicePixelRatio || 1);
}
// The second font's ascent as the DOM lays it out (a whole device pixel), from a probe canvas at the device size
function _sansAscentDev(fontPx) {
    const dpr = window.devicePixelRatio || 1;
    const c = document.createElement('canvas').getContext('2d');
    c.font = (fontPx * dpr) + 'px ' + _OVERLAY_SANS;
    const m = c.measureText('X');
    return Math.round(m.fontBoundingBoxAscent != null ? m.fontBoundingBoxAscent : fontPx * dpr * 0.8);
}
function _syncResidueWidthVar(fontSize) {
    if (!alignmentContainer) return;
    const inline = fontSize || alignmentContainer.style.fontSize;
    const key = (inline || 'default') + '@' + (window.devicePixelRatio || 1);
    if (_resWidthFor !== null && _resWidthFor === key) return;
    _resWidthFor = key;
    const size = inline || getComputedStyle(alignmentContainer).fontSize;
    const fontPx = parseFloat(size) || 13, rowH = _rowHeightPx(fontPx), fm = _fontMetrics(fontPx);
    alignmentContainer.style.setProperty('--row-h', rowH + 'px');
    const dprNow = window.devicePixelRatio || 1;
    // two device pixels, not one: a cell's painted box does not end where its layout box does (measured by colouring
    // rows: up to about a device pixel short), so with one pixel of overlap the white page showed through part of the
    // boundary row (measured 2026-11-09, software drawing as on Linux CI or Windows --disable-gpu: 240 at 110 %,
    // 327 at 125 %, out of 765; 0 with two pixels, GPU and software alike)
    alignmentContainer.style.setProperty('--row-overlap', (Math.abs(dprNow - Math.round(dprNow)) < 1e-6 ? 0 : (2 / dprNow).toFixed(4)) + 'px');
    // --text-dy moves the DOM's baseline down onto the device row _textBaselineDev. The browser puts the
    // text's content area at a half-leading it rounds to whole CSS px, then the glyphs at the floor of
    // (row top + content top + rounded ascent) in device px, and at a fractional browser zoom the row
    // top itself is fractional. So the candidate paddings are tried on a residue span laid out off
    // screen, its content top read from a Range, and the one whose baseline comes nearest to a quarter
    // of a device pixel past the wanted row is kept (measured 2026-10-23: the same rows as Canvas at
    // 100 %, 110 %, 200 %, one row off at some other zooms, where a DOM row top is itself fractional).
    // The second font (TSD marks, _OVERLAY_SANS: Arial with Courier's vertical metrics) rounds its own way,
    // so it gets its own value, --text-dy-sans, measured with the same probe.
    const dpr = window.devicePixelRatio || 1, B = _textBaselineDev(fontPx, rowH);
    const findDy = (family, ascentDev) => {
        const row = document.createElement('div');
        row.style.cssText = `position:absolute;left:-9999px;top:0;visibility:hidden;font:${fontPx}px ${family};line-height:${rowH}px;height:${rowH}px;white-space:pre;margin:0;padding:0;border:0`;
        const sp = document.createElement('span');
        sp.textContent = 'T';
        sp.style.cssText = 'display:inline-block;vertical-align:top;margin:0;border:0;padding:0';
        row.appendChild(sp); document.body.appendChild(row);
        let dy = 0, bestErr = Infinity;
        for (let cand = 0; cand <= 4; cand += 0.25) {
            sp.style.paddingTop = (2 * cand) + 'px';
            sp.style.lineHeight = (rowH - 2 * cand) + 'px';
            const rg = document.createRange(); rg.selectNodeContents(sp);
            const contentTop = rg.getBoundingClientRect().top - sp.getBoundingClientRect().top;
            const err = Math.abs(contentTop * dpr + ascentDev - (B + 0.25));
            if (err < bestErr - 1e-6) { bestErr = err; dy = cand; }
        }
        row.remove();
        return dy;
    };
    alignmentContainer.style.setProperty('--text-dy', findDy('"Courier New", monospace', fm.ascentDev) + 'px');
    alignmentContainer.style.setProperty('--text-dy-sans', findDy(_OVERLAY_SANS, _sansAscentDev(fontPx)) + 'px');
    const probe = document.createElement('span');
    probe.textContent = 'M';
    probe.style.cssText = `position:absolute;left:-9999px;top:0;visibility:hidden;display:inline-block;white-space:pre;`
        + `font:700 ${size}/1 "Courier New", monospace;padding:0;margin:0;border:0;letter-spacing:normal`;
    document.body.appendChild(probe);
    const w = probe.getBoundingClientRect().width;
    probe.remove();
    if (w > 0) alignmentContainer.style.setProperty('--res-w', w + 'px');
}
