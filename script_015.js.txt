    const AA_PAD_T = hasCodon ? dpx(1) : 0;
    const aaRowH = hasCodon ? AA_PAD_T + AA_BOX_H + dpx(3) : 0;
    const aaOnly = hasCodon && _aaOnlyActive();
    // Pitch of a row with k translation rows. Block layout: codon analysis by gene gives each band only
    // the lanes of the genes in its columns (as the DOM Block gives each block), so rowPitch is set per
    // band while a band is drawn (setBand); basePitch, the strip's, is what the hit tests see
    // (_canvasLocal maps a band's rows onto it).
    const pitchOf = k => (aaOnly && k > 0 ? 0 : CHAR_H) + k * aaRowH;
    const basePitch = pitchOf(aaRowCount);
    let rowPitch = basePitch;
    _canvasState.rowPitch = basePitch;
    _canvasState.aaRowH = aaRowH;
    _canvasState.aaRows = aaRowCount;
    _canvasState.aaOnly = aaOnly;
    _canvasState.charH = CHAR_H;
    const aaPal = hasCodon ? _canvasAaPalette() : null;
    const aaFontStr = 'bold ' + fontStr;                                   // .aa-data > span.aa-c
    const aaTagFont = 'italic ' + (Math.round(fontSizePx * 0.75 * 100) / 100) + 'px "Courier New", monospace';   // .aa-tag
    const CSS1 = Math.max(1 / dpr, Math.round(dpr) / dpr);                   // one CSS pixel on device pixels
    // Annotation track above the ruler: its height shifts the ruler and every
    // row down, and the hit tests read it from the metrics.
    const TRACK_H = _annotTrackHeightPx(CHAR_H);
    m.trackH = TRACK_H;
    // Consensus row: letters CHAR_H high plus the DOM row's 1 px rule and 4 px margin
    // (.consensus-top / .consensus-bottom). At the top it is part of the header, so hit
    // tests and scroll anchors (m.headerH) see the rows start below it.
    const CONS_GAP = Math.ceil(5 * dpr) / dpr;             // the DOM row's 1 px rule + 4 px margin, on device pixels
    const HAIR = 1 / dpr;                                   // one device pixel
    const CONS_H = consInfo ? CHAR_H + CONS_GAP : 0;
    const consTop = !!consInfo && consInfo.position !== 'bottom';
    _canvasState.consensus = consInfo ? { len, values: consInfo.values, position: consInfo.position,
        threshold: consInfo.threshold, caseCache: null } : null;
    m.headerH = CHAR_H + TRACK_H + (consTop ? CONS_H : 0);
    // Glyph cache: pre-rendered (char, bg-color, fg-color) -> off-screen canvas
    // Turns fillRect()+fillText() into a single drawImage() per cell after warm-up
    const _glyphCache = new Map();
    // Rounded end of a codon box (one cell wide), with its 1 px white seam, on an opaque white
    // cell: side 'l', 'r' or 'lr' (a one-column codon)
    const _boxEndCache = new Map();
    function _boxEnd(bg, side) {
        const k = bg + '|' + side;
        let g = _boxEndCache.get(k);
        if (!g) {
            g = document.createElement('canvas');
            g.width = Math.ceil(CHAR_W * dpr);
            g.height = Math.ceil(AA_BOX_H * dpr);
            const c2 = g.getContext('2d', { alpha: false });
            c2.setTransform(dpr, 0, 0, dpr, 0, 0);
            c2.fillStyle = '#fff'; c2.fillRect(0, 0, CHAR_W, AA_BOX_H);
            c2.fillStyle = bg;
            const l = side.includes('l'), r = side.includes('r');
            if (typeof c2.roundRect === 'function') { c2.beginPath(); c2.roundRect(0, 0, CHAR_W, AA_BOX_H, [l ? 3 : 0, r ? 3 : 0, r ? 3 : 0, l ? 3 : 0]); c2.fill(); }
            else c2.fillRect(0, 0, CHAR_W, AA_BOX_H);
            c2.fillStyle = '#fff';
            if (l) c2.fillRect(0, 0, CSS1, AA_BOX_H);
            if (r) c2.fillRect(CHAR_W - CSS1, 0, CSS1, AA_BOX_H);
            g._svg = { box: true, bg, side };   // what it holds, for the SVG recorder (_makeSvgRecorder)
            _boxEndCache.set(k, g);
        }
        return g;
    }
    // variant: '' or the overlay's font ('b' bold, 'i' italic, 's' second font), same cell size
    function _makeGlyph(ch, bg, fg, variant = '') {
        const k = ch + '| ' + (bg||' ') + '| ' + fg + '| ' + dpr + '| ' + variant;
        let g = _glyphCache.get(k);
        if (!g) {
            g = document.createElement('canvas');
            g.width = Math.ceil(CHAR_W * dpr);
            g.height = Math.ceil(CHAR_H * dpr);
            const c2 = g.getContext('2d', { alpha: false });
            c2.setTransform(dpr, 0, 0, dpr, 0, 0);
            const sans = variant.includes('s');
            c2.font = (variant.includes('i') ? 'italic ' : '') + (variant.includes('b') ? 'bold ' : '')
                + (sans ? fontSizePx + 'px ' + _OVERLAY_SANS : fontStr);
            c2.textBaseline = 'alphabetic';
            c2.fillStyle = '#fff'; c2.fillRect(0, 0, CHAR_W, CHAR_H);          // the row is white under every cell
            if (bg) { c2.fillStyle = bg; c2.fillRect(0, 0, CHAR_W, CHAR_H); }   // a translucent tint composes over it as in the DOM
            c2.fillStyle = fg;
            const base = _canvasTextBaseline(CHAR_H);                            // the font box centred in the row, as the DOM's half-leading does
            if (sans) { c2.textAlign = 'center'; c2.fillText(ch, CHAR_W / 2, base); }
            else c2.fillText(ch, 0, base);
            g._svg = { ch, bg, fg, font: c2.font, base, h: CHAR_H, sans };   // what it holds, for the SVG recorder (_makeSvgRecorder)
            _glyphCache.set(k, g);
        }
        return g;
    }
