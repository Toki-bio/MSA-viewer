function _sanitizeSnapshotFileBaseName(sourceTitle) {
    const base = String(sourceTitle || 'alignment_snapshot')
        .trim()
        .replace(/[^a-z0-9._-]+/gi, '_')
        .replace(/^_+|_+$/g, '')
        .toLowerCase();
    const stamp = new Date().toISOString().replace(/[:.]/g, '-');
    return `${base || 'alignment_snapshot'}_${stamp}`;
}

function downloadSnapshotJson(payload, fileBaseName) {
    const json = JSON.stringify(payload, null, 2);

// The snapshot link (?snapshot=, also in the HTML file) holds the whole alignment already: own-copy and file panes,
// each a whole alignment more, go only into the JSON file; mirrors (a place each) stay in the link
function _snapshotLinkPayload(payload) {
    if (!payload.panes || !Array.isArray(payload.panes.list)) return payload;
    const list = payload.panes.list.filter(p => p.mode === 'mirror');
    const out = { ...payload, panes: list.length ? { ...payload.panes, list } : undefined };
    out._panesLeftOut = payload.panes.list.length - list.length;
    return out;
}

async function openSnapshotFromPath(snapshotPath) {
    const normalizedPath = String(snapshotPath || '').trim();
    if (!normalizedPath) {
        showMessage('Select or enter a snapshot path first.', 3000);
        return;
    }
    try {
        showMessage('Loading snapshot...', 0);
        const resp = await fetch(normalizedPath, { signal: AbortSignal.timeout(10000) });
        if (!resp.ok) throw new Error(`HTTP ${resp.status}: ${resp.statusText}`);
        const payload = await resp.json();
        await _loadSnapshotPayload(payload);
        const pathInput = el('snapshotPathInput');
        if (pathInput) pathInput.value = normalizedPath;
        if (state.seqs.length) showMessage(`Snapshot loaded from ${normalizedPath}`, 2500);   // not over a load error (audit M6)
    } catch (err) {
        console.error('Snapshot open failed:', err);
        showMessage(`Failed to open snapshot: ${err.message}`, 5000);
    }
}

async function refreshSnapshotList(silent = false) {

        // Local server mode: dynamic directory listing

        if (apiResp.ok) {
            const data = await apiResp.json();
            const files = Array.isArray(data.files) ? data.files : [];
            _renderSnapshotSelect(files);
            _showSnapshotFolderRows(true);
            if (!silent) showMessage(`Snapshot list refreshed (${files.length} file${files.length === 1 ? '' : 's'})`, 2000);
            return;
        }
        throw new Error(`HTTP ${apiResp.status}`);
    } catch (_) {
        // Static mode fallback: optional manifest at snapshots/index.json
        try {
            const manifestResp = await fetch('./snapshots/index.json', { signal: AbortSignal.timeout(4000) });
            if (!manifestResp.ok) throw new Error(`HTTP ${manifestResp.status}`);
            const data = await manifestResp.json();
            const entries = Array.isArray(data.files) ? data.files : [];
            const files = entries.map(v => {
                const path = String(v.path || v).trim();
                const parts = path.split('/');
                return { path, name: String(v.name || parts[parts.length - 1] || path) };
            }).filter(v => v.path);
            _renderSnapshotSelect(files);
            _showSnapshotFolderRows(files.length > 0);
            if (!silent) showMessage(`Snapshot list refreshed (${files.length} via manifest)`, 2500);
            return;
        } catch (err2) {
            _renderSnapshotSelect([]);
            _showSnapshotFolderRows(false);
            if (!silent) showMessage('No snapshot list API/manifest. Use manual snapshot path input.', 3500);
        }
    } finally {
        if (refreshBtn) refreshBtn.disabled = false;
    }
}

function _getContinuousSelectedColumnRange() {
    if (state.selectedColumns.size === 0) return null;
    const cols = Array.from(state.selectedColumns).sort((a, b) => a - b);
    for (let i = 1; i < cols.length; i++) {
        if (cols[i] !== cols[i - 1] + 1) return null;
    }
    return { minCol: cols[0], maxCol: cols[cols.length - 1], count: cols.length };
}

function degapSelectedBlock(direction = 'left') {
    if (!state.seqs.length) {
        showMessage('No sequences loaded.', 2000);
        return;
    }

    const range = _getContinuousSelectedColumnRange();
    if (!range || range.count < 1) {
        showMessage('Select a continuous column block first (Ctrl+Alt+drag).', 3500);
        return;
    }

    const alignLeft = direction !== 'right';
    pushUndo(alignLeft ? 'degap-block-left' : 'degap-block-right');

    const blockWidth = range.maxCol - range.minCol + 1;
    state.seqs.forEach(seqObj => {
        const leftPart = seqObj.seq.slice(0, range.minCol);
        const blockPart = seqObj.seq.slice(range.minCol, range.maxCol + 1);
        const rightPart = seqObj.seq.slice(range.maxCol + 1);
        const residues = blockPart.replace(/[-.\s]/g, '');
        const compacted = alignLeft
            ? residues.padEnd(blockWidth, '-')
            : residues.padStart(blockWidth, '-');
        seqObj.seq = leftPart + compacted + rightPart;
        seqObj.gaplessPositions = calculateGaplessPositions(seqObj.seq);
    });

    // Remove gap-only columns generated within this selected block
    const colsToRemove = [];
    for (let col = range.maxCol; col >= range.minCol; col--) {
        const allGap = state.seqs.every(seqObj => {
            const ch = seqObj.seq[col];
            return ch === '-' || ch === '.';
        });
        if (allGap) colsToRemove.push(col);
    }

    if (colsToRemove.length > 0) {
        // Column-keyed state (residue selection, trim bounds, SNP letters, column repeats)
        // follows the removed columns, as with Delete columns; it used to stay on the old
        // column numbers, i.e. on other residues
        applyColumnEdit({ type: 'delete', cols: colsToRemove });
        state.seqs.forEach(seqObj => {
            const chars = seqObj.seq.split('');
            colsToRemove.forEach(col => {
                if (col >= 0 && col < chars.length) chars.splice(col, 1);
            });
            seqObj.seq = chars.join('');
            seqObj.gaplessPositions = calculateGaplessPositions(seqObj.seq);
        });
    }

    state.selectedColumns.clear();

    renderAlignment();
    showMessage(`Block degapped to ${alignLeft ? 'left' : 'right'}; removed ${colsToRemove.length} gap-only col(s).`, 2600);
}

function _escapeXmlText(value) {
    return String(value)
        .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, '') // not allowed in XML 1.0
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&apos;');
}

// ---- Export colours ----
// Exports built from the data rather than from the cells on screen (full SVG,
// RTF) take each cell's colours from the stylesheet itself: a hidden line in
// the alignment container is given the cell's class list once per distinct
// list, so an export always matches what the screen shows for that cell.
function _cssColourToHex(c) {
    const m = String(c || '').match(/rgba?\(\s*(\d+)[,\s]+(\d+)[,\s]+(\d+)(?:[,\s/]+([\d.]+))?/);
    if (!m) return null;
    if (m[4] !== undefined && parseFloat(m[4]) === 0) return null; // transparent
    return '#' + [m[1], m[2], m[3]].map(v => Number(v).toString(16).padStart(2, '0')).join('');
}

// Layout shared by the full SVG and the RTF export: blocks of columns (the
// Block-mode width, or the whole alignment in Full mode for the SVG), each
// with a position ruler, the consensus row where the screen shows it, and
// one line per sequence.
// Block (Canvas Block) wraps the exports at the band width on screen: the Block size set, or fewer columns
// when the window holds fewer (_canvasState.blockFitCols). Before batch 5 they wrapped at the width set,
// which Canvas Block can exceed on a narrow window. Every column is exported whatever the view hides
// (hidden columns, Variable sites only), as from the old DOM Block: then a band on screen holds that many
// SHOWN columns and an exported block that many alignment columns, so the blocks break at other columns
// than the bands on screen (GLM audit cb5 #1).
function _exportBlockWidth() {
    const slider = blockSizeRequested();
    const fit = isBlockLayout() && _canvasState.layout?.kind === 'block' ? _canvasState.blockFitCols : 0;
    return fit > 0 ? Math.min(slider, fit) : slider;
}

// Cells of one line between columns [start, end): runs of equal style
function _exportRuns(text, start, end, styleAt) {
    const runs = [];
    for (let pos = start; pos < end; pos++) {
        const st = styleAt(pos);
        const ch = text[pos] || '-';
        const last = runs[runs.length - 1];
        if (last && last.st === st) last.text += ch;
        else runs.push({ st, text: ch, start: pos });
    }
    return runs;
}

// ---- SVG of what the Canvas renderer draws (viewport SVG export, printing Block) ----
// Canvas and Canvas Block draw through a 2D context, so there are no cells to walk for an SVG (the
// viewport export refused in Canvas, and with Block drawn by the canvas it would have refused there
// too). For an SVG the renderer draws the frame again into this recorder (_canvasState.drawToSvg): it
// has the part of the CanvasRenderingContext2D API the renderer uses and writes each call as SVG.
// The SVG so holds what the canvas shows, from the same state and the same drawing code: names, ruler,
// consensus, residues with their colours and overlays, translation lanes, annotation track, length
// badges, selections. A residue cell is a cached glyph image (_makeGlyph); the renderer tags each glyph
// and codon-box end with what it holds (._svg), and the recorder writes the letter and its background.
// Runs of cells in one line with one look become one <rect> per background run and one <text> with an
// x per letter. Colours are composed over white as the opaque canvas composes them (a translucent tint,
// a dimmed column under Highlight diffs), so cells need no transparency.
// idPrefix keeps the clip-path ids of several SVGs in one page apart (printing).
function _makeSvgRecorder(dpr, { overlay = false, idPrefix = 'c' } = {}) {
    const mc = document.createElement('canvas').getContext('2d');   // colour parsing, text metrics
    const out = [];
    const clipDefs = new Map();   // clip rect key -> { id, r }
    const colourCache = new Map(), fontCache = new Map(), baseCache = new Map();
    const n2 = v => String(Math.round(v * 100) / 100);
    const hex = c => '#' + [c[0], c[1], c[2]].map(v => Math.max(0, Math.min(255, Math.round(v))).toString(16).padStart(2, '0')).join('');
    const WHITE = [255, 255, 255, 1];
    // a canvas colour -> [r, g, b, a]
    const rgba = c => {
        let v = colourCache.get(c);
        if (!v) {
            mc.fillStyle = '#000';
            mc.fillStyle = c;
            const s = String(mc.fillStyle);
            if (s[0] === '#') v = [parseInt(s.slice(1, 3), 16), parseInt(s.slice(3, 5), 16), parseInt(s.slice(5, 7), 16), 1];
            else { const mm = s.match(/[\d.]+/g) || [0, 0, 0]; v = [+mm[0], +mm[1], +mm[2], mm[3] === undefined ? 1 : +mm[3]]; }
            colourCache.set(c, v);
        }
        return v;
    };
    // top (with its alpha) over an opaque colour
    const over = (top, under) => [0, 1, 2].map(i => top[i] * top[3] + under[i] * (1 - top[3])).concat(1);
    // an opaque colour drawn at globalAlpha a over the white canvas
    const fade = (c, a) => a >= 1 ? c : over([c[0], c[1], c[2], a], WHITE);
    const fontAttrs = f => {
        let v = fontCache.get(f);
        if (v === undefined) {
            const mm = String(f).match(/^\s*((?:\S+\s+)*?)([\d.]+)px\s+(.+)$/);
            const pre = mm ? mm[1] : '', size = mm ? +mm[2] : 10, fam = mm ? mm[3].trim() : 'sans-serif';
            v = ` font-family="${_escapeXmlText(fam.replace(/"/g, '\''))}" font-size="${n2(size)}px"`
                + (/\bbold\b/.test(pre) ? ' font-weight="bold"' : '') + (/\bitalic\b/.test(pre) ? ' font-style="italic"' : '');
            fontCache.set(f, v);
        }
        return v;
    };
    // how far the alphabetic baseline (SVG's) lies below a canvas textBaseline position
    const baseOff = (font, tb) => {
        if (tb === 'alphabetic' || !tb) return 0;
        const k = font + '|' + tb;
        let v = baseCache.get(k);
        if (v === undefined) {
            mc.font = font; mc.textBaseline = tb;
            const tm = mc.measureText('M');
            const size = parseFloat((String(font).match(/([\d.]+)px/) || [0, 10])[1]);
            v = typeof tm.alphabeticBaseline === 'number' ? -tm.alphabeticBaseline
                : (tb === 'top' || tb === 'hanging' ? size * 0.76 : tb === 'middle' ? size * 0.26 : tb === 'bottom' ? -size * 0.24 : 0);
            mc.textBaseline = 'alphabetic';
            baseCache.set(k, v);
        }
        return v;
    };
    const rec = {
        _svgRecorder: true,
        fillStyle: '#000', strokeStyle: '#000', font: '10px sans-serif', textAlign: 'start', textBaseline: 'alphabetic',
        globalAlpha: 1, lineWidth: 1, imageSmoothingEnabled: true,
        skipImages: false,  // drawImage writes nothing (the name-column pass of a scrolled Canvas strip, drawToSvg)
        unknownImages: 0,  // images the recorder cannot write (none in the renderer's own drawing)
    };
    let m = [1, 1, 0, 0];   // x scale, y scale, x shift, y shift: user units -> CSS px
    let clip = null;        // [x0, y0, x1, y1] CSS px
    let dash = [];
    let path = [], cur = null;
    const stack = [];
    const X = x => x * m[0] + m[2], Y = y => y * m[1] + m[3];
    // '' when the box is inside the clip, null when it is outside, else a clip-path attribute
    const clipAttr = (x0, y0, x1, y1) => {
        if (!clip) return '';
        if (x1 <= clip[0] || x0 >= clip[2] || y1 <= clip[1] || y0 >= clip[3]) return null;
        if (x0 >= clip[0] - 0.01 && x1 <= clip[2] + 0.01 && y0 >= clip[1] - 0.01 && y1 <= clip[3] + 0.01) return '';
        const k = clip.map(n2).join(',');
        let d = clipDefs.get(k);
        if (!d) { d = { id: idPrefix + 'clip' + (clipDefs.size + 1), r: clip.slice() }; clipDefs.set(k, d); }
        return ` clip-path="url(#${d.id})"`;
    };
    const rectOut = (x0, y0, x1, y1, c) => {   // c: [r, g, b, a]; clipped by geometry
        if (clip) { x0 = Math.max(x0, clip[0]); y0 = Math.max(y0, clip[1]); x1 = Math.min(x1, clip[2]); y1 = Math.min(y1, clip[3]); }
        if (x1 - x0 < 0.005 || y1 - y0 < 0.005) return;
        const a = c[3];
        if (a <= 0) return;
        out.push(`<rect x="${n2(x0)}" y="${n2(y0)}" width="${n2(x1 - x0)}" height="${n2(y1 - y0)}" fill="${hex(c)}"${a < 1 ? ` fill-opacity="${n2(a)}"` : ''}/>`);
    };
    const paint = (style) => { const c = rgba(style); return [c[0], c[1], c[2], c[3] * rec.globalAlpha]; };

    // A run of glyph cells: one line, adjacent, one text colour and font
    let run = null;
    const flush = () => {
        if (!run) return;
        for (const s of run.bgs) out.push(`<rect x="${n2(s[0])}" y="${n2(run.y0)}" width="${n2(s[1] - s[0])}" height="${n2(run.y1 - run.y0)}" fill="${s[2]}"/>`);
        out.push(`<text x="${run.xs.map(n2).join(' ')}" y="${n2(run.base)}" fill="${run.fg}"${run.font}${run.sans ? ' text-anchor="middle"' : ''} xml:space="preserve">${_escapeXmlText(run.text)}</text>`);
        run = null;
    };
    const glyph = (g, x0, y0, x1, y1) => {
        const ca = clipAttr(x0, y0, x1, y1);
        if (ca === null) return;
        const a = rec.globalAlpha;
        const bgC = g.bg ? over(rgba(g.bg), WHITE) : WHITE;   // the glyph is white under its tint
        const bgHex = hex(fade(bgC, a));
        const fgHex = hex(fade(over(rgba(g.fg), bgC), a));
        const font = fontAttrs(g.font);
        const base = y0 + g.base * (y1 - y0) / g.h;
        const tx = g.sans ? (x0 + x1) / 2 : x0;
