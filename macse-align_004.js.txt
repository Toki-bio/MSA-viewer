                                    ps += (go | 0);

                                }
                            }
                        }
                    }
                }
                sc[R0 + col * 3 + 0] = best;
                cc = cPc << 2;
                code |= (cc | cDi) << 6;
                tb[off + col] = code;
            }
        }
        return R0;
    }
