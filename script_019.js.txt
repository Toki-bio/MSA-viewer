// Clustal (.aln) format parser. The first non-blank line is the program's
// header (CLUSTAL W/O/2.x, MUSCLE, MSAPROBS, PROBCONS, Kalign all write one).
// Conservation lines are indented, so any line starting with whitespace is
// skipped; an optional trailing residue count after the sequence is dropped.
function _isClustalInput(text) {
    const first = (text.match(/^[^\S\n]*\S.*$/m) || [''])[0].trim();
    return /^(CLUSTAL|MUSCLE|MSAPROBS|PROBCONS|Kalign)\b/i.test(first);
}

// Rows of an interleaved file (Clustal, Stockholm), joined over its blocks. A row is the n-th line with its name
// within a block: keyed by name alone, two sequences with the same name (names cut short by the writer) became
// one row of double length and a sequence was lost (audit 2026-11-10, H7).
function _interleavedRows() {
    const rows = new Map();      // name + "\u0000" + occurrence in the block -> { name, seq }
    let seenInBlock = new Map();
    return {
        blockEnd() { seenInBlock = new Map(); },
        add(name, piece) {
            const n = seenInBlock.get(name) || 0;
            seenInBlock.set(name, n + 1);
            const key = name + '\u0000' + n;
            if (!rows.has(key)) rows.set(key, { name, seq: '' });
            rows.get(key).seq += piece;
        },
        seqs() {
            return [...rows.values()].map(r => ({ header: r.name, fullHeader: r.name, seq: r.seq, gaplessPositions: calculateGaplessPositions(r.seq) }));
        },
        get size() { return rows.size; }
    };
}
function parseClustal(text) {
    const lines = text.split(/\r\n|\r|\n/);
    const rows = _interleavedRows();
    let started = false;
    for (const line of lines) {
        if (!started) {
            if (line.trim()) started = true;
            continue;
        }
        if (!line.trim()) { rows.blockEnd(); continue; }   // a blank line ends a block
        if (/^\s/.test(line)) { rows.blockEnd(); continue; }   // so does the conservation line under a block
        if (line.startsWith('//')) continue;
        const m = line.match(/^(\S+)\s+(\S+)(?:\s+\d+)?\s*$/);
        if (!m) continue;
        rows.add(m[1], m[2]);
    }
    if (rows.size === 0) return null;
    return rows.seqs();
}

// PHYLIP format parser. Strict PHYLIP gives names a fixed 10-character field
// (names may contain spaces and may touch the sequence); relaxed PHYLIP (RAxML,
// PhyML, IQ-TREE) separates a name of any length by whitespace. Either may be
// sequential (one sequence after another, possibly over several lines) or
// interleaved. Each combination is tried, and the first that gives every
// sequence exactly the declared length is used; relaxed is tried first, since
// a strict file only parses as relaxed by coincidence of its spacing.
function parsePhylip(text) {
    const lines = text.split(/\r\n|\r|\n/).filter(l => l.trim());
    if (lines.length < 2) return null;
    const hm = lines[0].trim().match(/^(\d+)\s+(\d+)/);
    if (!hm) return null;
    const nSeqs = parseInt(hm[1]), alignLen = parseInt(hm[2]);
    if (nSeqs < 1 || alignLen < 1) return null;
    const body = lines.slice(1);
    // Whitespace and position numbers some writers put at line ends are not residues
    let strip = s => s.replace(/[\s\d]/g, '');
    const splitName = {
        relaxed: l => { const m = l.trim().match(/^(\S+)\s+(.*)$/); return m ? [m[1], strip(m[2])] : null; },
        strict: l => l.length > 10 ? [l.substring(0, 10).trim(), strip(l.substring(10))] : null
    };
    function sequential(split) {
        const out = [];
        let i = 0;
        while (out.length < nSeqs && i < body.length) {
            const first = split(body[i++]);
            if (!first || !first[0]) return null;
            let seq = first[1];
            while (seq.length < alignLen && i < body.length) seq += strip(body[i++]);
            out.push({ name: first[0], seq });
        }
        return i === body.length ? out : null;
    }
    function interleaved(split) {
        if (body.length < nSeqs) return null;
        const out = [];
        for (let k = 0; k < nSeqs; k++) {
            const first = split(body[k]);
            if (!first || !first[0]) return null;
            out.push({ name: first[0], seq: first[1] });
        }
        for (let i = nSeqs; i < body.length; i++) {
            const e = out[(i - nSeqs) % nSeqs];
            // Some writers repeat the name on continuation lines; drop it
            const named = split(body[i]);
            e.seq += named && named[0] === e.name ? named[1] : strip(body[i]);
        }
        return out;
    }
    const attempts = [
        () => sequential(splitName.relaxed), () => interleaved(splitName.relaxed),
        () => sequential(splitName.strict), () => interleaved(splitName.strict)
    ];
    let entries = null;
    const results = attempts.map(a => a());
    const fits = got => got && got.length === nSeqs && got.every(e => e.seq.length === alignLen);
    entries = results.find(fits) || null;
    if (!entries) {
        // A matrix of states (binary 0/1, restriction sites): its digits are the data, not position
        // numbers. Removing them left empty rows and the file was refused (audit L15). Read again keeping
        // digits; taken only when every row is states and fits the header.
        const stripLetters = strip;
        strip = s => s.replace(/\s/g, '');
        const asStates = attempts.map(a => a()).find(got => fits(got) && got.every(e => /\d/.test(e.seq) && /^[0-9?.\-]+$/.test(e.seq)));
        if (asStates) return asStates.map(e => ({ header: e.name, fullHeader: e.name, seq: e.seq, gaplessPositions: calculateGaplessPositions(e.seq) }));
        strip = stripLetters;
    }
    // A number inside the 10 name columns is part of the name ("Clone 1", "Clone 12"), not a position
    // number: the relaxed reading dropped it and gave three rows named "Clone" (audit M2). Taken only
    // when the strict reading gives the same sequences and its names are the relaxed ones plus digits.
    const ri = results.indexOf(entries);
    if (entries && ri < 2 && fits(results[ri + 2])) {
        const st = results[ri + 2];
        let differs = false;
        const same = st.every((e, k) => {
            const r = entries[k];
            if (e.seq !== r.seq) return false;
            if (e.name === r.name) return true;
            differs = true;
            return e.name.startsWith(r.name) && /^\s+\d[\d\s]*$/.test(e.name.slice(r.name.length));
        });
        if (same && differs) entries = st;
    }
    if (!entries) {
        // Nothing fits the header's length. Prefer a reading that gives every
        // sequence the same length (relaxed names first), and say so: a stale
        // header must not silently move residues into the names.
        entries = results.find(got => got && got.length === nSeqs && got.every(e => e.seq.length === got[0].seq.length && e.seq.length > 0))
            || interleaved(splitName.strict);
        if (!entries || entries.length < nSeqs) {
            throw new Error(`PHYLIP header says ${nSeqs} sequences of ${alignLen} columns, but the file does not have ${nSeqs} sequences in either sequential or interleaved layout`);
        }
        const lens = Array.from(new Set(entries.map(e => e.seq.length)));
        _parseNotice = `PHYLIP header says ${alignLen} columns, but the sequences have ${lens.join('/')} - check the header line.`;
    }
    return entries.map(e => ({ header: e.name, fullHeader: e.name, seq: e.seq, gaplessPositions: calculateGaplessPositions(e.seq) }));
}
