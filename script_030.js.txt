                        const where = first ? `; first: ${first.f.name || 'CDS'} at column ${(first.cs + 1).toLocaleString()}` : '';
                        // a third overlapping CDS has no lane: say so instead of claiming every CDS is shown
                        const hid = byGene.hidden && byGene.hidden.length ? `. Not shown where two other CDS already overlap: ${byGene.hidden.slice(0, 3).join(', ')}${byGene.hidden.length > 3 ? ' and more' : ''}` : '';
                        showMessage(`Codon analysis: ${byGene.genes} CDS from the annotation track, each in its own frame and strand${where}${hid}`, hid ? 8000 : 5000);
                    }
                    return;
                }
                if (frameSel === 'genes') showMessage('Frame "Genes" needs an annotation track with CDS features (Annotation menu); using Auto', 4000);
            }
            state._codonFrames = _computeMultiFrameCodonAnalysis(state.seqs, len);
            const validFrames = state._codonFrames?.frames?.filter(Boolean);
            if (state._codonFrames && validFrames && validFrames.length > 0) {
                let activeFrame;

                    activeFrame = state._codonFrames.bestFrame;
                } else if (frameSel === 'all') {
                    activeFrame = -1;
                } else {
                    activeFrame = parseInt(frameSel) || 0;
                }

                state._codonData = state._codonFrames.frames[displayFrame] || validFrames[0];

                        showMessage(`Codon analysis: auto-selected frame Pos ${bf + 1} (ATG start)`, 2500);   // as the Pos 1/2/3 buttons (said 'frame 0')
                    }
                }
            } else {
                state._codonFrames = null;
                state._codonData = null;
                state._codonActiveFrame = 0;
                document.body.classList.remove('codon-mode');
                showMessage('Codon analysis requires a nucleotide alignment', 4000);
            }
        } catch (codonErr) {

            state._codonFrames = null;
            state._codonData = null;
            state._codonActiveFrame = 0;

            showMessage('Codon analysis failed: ' + codonErr.message, 4000);
        }
    } else {
        state._codonData = null;
        state._codonFrames = null;
        state._codonActiveFrame = 0;
        document.body.classList.remove('codon-mode');
    }
}

const _FOLD_STYLE = { color: '#f2c14e', symbol: '\u2194' };
function _foldNormalize(ranges, len) {
    const r = (ranges || []).map(x => [Math.max(0, x[0] | 0), Math.min(len - 1, x[1] | 0)]).filter(x => x[0] <= x[1]).sort((a, b) => a[0] - b[0]);
    const out = [];
    for (const x of r) {
        const last = out[out.length - 1];
        if (last && x[0] <= last[1] + 1) last[1] = Math.max(last[1], x[1]);
        else out.push([x[0], x[1]]);
    }
    return out;
}
function _foldLen() { return state.seqs && state.seqs.length ? _maxOf(state.seqs, q => q.seq.length) : 0; }
// Full / Block (A2): a hidden column's spans carry .fold-hidden (display: none) in every row, the consensus row and
// the ruler, and one marker cell (.col-fold) stands before the first column shown after each run (at len for a run
// that ends the alignment), so the ruler stays over its columns. Not in the windowed view (it builds only the
// columns near the viewport, by column number): the commands go to Canvas there. null when nothing is folded here.
function _foldDomInfo() {
    const f = state.hiddenCols;
    if (!f || !f.length || isCanvasMode() || state._needsWindowedDom) return null;
    if (state._foldDom && state._foldDom.f === f) return state._foldDom;
    const len = _foldLen();
    const hidden = new Uint8Array(len + 1), before = new Map();
    for (const r of f) {
        for (let c = r[0]; c <= r[1] && c < len; c++) hidden[c] = 1;
        before.set(Math.min(r[1] + 1, len), r);
    }
    state._foldDom = { f, len, hidden, before };
    return state._foldDom;
}

// The fold holding a column, or null
function _foldAt(col) {
    const f = state.hiddenCols;
    if (!f || !f.length) return null;
    let lo = 0, hi = f.length - 1;
    while (lo <= hi) {
        const mid = (lo + hi) >> 1;
        if (f[mid][1] < col) lo = mid + 1;
        else if (f[mid][0] > col) hi = mid - 1;
        else return f[mid];
    }
    return null;
}

// Columns with at least pct % gaps ('-' or '.') among all rows
async function hideGappyColumns() {
    const pct = Math.min(100, Math.max(1, parseFloat(el('hideGappyPct')?.value) || 100));
    const len = _foldLen(), seqs = state.seqs, n = seqs.length;
    if (!len) return;
    const need = Math.ceil(pct / 100 * n - 1e-9), hide = [];
    // gaps per column, row by row (a row shorter than the alignment counts as gaps at its end), in slices of
    // ~30 ms so a large alignment does not freeze the page
    const gaps = new Uint32Array(len);
    let t0 = performance.now(), told = false;
    for (let i = 0; i < n; i++) {
        const q = seqs[i].seq, m = Math.min(q.length, len);
        for (let c = 0; c < m; c++) { const k = q.charCodeAt(c); if (k === 45 || k === 46) gaps[c]++; }
        for (let c = m; c < len; c++) gaps[c]++;
        if (performance.now() - t0 > 30) {
            if (!told) { told = true; showMessage('Looking for gappy columns...', 0); }
            await new Promise(r => setTimeout(r, 0));
            if (state.seqs !== seqs) return;   // another file came in meanwhile
            t0 = performance.now();
        }
    }
    for (let c = 0; c < len; c++) if (gaps[c] >= need) hide.push(c);
    if (!hide.length) { showMessage(`No column has ${pct} % or more gaps.`, 3000); return; }
    if (!await _foldEnsureCanvas()) return;
    if (_foldApply([...(state.hiddenCols || []), ..._foldRunsOf(hide)]) === false) return;
    showMessage(`${hide.length} column${hide.length > 1 ? 's' : ''} with ${pct} % or more gaps hidden.`, 3000);
}

function _computeVarSites(len) {
    const highlightDiffs = document.getElementById('highlightDiffs')?.checked;
    const varSites = document.getElementById('varSitesOnly')?.checked;
    updateVarThresholdBounds();
    const varThresholdMode = document.getElementById('varThresholdMode')?.value || 'pct';
    const varThresholdRaw = parseInt(document.getElementById('varSitesThreshold')?.value) || 0;
    const nSeq = state.seqs.length;
    // At most nSeq - 1 sequences can differ from a column's consensus (the consensus is one of
    // them), so a threshold of nSeq (the old count maximum, or 100%) could never be met and
    // "Variable sites only" blanked the whole alignment
    const varThreshold = Math.min(Math.max(nSeq - 1, 1), varThresholdMode === 'count'
        ? varThresholdRaw
        : (varThresholdRaw === 0 ? 0 : Math.ceil((varThresholdRaw / 100) * nSeq)));
    const showBreakpoints = document.getElementById('varSitesBreakpoints')?.checked !== false;
    if ((highlightDiffs || varSites) && state.seqs.length > 1) {
        const diffCols = new Set();
        const spans = state.seqs.map(s => {
            const seq = s.seq;
            let first = -1, last = -1;
            for (let i = 0; i < seq.length; i++) {
                const c = seq[i];
                if (c !== '-' && c !== '.') { if (first === -1) first = i; last = i; }
            }
            return { first, last };
        });
        // An unknown residue (?, X, and N unless the alignment is protein, where N is asparagine) is not
        // a difference: a column of A and N used to be flagged as variable (audit M19)
        const nIsUnknown = !isProteinAlignment();
        for (let pos = 0; pos < len; pos++) {
            const counts = {};
            let covered = 0;
            for (let i = 0; i < state.seqs.length; i++) {
                const { first, last } = spans[i];
                if (first === -1 || pos < first || pos > last) continue;
                const base = (state.seqs[i].seq[pos] || '-').toUpperCase();
                if (base === '?' || base === 'X' || (base === 'N' && nIsUnknown)) continue;
                const ch = (base === '-' || base === '.') ? '-' : base;
                counts[ch] = (counts[ch] || 0) + 1;
                covered++;
            }
            const diffCount = covered > 0 ? covered - Math.max(...Object.values(counts)) : 0;
            if (varThreshold === 0 || diffCount >= varThreshold) diffCols.add(pos);
        }
        state._diffColumns = diffCols;
        if (varThreshold > 0 && diffCols.size === 0 && typeof showMessage === 'function') {
            showMessage(`No column has ${varThreshold} or more sequences differing from its consensus; lower the threshold to see columns.`, 4000);
        }
        if (varSites) {
            const hiddenRanges = [];
            let hiddenStart = -1;
            for (let pos = 0; pos < len; pos++) {
                if (!diffCols.has(pos)) {
                    if (hiddenStart === -1) hiddenStart = pos;
                } else {
                    if (hiddenStart !== -1) {
                        hiddenRanges.push({ start: hiddenStart, end: pos - 1, count: pos - hiddenStart });
                        hiddenStart = -1;
                    }
                }
            }
            if (hiddenStart !== -1) hiddenRanges.push({ start: hiddenStart, end: len - 1, count: len - hiddenStart });
            state._varSiteHiddenRanges = hiddenRanges;
            document.body.classList.toggle('hide-breakpoints', !showBreakpoints);
        } else {
            state._varSiteHiddenRanges = null;
        }
        if (highlightDiffs) {
            document.body.classList.add('highlight-diffs');
            document.body.classList.remove('var-sites-only');
        }
        if (varSites) {
            document.body.classList.add('var-sites-only');
            document.body.classList.remove('highlight-diffs');
        }
    } else {
        state._diffColumns = null;
        state._varSiteHiddenRanges = null;
        document.body.classList.remove('highlight-diffs');
        document.body.classList.remove('var-sites-only');
        document.body.classList.remove('hide-breakpoints');
    }

    const _varSitesActive = document.body.classList.contains('var-sites-only');
    const _showBrk = _varSitesActive && !document.body.classList.contains('hide-breakpoints');
    state._brkBeforePos = new Set();
    state._brkInfo = {};
    if (_showBrk && state._varSiteHiddenRanges && state._diffColumns) {
        for (const r of state._varSiteHiddenRanges) {
            const afterPos = r.end + 1;
            if (state._diffColumns.has(afterPos)) {
                state._brkBeforePos.add(afterPos);
                state._brkInfo[afterPos] = r;
            }
        }
    }
}
