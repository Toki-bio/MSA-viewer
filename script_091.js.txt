function _finishRepeatJob(job) {
    if (_repeatJob !== job) return;
    _repeatJob = null;
    _setRepeatRunning(false);
    _repeatJobStats = { slices: job.slices, maxSliceMs: Math.round(job.maxSliceMs * 10) / 10, ms: Math.round(performance.now() - job.t0) };
    const { resultsEl, mode, minLen, maxDiv, unit, targets, scope, allResults } = job;
    const notes = job.notes.slice();
    const searched = job.done + (job.partial ? 1 : 0);
    if (job.cancelled && job.M > 1) notes.unshift(`stopped by you after ${job.done} of ${job.M} sequences${job.partial ? ' (and part of the next)' : ''}`);
    else if (job.cancelled && !job.partial && !job.done) notes.unshift('stopped by you before the search began');
    _carryRepeatHighlights(mode, allResults);
    if (allResults.length === 0) {
        // forget the previous table, or Clear highlights brings it back
        _repeatResultsShown = null;
        _lastRepeatResults = [];
        _lastRepeatView = null;
        resultsEl.textContent = `No ${mode} repeats found (min ${mode === 'tandem' ? 'array length ' : ''}${minLen} ${unit}, max ${maxDiv}% divergence)${notes.length ? '; ' + notes.join('; ') : ''}.`;
        return;
    }
    _lastRepeatResults = allResults;
    const name = targets.length === 1 ? job.displayName : `${job.cancelled ? searched : targets.length} sequences`;
    // obj: the sequence a One-sequence table is about, so reopening on another sequence does not keep it
    _lastRepeatView = { mode, name, totalLen: job.totalLen, unit, note: notes.join('; '), obj: scope === 'single' && targets.length === 1 ? targets[0].obj : null,
        // B2: microsatellite groups (each { id, unit, n, longest, bp, arrays: [[col, endCol)], seqIndex, obj }) and the
        // pairs each shows when opened (20 at a time)
        groups: job.groups, groupedPairs: job.groupedPairs || 0, ungroupedTotal: job.ungroupedTotal || 0, expanded: new Map() };
    _renderRepeatResultsHTML(resultsEl, allResults, mode, name, job.totalLen, _lastRepeatView);
}

function runTsdAnalysis() {
    const resultsEl = document.getElementById('repeatResults');
    if (!resultsEl) return;
    const tsdMode = document.querySelector('input[name="tsdMode"]:checked')?.value || 'auto';
    const params = _readTsdParams();
    const aliLen = _maxOf(state.seqs, s => s.seq.length);
    // a search that stops with a message forgets the previous results: Mark, Copy table and the faint preview still
    // worked on the last run's TSDs under "Invalid pre-SINE region" (GLM db2 #9)
    const fail = (msg) => {
        _lastTsdResults = [];
        _repeatResultsShown = null;
        _syncTsdResultTools(true);
        _clearTsdPreview();
        resultsEl.textContent = msg;
    };

    if (tsdMode === 'manual') {
        if (params.upStart < 1 || params.upEnd < params.upStart) {
            fail('Invalid pre-SINE region: check start/end columns.');
            return;
        }
        if (params.downStart < 1 || params.downEnd < params.downStart) {
            fail('Invalid post-SINE region: check start/end columns.');
            return;
        }
        if (params.upEnd >= params.downStart) {
            fail('Pre-SINE and post-SINE regions must not overlap (leave room for the SINE body between them).');
            return;
        }
        if (params.downEnd > aliLen) {
            fail(`Post-SINE end column exceeds alignment length (${aliLen}).`);
            return;
        }
    }

    resultsEl.textContent = `Searching TSDs in ${state.seqs.length} sequence(s)...`;
    setTimeout(() => {
        try {
            const results = _findTSD(state.seqs, tsdMode, params);
            _lastTsdResults = results;
            _renderTsdResultsHTML(resultsEl, results, tsdMode, params);
            _setTsdPreview(results);
        } catch (e) {
            fail(`Error: ${e.message}`);
        }
    }, 30);
}

// Scroll the alignment to a row (if it is drawn) and a column, and flash the row name
// (bandCol: Block view only, when col is not given: the band to show the row in, e.g. the SINE body's
// first column for a TSD miss; without it the band now in the middle of the view)
function _showRowInAlignment(row, col, bandCol) {
    const container = document.getElementById('alignmentContainer');
    // Block view (Canvas Block): the band holding the column, the row's line in it (it set the strip's
    // offsetY = row * pitch, which is band 0's rows, and panned offsetX, which Block holds at 0)
    if (isBlockLayout() && _canvasState.layout) {
        if (Number.isInteger(col)) _foldRevealCol(col);
        const c = Number.isInteger(col) ? col : (Number.isInteger(bandCol) ? bandCol : null);
        if (_canvasBlockReveal(row, c)) return;
    }
    if (document.getElementById('modeCanvas')?.checked && _canvasState.canvas) {
        // Canvas: the row a third of the way down (as _geScrollToSeq), then the column
        const rowPitch = _canvasState.rowPitch || _canvasState.metrics?.charH || 16;
        const cv = _canvasState.canvas;
        const viewH = (parseInt(cv.style.height) || cv.clientHeight || container?.clientHeight || 240);
        const maxY = Math.max(0, (_canvasState.totalContentH || 0) - viewH);   // not past the content's end (GLM audit)
        _canvasState.offsetY = Math.min(maxY, Math.max(0, row * rowPitch - Math.min(viewH / 3, rowPitch * 4)));
        _canvasState.scheduleDraw?.();
        _canvasState.onOffsetChange?.();
        if (Number.isInteger(col)) _scrollToColumn(col);
        return;
    }
    // Block mode draws a row once per block: go to the block that holds the column
    const cell = Number.isInteger(col) ? container?.querySelector(`.seq-line[data-seq-index="${row}"] .seq-data > span[data-pos="${col}"]`) : null;
    if (cell) {
        cell.scrollIntoView({ block: 'center', inline: 'center' });
        const name = cell.closest('.seq-line')?.querySelector('.seq-name');
        if (name) { name.classList.remove('rf-flash'); void name.offsetWidth; name.classList.add('rf-flash'); }
        return;
    }
    const line = container?.querySelector(`.seq-line[data-seq-index="${row}"]`);
    if (line) {
        line.scrollIntoView({ block: 'center', inline: 'nearest' });
        const name = line.querySelector('.seq-name');
        if (name) { name.classList.remove('rf-flash'); void name.offsetWidth; name.classList.add('rf-flash'); }
    }
    if (Number.isInteger(col)) requestAnimationFrame(() => _scrollToColumn(col));
}

// The columns a result row highlights, "start-end[;start-end]" (end exclusive)
function _repeatResultSegs(r, mode) {
    if (mode === 'tandem') return `${r.start}-${r.end}`;
    const endA = r.alignEnd != null ? r.alignEnd : (r.posA + r.length);
    const endB = r.alignEndB != null ? r.alignEndB : (r.posB + r.length);
    return `${r.posA}-${endA};${r.posB}-${endB}`;
}
// A new search of a mode replaces that mode's table, and its highlights were keyed by place in the old table
// ("direct-3"): the new row 3 came up highlighted while the old repeat stayed painted, and clicking it removed the
// old paint. A highlighted repeat the new search finds again keeps its highlight under its new place; the others go.
function _carryRepeatHighlights(mode, results) {
    const hl = state.repeatHighlights;
    const old = [...hl.entries()].filter(([k]) => k.startsWith(mode + '-'));
    if (!old.length) return;
    const prev = _lastRepeatView?.mode === mode ? (_lastRepeatResults || []) : [];
    old.forEach(([k]) => hl.delete(k));
    old.forEach(([k, info]) => {
        const was = prev[parseInt(k.split('-').pop(), 10)];
        if (!was) return;
        const segs = _repeatResultSegs(was, mode);
        const j = results.findIndex(r => r.obj === was.obj && _repeatResultSegs(r, mode) === segs);
        if (j >= 0 && !hl.has(mode + '-' + j)) hl.set(mode + '-' + j, { ...info, color: _repeatColorPalette[j % _repeatColorPalette.length] });
    });
    _paintRepeatHighlight(null, false);
}

                `<td><button class="rf-x repeat-remove-btn" title="Remove this highlight" onclick="event.stopPropagation();_removeRepeatHighlight(this.closest('tr'))">×</button></td></tr>`;
            if (!open) continue;
            const list = members.get(g.id), shown = Math.min(list.length, view.expanded.get(g.id) || 20);
            for (let k = 0; k < shown; k++) pairRow(results[list[k]], list[k], 'rf-member');
            if (shown < list.length) html += `<tr class="rf-more" onclick="_moreRepeatGroup(${g.id})"><td></td><td colspan="${multi ? 7 : 6}">show ${Math.min(20, list.length - shown)} more of this group (${fmt(list.length - shown)} left)</td></tr>`;
        }
    }
    html += '</tbody></table>';
    el.innerHTML = html;
}
