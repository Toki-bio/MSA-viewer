// One pass over the rows, changing only those whose state differs. Was: remove the class
// from every selected row, then one document-wide querySelectorAll per selected row to add
// it back: 150 selected rows cost ~760 ms of script on a 316k-span alignment, plus a style
// recalc over every row whose class was removed and re-added.
function updateRowSelections() {
    const sel = state.selectedRows;
    // Only sequences can be selected rows. Anything else (the consensus line's -1, NaN) made
    // Delete remove the wrong sequence and Copy / Move to top throw or insert undefined.
    for (const v of sel) { const n = +v; if (!Number.isInteger(n) || n < 0) sel.delete(v); }
    const els = document.querySelectorAll('.seq-line[data-seq-index], .seq-name[data-seq-index]');
    for (let i = 0; i < els.length; i++) {
        const e = els[i];
        const raw = e.getAttribute('data-seq-index');
        const want = sel.has(+raw) || sel.has(raw);
        if (e.classList.contains('selected') !== want) e.classList.toggle('selected', want);
    }
    _updateSplitHint();
    scheduleSelectionsPanelRefresh();
    // Canvas draws selections on its overlay: without this, Ctrl+A said "All sequences selected." and showed nothing
    if (isCanvasMode()) _canvasState.scheduleOverlay?.();
}
// Column highlight: toggle .column-selected on the spans of the columns that changed since
// the last call, nothing else. Rows drawn later (render, windowed scroll) bake the class in
// from state.selectedColumns, so syncing a column to the state is always correct.
// History: first a classList change per span of every selected column (O(selected x rows)
// per click), then one <style> rule listing `.seq-data > span[data-pos="N"]` for every
// selected column. The rule made every change re-match that selector list against all
// spans: on 316k spans one column cost ~1.7 s of style recalc, 201 columns 2.6 s, and
// clearing 2 s. The delta costs one class change per span of the changed columns only.
let _appliedColumnSel = new Set();
// skipConsensus: leave the consensus row out (column selection; Highlight diffs marks it too)
function _forEachSpanInColumns(cols, callback, skipConsensus = false) {
    if (!cols.length) return;
    const datas = document.querySelectorAll(skipConsensus ? '.seq-line:not(.consensus-line) > .seq-data' : '.seq-data');
    for (let d = 0; d < datas.length; d++) {
        const kids = datas[d].children;
        if (!kids.length) continue;
        // Spans are laid out in position order; breakpoint markers are extra children,
        // so the span for pos sits at index (pos - first) or later.
        let first = null;
        let k0 = 0;
        for (; k0 < kids.length; k0++) {
            const r = kids[k0].getAttribute('data-pos');
            if (r !== null) { first = +r; break; }
        }
        if (first === null) continue;
        let last = first;
        for (let k = kids.length - 1; k >= k0; k--) {
            const r = kids[k].getAttribute('data-pos');
            if (r !== null) { last = +r; break; }
        }
        for (let c = 0; c < cols.length; c++) {
            const pos = cols[c];
            if (pos < first || pos > last) continue;
            for (let k = k0 + (pos - first); k < kids.length; k++) {
                const r = kids[k].getAttribute('data-pos');
                if (r === null) continue;
                const p = +r;
                if (p === pos) { callback(kids[k], pos); break; }
                if (p > pos) break;
            }
        }
    }
}
function updateColumnSelections() {
    scheduleSelectionsPanelRefresh();
    if (isCanvasMode()) _canvasState.scheduleOverlay?.();   // Canvas: the overlay holds the column selection too
    const sel = state.selectedColumns;
    const changed = [];
    _appliedColumnSel.forEach(pos => { if (!sel.has(pos)) changed.push(pos); });
    const next = new Set();
    sel.forEach(pos => {
        const n = Number(pos);
        if (!Number.isInteger(n)) return;
        next.add(n);
        if (!_appliedColumnSel.has(n)) changed.push(n);
    });
    _appliedColumnSel = next;
    // the edges of a run (.col-sel-l / .col-sel-r) change for the changed columns and their neighbours
    const touched = new Set();
    changed.forEach(p => { touched.add(p); touched.add(p - 1); touched.add(p + 1); });
    // the consensus row is not part of a column selection (it was highlighted with it)
    _forEachSpanInColumns([...touched].filter(p => p >= 0), (span, pos) => {
        const on = next.has(pos);
        span.classList.toggle('column-selected', on);
        span.classList.toggle('col-sel-l', on && !next.has(pos - 1));
        span.classList.toggle('col-sel-r', on && !next.has(pos + 1));
    }, true);
}

function detachNucSelectedHandlers(span) {
    if (span.dataset.nucSelectedBound !== '1') return;
    const handlers = span._nucSelectedHandlers;
    if (handlers) {
        span.removeEventListener('mouseover', handlers.mouseover);

function attachNucSelectedHandlers(span) {
    if (span.dataset.nucSelectedBound === '1') return;
    const mouseover = () => {
        if (_pointerDragInProgress()) return;
        showTooltipAt('Copy', span);
    };
    const mouseout = () => {

    };
    const click = (event) => {
        event.stopPropagation();
        event.preventDefault();
        copySelected();

    span.addEventListener('mouseover', mouseover);
    span.addEventListener('mouseout', mouseout);
    span.addEventListener('click', click);

// All residue spans of a row (every block line of it) by alignment position, in one query
function _rowSpansByPos(row) {
    const byPos = new Map();
    if (!alignmentContainer) return byPos;
    const sel = row === CONSENSUS_ROW_INDEX ? '.seq-line.consensus-line .seq-data > span[data-pos]' : `.seq-line[data-seq-index="${row}"] .seq-data > span[data-pos]`;
    alignmentContainer.querySelectorAll(sel).forEach(span => {
        const pos = parseInt(span.dataset.pos, 10);
        const list = byPos.get(pos);
        if (list) list.push(span); else byPos.set(pos, [span]);
    });
    return byPos;
}
let _nucRowsTouched = new Map();   // row -> positions whose run-end classes may change in the current refresh
function refreshNucleotideSelectionsImmediate() {
    const container = alignmentContainer;
    if (!container) return;
    const desiredSelections = state.selectedNucs;
    _nucRowsTouched = new Map();

    // Remove selections that are no longer desired
    for (const [row, posMap] of Array.from(state.domSelectedNucs.entries())) {
        const desiredSet = desiredSelections.get(row);
        if (!desiredSet || desiredSet.size === 0) {
            for (const [pos, span] of Array.from(posMap.entries())) {
                span.classList.remove('nuc-selected', 'nuc-sel-start', 'nuc-sel-end');
                detachNucSelectedHandlers(span);
                posMap.delete(pos);
            }
            state.domSelectedNucs.delete(row);
            continue;
        }
        let t = null;
        for (const [pos, span] of Array.from(posMap.entries())) {
            if (!desiredSet.has(pos)) {
                span.classList.remove('nuc-selected', 'nuc-sel-start', 'nuc-sel-end');
                detachNucSelectedHandlers(span);
                posMap.delete(pos);
                if (!t) { t = new Set(); _nucRowsTouched.set(row, t); }
                t.add(pos - 1); t.add(pos + 1);
            }
        }
        if (posMap.size === 0) {
            state.domSelectedNucs.delete(row);
        }
    }

    // Add new selections: one DOM query per row that gained cells (a query per cell made a 30,000-cell
    // rectangle cost ~90 ms per drag frame; GLM audit, v278)
    const touched = _nucRowsTouched;   // row -> Set of positions whose run-end classes may have changed
    desiredSelections.forEach((posSet, row) => {
        const have = state.domSelectedNucs.get(row);
        let missing = null;
        posSet.forEach(pos => { if (!have || !have.has(pos)) (missing || (missing = [])).push(pos); });
        if (!missing) return;
        let byPos = null;   // the row's spans, queried once only when the span cache has no entry
        let rowMap = have;
        if (!rowMap) { rowMap = new Map(); state.domSelectedNucs.set(row, rowMap); }
        let t = touched.get(row); if (!t) { t = new Set(); touched.set(row, t); }
        missing.forEach(pos => {
            const cached = getCachedSpan(row, pos);
            let spans;
            if (cached) spans = [cached];
            else { if (!byPos) byPos = _rowSpansByPos(row); spans = byPos.get(pos); }
            if (!spans) return;
            spans.forEach(span => { span.classList.add('nuc-selected'); attachNucSelectedHandlers(span); rowMap.set(pos, span); });
            t.add(pos - 1); t.add(pos); t.add(pos + 1);
        });
    });
    // The first and last cell of each run of selected cells carry the frame's sides (styles.css), judged by
    // the cells that exist on the page (a hidden column is not a neighbour: Canvas frames by shown slots the
    // same way); only cells whose neighbours changed are touched, so a 30,000-cell rectangle drag stays cheap
    // (a neighbour hidden by Variable sites only is on the page but not shown: it breaks the run; the layout
    // reads of pass 1 are kept apart from the class writes of pass 2 so the browser lays out once)
    const shownCache = new Map();
    const shown = span => { let v = shownCache.get(span); if (v === undefined) { v = span.offsetParent !== null; shownCache.set(span, v); } return v; };
    const writes = [];
    touched.forEach((set, row) => {
        const posMap = state.domSelectedNucs.get(row);
        if (!posMap) return;
        set.forEach(pos => {
            const span = posMap.get(pos);
            if (!span) return;
            const left = posMap.get(pos - 1), right = posMap.get(pos + 1);
            writes.push([span, !(left && shown(left)), !(right && shown(right))]);
        });
    });
    writes.forEach(([span, start, end]) => { span.classList.toggle('nuc-sel-start', start); span.classList.toggle('nuc-sel-end', end); });

    // Update pending indicator
    if (state.domPendingNuc) {
        const { span } = state.domPendingNuc;
        span.classList.remove('nuc-pending');
        detachNucPendingHandlers(span);
        state.domPendingNuc = null;
    }
    if (state.pendingNucStart) {
        const { row, pos } = state.pendingNucStart;
        forEachRowSpanAtPosition(row, pos, (span) => {
            if (!span.classList.contains('nuc-selected')) {
                span.classList.add('nuc-pending');
                attachNucPendingHandlers(span);
                state.domPendingNuc = { row, pos, span };
            }
        });
    }
}

function scheduleNucSelectionRefresh() {
    if (pendingNucDomUpdate) return;
    pendingNucDomUpdate = true;
    const raf = (typeof window !== 'undefined' && window.requestAnimationFrame)
        ? window.requestAnimationFrame.bind(window)
        : (cb) => setTimeout(cb, 16);
    raf(() => {
        pendingNucDomUpdate = false;
        refreshNucleotideSelectionsImmediate();
        scheduleSelectionsPanelRefresh();
    });
}

    if (undoDropdownBtn) {
        undoDropdownBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            showUndoRedoDropdown(undoDropdownBtn, state.deletedHistory, undoDelete, 'Undo');
        });
    }
    const redoDropdownBtn = el('redoDropdownBtn');
    if (redoDropdownBtn) {
        redoDropdownBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            showUndoRedoDropdown(redoDropdownBtn, state.redoHistory, redoAction, 'Redo');
        });
    }

    // Initialize clustering preset list
    updateClusteringPresetList();

    // Auto-create and load "optimal" preset as default if not already present
    setTimeout(() => {
        const presets = safeLocalGet('clusteringPresets');
        if (!presets['optimal']) {
            // Create optimal preset silently
            createOptimalPreset(true);
        } else {
            // Load existing optimal preset
            const presetSelect = el('clusteringPresetList');
            if (presetSelect) {
                presetSelect.value = 'optimal';
                loadClusteringPreset(true);
            }
        }
    }, 100);

    const searchInputEl = el('searchInput');
    if (searchInputEl) {
        searchInputEl.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') {
                e.preventDefault();
                searchMotif();
            }
        });
    }
    const reverseQueryButton = el('reverseComplementQueryButton');
    if (reverseQueryButton && searchInputEl) {
        reverseQueryButton.addEventListener('click', () => {
            const raw = searchInputEl.value || '';
            const motif = raw.trim().replace(/\s+/g, '').toUpperCase();
            if (!motif) { showMessage('Type a motif in the Search box first.', 2500); return; }
            searchInputEl.value = reverseComplement(motif);
        });
    }

    minimizeBar?.addEventListener('click', expandMenu);
    el('expandButton')?.addEventListener('click', expandMenu);

    // UI listener wiring
    attachUIListeners();

    // Apply default consensus thresholds
    setConsensusThresholdUI(DEFAULTS.consensusThreshold);
    setGroupConsensusThresholdUI(DEFAULTS.groupConsensusThreshold);
    setNameLengthUI(DEFAULTS.nameLength);

    // Set initial state for controls
    el('modeBlocks').checked = true;
    el('modeSingle').checked = false;
