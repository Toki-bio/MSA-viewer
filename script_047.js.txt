// One SVG document from recorded parts ({ body, defs }, in paint order)
function _svgDocument(parts, width, height, title, { xmlDecl = true } = {}) {
    const W = String(Math.max(1, Math.ceil(width))), H = String(Math.max(1, Math.ceil(height)));
    const defs = parts.map(p => p.defs).filter(Boolean).join('\n');
    return [
        xmlDecl ? '<?xml version="1.0" encoding="UTF-8"?>' : '',
        `<svg xmlns="http://www.w3.org/2000/svg" width="${W}" height="${H}" viewBox="0 0 ${W} ${H}">`,
        `<title>${_escapeXmlText(title)}</title>`,
        defs ? `<defs>\n${defs}\n</defs>` : '',
        `<rect x="0" y="0" width="${W}" height="${H}" fill="#ffffff"/>`,
        ...parts.map(p => p.body).filter(Boolean),
        '</svg>'
    ].filter(Boolean).join('\n');
}

function _exportAlignmentAsSvg(mode = 'viewport') {
    if (!alignmentContainer) {
        showMessage('Alignment container not found.', 3000);
        return;
    }
    if (!state.seqs || state.seqs.length === 0) {
        showMessage('Load an alignment first.', 3000);
        return;
    }
    // The whole alignment is drawn from the data: in windowed views most rows
    // and columns are not in the DOM, so walking the cells exported a fraction
    if (mode === 'full') { _exportFullAlignmentSvgFromData(); return; }
    // Canvas and Block (Canvas Block) draw on a canvas: the frame is drawn again into an SVG recorder
    // (_canvasState.drawToSvg). Before batch 5 this walked DOM spans only and refused there, which with
    // Block drawn by the canvas would have taken the viewport SVG away from Block.
    if (isCanvasMode() && _canvasState.drawToSvg && _canvasState.canvas && _canvasState.canvas.isConnected) {
        _exportCanvasViewportSvg();
        return;
    }
    // This walks the rendered residue spans, which Reads never creates. Without
    // this guard the export silently produced an empty file.
    if (!alignmentContainer.querySelector('.seq-data span[data-pos]')) {
        showMessage('SVG export needs Full, Block or Canvas view - the current view draws no per-residue cells.', 4500);
        return;
    }

    const containerRect = alignmentContainer.getBoundingClientRect();
    const fullMode = mode === 'full';
    const width = Math.max(1, Math.round(fullMode ? alignmentContainer.scrollWidth : alignmentContainer.clientWidth));
    const height = Math.max(1, Math.round(fullMode ? alignmentContainer.scrollHeight : alignmentContainer.clientHeight));
    const viewRight = containerRect.left + (fullMode ? alignmentContainer.scrollWidth : alignmentContainer.clientWidth);
    const viewBottom = containerRect.top + (fullMode ? alignmentContainer.scrollHeight : alignmentContainer.clientHeight);

    const spanNodes = Array.from(alignmentContainer.querySelectorAll('.seq-line span, .scale-ruler-line span'));
    if (spanNodes.length === 0) {
        showMessage('Nothing visible to export.', 2500);
        return;
    }

    const bgRects = [];
    const textNodes = [];

    spanNodes.forEach(span => {
        const rawText = span.textContent || '';
        if (!rawText) return;

        const r = span.getBoundingClientRect();
        if (r.right <= containerRect.left || r.left >= viewRight || r.bottom <= containerRect.top || r.top >= viewBottom) return;

        // true position (negative when the span is cut by the left or top edge): the text is drawn there and the
        // SVG viewport clips it, so the export shows the visible slice, not the span's start moved to the edge
        const x0 = r.left - containerRect.left + (fullMode ? alignmentContainer.scrollLeft : 0);
        const y0 = r.top - containerRect.top + (fullMode ? alignmentContainer.scrollTop : 0);
        const x = Math.max(0, x0), y = Math.max(0, y0);
        const w = Math.min(width, x0 + r.width) - x;
        const h = Math.min(height, y0 + r.height) - y;
        if (w <= 0 || h <= 0) return;

        const style = window.getComputedStyle(span);
        const bg = style.backgroundColor;
        const color = style.color || '#000';
        const fontSize = parseFloat(style.fontSize || '12') || 12;
        const fontFamily = style.fontFamily || 'monospace';
        const fontWeight = style.fontWeight || '400';
        const fontStyle = style.fontStyle || 'normal';
        // The baseline from where the letters really are: a residue span has a top padding of 2 * --text-dy and a
        // line box that much shorter, which moves its letters down by half the padding (--text-dy), and a bottom
        // padding (--row-overlap) that is painted under the row below and holds no text. Using the whole box put
        // the residues --text-dy above their names and above the screen where --text-dy is not 0 (1 px with
        // Liberation Mono on Linux, 2026-11-09).
        const padT = parseFloat(style.paddingTop) || 0, padB = parseFloat(style.paddingBottom) || 0;
        const boxH = r.height - padB;
        const textY = y0 + padT / 2 + boxH - Math.max(1, boxH * 0.18);

        if (!_isTransparentCssColor(bg)) {
            bgRects.push(`<rect x="${x.toFixed(2)}" y="${y.toFixed(2)}" width="${w.toFixed(2)}" height="${h.toFixed(2)}" fill="${_escapeXmlText(bg)}" />`);
        }

        const escaped = _escapeXmlText(rawText);
        textNodes.push(`<text x="${x0.toFixed(2)}" y="${textY.toFixed(2)}" fill="${_escapeXmlText(color)}" font-family="${_escapeXmlText(fontFamily)}" font-size="${fontSize.toFixed(2)}px" font-weight="${_escapeXmlText(fontWeight)}" font-style="${_escapeXmlText(fontStyle)}" xml:space="preserve">${escaped}</text>`);
    });

    if (textNodes.length === 0) {
        showMessage('Nothing visible to export.', 2500);
        return;
    }

    const title = _escapeXmlText(state.currentFilename || 'alignment');
    const label = fullMode ? 'full alignment export' : 'viewport export';
    const svg = [
        `<?xml version="1.0" encoding="UTF-8"?>`,
        `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">`,
        `<title>${title} ${label}</title>`,
        `<rect x="0" y="0" width="${width}" height="${height}" fill="#ffffff" />`,
        ...bgRects,
        ...textNodes,
        `</svg>`
    ].join('\n');

    const blob = new Blob([svg], { type: 'image/svg+xml;charset=utf-8' });
    const dlUrl = URL.createObjectURL(blob);
    const a = document.createElement('a');
    const safeBaseName = (state.currentFilename || 'alignment').replace(/[^a-z0-9._-]+/gi, '_').replace(/^_+|_+$/g, '');
    a.href = dlUrl;
    a.download = `${safeBaseName || 'alignment'}_${fullMode ? 'full' : 'viewport'}.svg`;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(dlUrl), 2000);
    showMessage(fullMode ? 'Full alignment exported as SVG.' : 'Visible viewport exported as SVG.', 2500);
}

// ---- Printing Block ----
// Block is drawn on a canvas that holds only the bands in view, so a print gave that one screen as a
// bitmap (measured 2026-10-31: 1 page, the viewport). The old DOM Block printed its blocks as page text,
// each cut at the page's right edge, and above ALIGN_WINDOWED_DOM_THRESHOLD only the rows and blocks in
// its window. Now, before printing, every band is drawn as SVG (_canvasState.drawToSvg) into #printBlock,
// which the print stylesheet shows in place of the page; it is scaled to the paper's width, so every
// column of a band is on paper. A band taller than a page is cut between rows into page-sized pieces.
// Above PRINT_BLOCK_MAX_CELLS the bands from the one in view on are printed, up to the limit, under a
// note naming them (the full-alignment SVG export has the rest).
const PRINT_BLOCK_MAX_CELLS = 400_000;
// A failed preparation is not retried within the same print: beforeprint and the print media query both call it,
// and the second call undid the "the print shows the screen only" fallback. Reset when the print ends.
let _printBlockFailed = false;
function _preparePrintBlock() {
    if (_printBlockFailed) return;
    _clearPrintBlock();
    try {
        _buildPrintBlock();
    } catch (e) {
        // nothing half-made goes to paper: the page prints as it is (the canvas), and the user is told. The whole
        // band build is covered, not only drawToSvg (an error in _svgDocument or the insert escaped unreported)
        _printBlockFailed = true;
        document.getElementById('printBlock')?.remove();
        document.body.classList.remove('print-block');
        console.error('Printing Block failed:', e);
        showMessage('Printing the blocks failed (' + (e && e.message ? e.message : e) + '): the print shows the screen only.', 6000);
    }
}
function _buildPrintBlock() {
    if (!isBlockLayout() || !_canvasState.drawToSvg || !_canvasState.svgBandRegions || !state.seqs?.length) return;
    const bands = _canvasState.svgBandRegions();
    if (!bands || !bands.length) return;
    const t0 = performance.now();
    const nSeqs = state.seqs.length;
    const m = _canvasState.metrics || {};
    const L = _canvasState.layout;
    // the bands to print: all, or from the band in view while the cells stay under the limit
    let b0 = 0, b1 = bands.length;
    const cellsOf = b => nSeqs * (bands[b].c1 - bands[b].c0);
    let total = 0;
    for (let b = 0; b < bands.length; b++) total += cellsOf(b);
    if (total > PRINT_BLOCK_MAX_CELLS) {
        b0 = L && L.bandAt ? L.bandAt(_canvasState.offsetY) : 0;
        let cells = 0;
        for (b1 = b0; b1 < bands.length && (b1 === b0 || cells + cellsOf(b1) <= PRINT_BLOCK_MAX_CELLS); b1++) cells += cellsOf(b1);
    }
    const holder = document.createElement('div');
    holder.id = 'printBlock';
    const title = state.currentFilename || 'alignment';
    if (b0 > 0 || b1 < bands.length) {
        const note = document.createElement('p');
        note.className = 'print-block-note';
        note.textContent = `${title}: bands ${b0 + 1}-${b1} of ${bands.length} (columns ${bands[b0].c0 + 1}-${bands[b1 - 1].c1} of ${bands[bands.length - 1].c1}). `
            + 'The whole alignment is too large to print from the page: use Full SVG in the Input menu.';
        holder.appendChild(note);
    }
    const headerH = m.headerH || 0;
    const parts = [];
    let k = 0;
    for (let b = b0; b < b1; b++) {
        const r = bands[b];
        // pieces of at most about a page (the SVG is scaled to the paper's width), cut between rows
        const maxH = Math.max(r.w * 1.3, headerH + 3 * (L?.bandPitch?.[b] || 15));
        const pitch = L?.bandPitch?.[b] || ((r.h - headerH) / Math.max(1, nSeqs));
        let y = 0;
        while (y < r.h - 0.5) {
            let h = r.h - y;
            if (h > maxH) {
                const rowsFrom = Math.max(0, Math.ceil((y - headerH) / pitch - 1e-6));
                const rowsTo = Math.floor((y + maxH - headerH) / pitch + 1e-6);
                h = Math.max(pitch, headerH + Math.max(rowsFrom + 1, rowsTo) * pitch - y);
            }
            // an exception here (or below) is caught by _preparePrintBlock, which tells the user (GLM audit cb5 #7)
            const s = _canvasState.drawToSvg({ ox: 0, oy: r.oy + y, w: r.w, h }, 'p' + (k++) + '_');
            parts.push(_svgDocument(s.parts, s.width, s.height, `${title} band ${b + 1}`, { xmlDecl: false }));
            y += h;
        }
    }
    holder.insertAdjacentHTML('beforeend', parts.join('\n'));
    document.body.appendChild(holder);
    document.body.classList.add('print-block');
    _printBlockWindowSize = window.innerWidth + 'x' + window.innerHeight;
    _canvasState.printMs = performance.now() - t0;
}
let _printBlockWindowSize = null;   // the window's size when the print began (_clearPrintBlock)
function _clearPrintBlock() {
    document.getElementById('printBlock')?.remove();
    if (!document.body.classList.contains('print-block')) return;
    document.body.classList.remove('print-block');
    // the viewer is shown again: lay it out for the window as it is now (a resize while printing was skipped,
    // and the controls' height, which places the viewer, was measured for the old window; GLM audit cb5 #4/#5)
    // A window resized meanwhile: the page's own resize listeners (the controls' rows among them) ran on a hidden
    // page, so they are run again.
    const size0 = _printBlockWindowSize;
    _printBlockWindowSize = null;
    requestAnimationFrame(() => {
        if (document.body.classList.contains('print-block')) return;
        if (!size0 || size0 !== window.innerWidth + 'x' + window.innerHeight) window.dispatchEvent(new Event('resize'));
        updateControlsOffset();
        if (isCanvasMode()) _canvasState.relayout?.();
    });
}
if (typeof window !== 'undefined') {
    window.addEventListener('beforeprint', _preparePrintBlock);
    window.addEventListener('afterprint', () => { _printBlockFailed = false; _clearPrintBlock(); });

}

function _exportFullAlignmentSvgFromData() {
