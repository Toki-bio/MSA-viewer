function _alignerRestoreResidues(result, recs) {
    if (typeof result !== 'string') return result;
    const byName = new Map();
    for (const r of recs) { const k = r.name; if (!byName.has(k)) byName.set(k, []); byName.get(k).push(r); }
    const out = parseMafftOutput(result);
    for (const o of out) {
        const q = byName.get(o.name) || byName.get(o.name.split(/\s+/)[0]);
        const src = q && q.shift();
        if (!src || /[-.]/.test(src.seq)) continue;   // not one of the rows sent, or sent with gaps: left as returned
        const n = o.seq.replace(/[-.]/g, '').length;
        if (n !== src.seq.length) {
            const shown = /^r\d+$/.test(o.name) && typeof _alignIdName === 'function' ? _alignIdName(o.name) : o.name;
            throw new Error(`the aligner returned ${n} residues for "${shown}", which has ${src.seq.length}; the alignment was not changed`);
        }
        let i = 0, t = '';
        for (let k = 0; k < o.seq.length; k++) { const c = o.seq[k]; t += (c === '-' || c === '.') ? c : src.seq[i++]; }
        o.seq = t;
    }
    return out.map(o => `>${o.name}\n${o.seq}\n`).join('');
}
async function _mafftAlignWithUiRun(fasta, extraArgs, label) {
    if (_isCodonAlignMode()) return _codonAlignWithUi(fasta, label && label.replace(/MAFFT/g, 'codon aligner'));
    const guarded = _alignerGuardInput(fasta);
    fasta = guarded.fasta;
    const stats = _mafftFastaStats(fasta);
    const confirmed = _confirmMafftJob(stats, extraArgs);
    if (!confirmed.ok) return null;
    extraArgs = confirmed.extraArgs;

    let cancelled = false;
    try {
        const result = await runWithProgress(
            label || 'Aligning with MAFFT...',
            async (updateBusy) => {
                updateBusy(`${stats.seqCount} seqs, ${stats.totalResidues.toLocaleString()} residues`);
                await yieldToPaint();
                return _runMafftInWorker(fasta, extraArgs);
            },
            '',
            () => { cancelled = true; _cancelActiveMafftWorker(); }
        );
        if (cancelled) {
            showMessage('MAFFT alignment cancelled.', 2500);
            return null;
        }
        return _alignerRestoreResidues(result, guarded.recs);
    } catch (err) {
        if (cancelled || String(err?.message || '').includes('cancelled')) return null;
        throw err;
    }
}

async function realignSelectedBlock(opts) {
    opts = opts || {};
    if (state.seqs.length === 0) {
        showMessage("No sequences loaded.", 2000);
        return;
    }
    if (typeof RealignRegion === 'undefined') {
        showMessage("Region realign module not loaded.", 3000);
        return;
    }
    // a block of rows x columns (a residue selection shaped as a rectangle): only those rows, within those columns
    const rect = !state.selectedColumns.size ? _nucSelectionRectangle() : null;
    if (rect) return _realignRectangle(rect, opts);
    const runs = RealignRegion.columnRuns(state.selectedColumns);
    if (!runs.length) {
        showMessage("Select at least one span of 2+ columns (or drag a block of rows x columns) to realign.", 3000);
        return;
    }

    const extra = getMafftExtraArgs();
    if (extra.seqType === 'codon' && !extra.reorderOnly) {
        // A column span cuts each row at a different codon position, so a codon-aware realignment of the block
        // would be out of frame. Whole sequences only.
        showMessage("Codon-aware alignment works on whole coding sequences: use Realign All or Realign Selected, or switch Sequence type to Nucleotide to realign a block.", 6000);
        return;
    }
    const extraArgs = extra.args.slice();

    let applyAdjustFull = opts.applyAdjustFull;
    let applyReorder = opts.applyReorder;

    if (extra.adjustDir && extra.seqType === '2' && applyAdjustFull === undefined) {
        applyAdjustFull = await _mafftAskYesNo({
            message: 'Adjust direction: reverse-complement the full sequences that look reversed in the selected region?\n\nYes = full sequences (recommended)\nNo = selected region only',
            defaultYes: true,
            dontAskKey: MAFFT_ASK_ADJUST,
            defaultKey: MAFFT_DEF_ADJUST,
            skipPrompt: opts.skipPrompt
        });
    }
    if ((extra.reorder || extra.reorderOnly) && applyReorder === undefined) {
        applyReorder = await _mafftAskYesNo({
            message: 'Apply the region similarity order to the whole MSA?\n\nYes = reorder all rows\nNo = keep current row order (recommended)',
            defaultYes: false,
            dontAskKey: MAFFT_ASK_REORDER,
            defaultKey: MAFFT_DEF_REORDER,
            skipPrompt: opts.skipPrompt
        });
    }

    if (extra.reorderOnly && !applyReorder) {
        showMessage('Reorder only: row order not applied. Nothing to do.', 3000);
        return;
    }

    try {
        _mafftResultStale = false;
        _alignRowIds = true;   // RealignRegion sends rows as r<index> too
        const result = await RealignRegion.run(state.seqs, state.selectedColumns, {
            adjustDir: extra.adjustDir && extra.seqType === '2',
            reorder: extra.reorder,
            reorderOnly: extra.reorderOnly,
            applyAdjustFull: applyAdjustFull !== false,
            applyReorder: !!applyReorder,
            extraArgs,
            mafftAlign: (fa, args) => _mafftAlignWithUi(fa, args, extra.reorderOnly ? 'Reordering by selected region…' : 'Realigning region with MAFFT...'),
            parseFasta: parseMafftOutput,
            adjustDirection: (fa) => _adjustDirection(fa),
            reorderByGuideTree: (fa) => _reorderByGuideTree(fa),
            reverseComplement: _rcForAlignment()
        }).finally(() => { _alignRowIds = false; });
        if (result.cancelled) {
            if (!_mafftResultStale) showMessage('Region realign cancelled.', 2500);
            return;
        }
        if (result.noop && !result.aligned && !result.orderApplied && !(result.flipped && result.flipped.length)) {
            showMessage(result.message || 'Nothing to do.', 2500);
            return;
        }

        pushUndo('realign-block');
        // each row comes back with its own names (RealignRegion matches rows by an internal key);
        // looking them up by name gave rows whose short names repeat the first one's full name
        state.seqs = result.seqs.map(r => {
            return {
                header: r.header,
                fullHeader: r.fullHeader || r.header,
                seq: r.seq,
                gaplessPositions: calculateGaplessPositions(r.seq)
            };
        });
        state.selectedColumns.clear();
        // aligned columns, or whole rows reverse-complemented (reorder-only with Adjust direction): the
        // groups and 2D blocks describe columns that no longer hold what they did
        if (result.aligned || (result.flipped && result.flipped.length)) _dropColumnAnalysesAfterRealign();
        renderAlignment();
        const bits = [];
        if (result.aligned) bits.push('Region realigned');
        else if (result.orderApplied) bits.push('Rows reordered (no alignment)');
        if (result.orderApplied && result.aligned) bits.push('rows reordered');
        if (result.flipped && result.flipped.length) bits.push("RC'd: " + result.flipped.join(', '));
        showMessage((bits.join('; ') || result.message || 'Done') + '.', 3500);
    } catch (err) {
        showMessage("Realign block error: " + err.message, 4000);
        console.error("Realign block error:", err);
    }
}

// Realigns the rows of a rectangle selection within its columns (owner, 2026-10-23: "re-align such block").
// The other rows are untouched; the alignment stays rectangular: when the realigned piece is wider than the
// span, the other rows get that many gaps right after the span; when it is narrower, the block rows get gaps
// right after the piece. The selection then covers the new block.
async function _realignRectangle(rect, opts) {
    opts = opts || {};
    const extra = getMafftExtraArgs();
    if (extra.seqType === 'codon' && !extra.reorderOnly) {
        showMessage("Codon-aware alignment works on whole coding sequences: switch Sequence type to Nucleotide to realign a block.", 6000);
        return;
    }
    if (extra.reorderOnly) {
        showMessage("Reorder only works on column spans of all rows; a block of rows x columns is realigned, not reordered.", 4000);
        return;
    }
    if (extra.adjustDir || extra.reorder) showMessage('Adjust direction and Reorder apply to column spans of all rows; the block is realigned without them.', 4000);
    const cols = new Set();
    for (let c = rect.c0; c <= rect.c1; c++) cols.add(c);
    // a ragged alignment (rows of unequal length) is squared with gaps first, so the width change is measured
    // against one length (the first block row's own length gave a wrong delta and padded the wrong rows)
    const L = _maxOf(state.seqs, s => s.seq.length);
    const padTo = (seq, n) => seq.length >= n ? seq : seq + '-'.repeat(n - seq.length);
    const subset = rect.rows.map(i => ({ ...state.seqs[i], seq: padTo(state.seqs[i].seq, L) }));
    const before = subset.map(s => s.seq);
    try {
        _mafftResultStale = false;
        _alignRowIds = true;
        const result = await RealignRegion.run(subset, cols, {
            adjustDir: false, reorder: false, reorderOnly: false, applyAdjustFull: false, applyReorder: false,
            extraArgs: extra.args.slice(),
            mafftAlign: (fa, args) => _mafftAlignWithUi(fa, args, 'Realigning block with MAFFT...'),
            parseFasta: parseMafftOutput,
            adjustDirection: (fa) => _adjustDirection(fa),
            reorderByGuideTree: (fa) => _reorderByGuideTree(fa),
            reverseComplement: _rcForAlignment()
        }).finally(() => { _alignRowIds = false; });
        if (result.cancelled) { if (!_mafftResultStale) showMessage('Block realign cancelled.', 2500); return; }
        if (!result.aligned) { showMessage(result.message || 'Nothing to do.', 2500); return; }
        const spanLen = rect.c1 - rect.c0 + 1;
        const newRows = result.seqs;
        const delta = newRows[0].seq.length - before[0].length;   // the realigned piece is spanLen + delta wide
        const alignedLen = spanLen + delta;
        pushUndo('realign-block');
        state.seqs.forEach((s, i) => { if (!rect.rowSet.has(i) && s.seq.length < L) { s.seq = padTo(s.seq, L); s.gaplessPositions = calculateGaplessPositions(s.seq); } });
        if (delta > 0) {
            const pad = '-'.repeat(delta);
            state.seqs.forEach((s, i) => {
                if (rect.rowSet.has(i)) return;
                s.seq = s.seq.slice(0, rect.c1 + 1) + pad + s.seq.slice(rect.c1 + 1);
                s.gaplessPositions = calculateGaplessPositions(s.seq);
            });
        } else if (delta < 0) {
            const pad = '-'.repeat(-delta), cut = rect.c0 + alignedLen;
            newRows.forEach(r => { r.seq = r.seq.slice(0, cut) + pad + r.seq.slice(cut); });
        }
        rect.rows.forEach((i, k) => {
            const r = newRows[k];
            state.seqs[i] = { header: r.header, fullHeader: r.fullHeader || r.header, seq: r.seq, gaplessPositions: calculateGaplessPositions(r.seq) };
        });
        const width = Math.max(spanLen, alignedLen);
        state.selectedNucs.clear();
        rect.rows.forEach(i => state.selectedNucs.set(i, buildNucSelectionSet(i, rect.c0, rect.c0 + width - 1)));
        state.pendingNucStart = null;
        _dropColumnAnalysesAfterRealign();
        renderAlignment();
        scheduleNucSelectionRefresh();
        showMessage(`Block realigned: ${rect.rows.length} rows, columns ${rect.c0 + 1}-${rect.c1 + 1}` + (delta ? ` (now ${width} columns wide)` : '') + '.', 3500);
    } catch (err) {
        showMessage("Realign block error: " + err.message, 4000);
        console.error("Realign block error:", err);
    }
}

/**
 * Realign all loaded sequences using MAFFT WASM.
 */
async function realignAll() {
    if (state.seqs.length < 2) {
        showMessage("Need at least 2 sequences to align.", 2000);
        return;
    }

    // Save undo state
    const redoBeforeRealign = state.redoHistory;
    pushUndo('realign-all');

    // Build ungapped FASTA. Rows go to MAFFT under internal ids (r0, r1, ...) and come back by row,
    // not by name: two rows with the same name used to both receive the first one's sequence.
    const rowById = new Map(state.seqs.map((s, i) => ['r' + i, s]));
    const nameOfId = (id) => { const s = rowById.get(id); return s ? s.header : id; };
    let fasta = '';
    state.seqs.forEach((s, i) => {
        fasta += `>r${i}\n${s.seq.replace(/[-.]/g, '')}\n`;
    });

    const { args: extraArgs, seqType, adjustDir, reorder, reorderOnly } = getMafftExtraArgs();

    // Pre-alignment: adjust direction if requested
    let flippedNames = new Set();
    if (adjustDir && (seqType === '2' || seqType === 'codon')) {
        const adj = _adjustDirection(fasta);
        fasta = adj.fasta;
        flippedNames = adj.flipped;
        if (flippedNames.size > 0) {
            console.log('Adjust direction: reverse-complemented', [...flippedNames]);
        }
    }

    // Reorder only: skip MAFFT. Apply RC to gapped rows if needed, then permute.
    if (reorderOnly) {
        if (flippedNames.size > 0) {
            const rcAln = _rcForAlignment();
            for (const [id, s] of rowById) {
                if (flippedNames.has(id)) {
                    s.seq = rcAln(s.seq);
                    s.gaplessPositions = calculateGaplessPositions(s.seq);
                    // these positions now hold other bases (ids are r<row>)
                    const row = Number(id.slice(1));
                    state.selectedNucs.delete(row);
                    if (state.pendingNucStart?.row === row) state.pendingNucStart = null;
                    if (state.editCell?.row === row) state.editCell = null;
                }
            }
            _dropColumnAnalysesAfterRealign();   // a turned row changes its bases in every column
        }
        const reordered = _reorderByGuideTree(fasta);
        const newSeqs = [];
        const used = new Set();
        for (const id of reordered.order) {
            const s = rowById.get(String(id).split(/\s+/)[0]);
            if (s && !used.has(s)) {
                newSeqs.push(s);
                used.add(s);
            }
        }
        for (const s of state.seqs) if (!used.has(s)) newSeqs.push(s);
        state.seqs = newSeqs;
        state.selectedColumns.clear();
        renderAlignment();
        const msgs = [`Reordered ${state.seqs.length} sequences (no alignment).`];
        if (flippedNames.size > 0) msgs.push(`RC'd: ${[...flippedNames].map(nameOfId).join(', ')}`);
        showMessage(msgs.join(' '), flippedNames.size > 0 ? 5000 : 2000);
        return;
    }

    // originalCaseByHeader captures case from the fasta TEXT actually sent to
    // MAFFT -- i.e. AFTER any adjustDir reverse-complementing, so a flipped
    // sequence's case is recorded in the same (flipped) base order the
    // aligned output will come back in. See _remapCaseOntoAligned.
    const originalCaseByHeader = new Map();
    for (const entry of parseMafftOutput(fasta)) originalCaseByHeader.set(entry.name, entry.seq);

    // Pre-alignment: reorder by guide tree (6-mer distances) like MAFFT --reorder
    let guideOrder = null;
    if (reorder) {
        const reordered = _reorderByGuideTree(fasta);
        fasta = reordered.fasta;
        guideOrder = reordered.order;
    }

    try {
        _alignRowIds = true;
        let result;
        try { result = await _mafftAlignWithUi(fasta, extraArgs, 'Aligning all sequences with MAFFT...'); }
        finally { _alignRowIds = false; }
        if (!result) { _dropNoOpUndo(redoBeforeRealign); return; }   // cancelled: no change, no undo step
        const aligned = parseMafftOutput(result);
        if (aligned.length === 0) {
            showMessage("Error: MAFFT returned no sequences.", 3000);
            return;
        }

        // Build new seqs array
        const newSeqs = [];
        // Determine output order: guide tree order or original input order
        const outputOrder = guideOrder || state.seqs.map((s, i) => 'r' + i);
        const alignedById = new Map(aligned.map(x => [x.name.split(/\s+/)[0], x]));
        // Every row must come back. A row with no residues (all gaps) may be dropped by the aligner
        // (the codon/MACSE engine does): it returns as a gap row of the new width, in its place. Any
        // other missing row means the result is incomplete, and nothing is changed.
        const hasResidues = (q) => /[^-.\s]/.test(q.seq);
        const lost = [...rowById].filter(([id, q]) => !alignedById.has(id) && hasResidues(q)).map(([, q]) => q.header);
        if (lost.length) {
            _dropNoOpUndo(redoBeforeRealign);   // the undo entry pushed for this realign: nothing changed
            showMessage(`The aligner returned no sequence for ${lost.slice(0, 3).join(', ')}${lost.length > 3 ? ` and ${lost.length - 3} more` : ''}; the alignment was not changed.`, 6000);
            return;
        }
        const newWidth = aligned.reduce((m, x) => Math.max(m, x.seq.length), 0);
        const placed = new Set();
        const gapRow = (orig) => ({ header: orig.header, fullHeader: orig.fullHeader, seq: '-'.repeat(newWidth), gaplessPositions: calculateGaplessPositions('-'.repeat(newWidth)) });
        for (const id of outputOrder) {
            const key = String(id).split(/\s+/)[0];
            const a = alignedById.get(key);
            const orig = rowById.get(key);
            placed.add(key);
            if (!a && orig) { newSeqs.push(gapRow(orig)); continue; }
            if (a) {
                const origCase = originalCaseByHeader.get(a.name);
                newSeqs.push({
                    header: orig ? orig.header : a.name.split(/\s+/)[0],
                    fullHeader: orig ? orig.fullHeader : a.name,
                    seq: origCase ? _remapCaseOntoAligned(origCase, a.seq) : a.seq,
                    gaplessPositions: calculateGaplessPositions(a.seq)
                });
            }
        }
        for (const [key, orig] of rowById) if (!placed.has(key)) newSeqs.push(alignedById.has(key) ? null : gapRow(orig));   // not in the guide-tree order
        for (let i = newSeqs.length - 1; i >= 0; i--) if (!newSeqs[i]) newSeqs.splice(i, 1);
        // Fallback: use aligned in MAFFT output order
        if (newSeqs.length === 0) {
            for (const a of aligned) {
                const orig = rowById.get(a.name.split(/\s+/)[0]);
                const origCase = originalCaseByHeader.get(a.name);
                newSeqs.push({
                    header: orig ? orig.header : a.name.split(/\s+/)[0],
                    fullHeader: orig ? orig.fullHeader : a.name,
                    seq: origCase ? _remapCaseOntoAligned(origCase, a.seq) : a.seq,
                    gaplessPositions: calculateGaplessPositions(a.seq)
                });
            }
        }

        state.seqs = newSeqs;
        state.selectedRows.clear();
        state.selectedColumns.clear();
        state.selectedNucs.clear();
        _dropColumnAnalysesAfterRealign();
        renderAlignment();
        const msgs = [`Aligned ${state.seqs.length} sequences successfully!`];
        if (flippedNames.size > 0) msgs.push(`RC'd: ${[...flippedNames].map(nameOfId).join(', ')}`);
        showMessage(msgs.join(' '), flippedNames.size > 0 ? 5000 : 2000);
    } catch (err) {
        _dropNoOpUndo(redoBeforeRealign); showMessage("MAFFT alignment error: " + err.message, 4000);
        console.error("MAFFT alignment error:", err);
    }
}

/**
 * Realign only the selected sequences using MAFFT WASM.
 * Non-selected sequences stay unchanged. Selected sequences are realigned
 * among themselves and padded/trimmed to fit back into the alignment.
 */
function realignSelected() {
    if (state.selectedRows.size < 2) {
        showMessage("Select at least 2 sequences to realign.", 2000);
        return;
    }

    // Save undo state
    const redoBeforeRealign = state.redoHistory;
    pushUndo('realign-selected');

    const selectedIndices = Array.from(state.selectedRows).sort((a, b) => a - b);

    // Build ungapped FASTA from selected sequences, under internal ids (r<row>) so rows with the
    // same name each get their own aligned sequence back
    let fasta = '';
    for (const idx of selectedIndices) {
        const s = state.seqs[idx];
        fasta += `>r${idx}\n${s.seq.replace(/[-.]/g, '')}\n`;
    }

    const { args: extraArgs, seqType, adjustDir } = getMafftExtraArgs();

    // Pre-alignment: adjust direction if requested
    if (adjustDir && (seqType === '2' || seqType === 'codon')) {
        const adj = _adjustDirection(fasta);
        fasta = adj.fasta;
        if (adj.flipped.size > 0) {
            console.log('Adjust direction (selected): reverse-complemented', [...adj.flipped].map(id => state.seqs[+id.slice(1)]?.header || id));
        }
    }

    // See originalCaseByHeader note in realignAll -- captured post-adjustDir.
    const originalCaseByHeader = new Map();
    for (const entry of parseMafftOutput(fasta)) originalCaseByHeader.set(entry.name, entry.seq);

    _alignRowIds = true;
    _mafftAlignWithUi(
        fasta,
        extraArgs,
        `Realigning ${selectedIndices.length} selected sequences with MAFFT...`
    ).finally(() => { _alignRowIds = false; }).then(result => {
        if (!result) { _dropNoOpUndo(redoBeforeRealign); return; }   // cancelled: no change, no undo step
        const aligned = parseMafftOutput(result);
        if (aligned.length === 0) {
            showMessage("Error: MAFFT returned no sequences.", 3000);
            return;
        }

        // Replace selected sequences with aligned versions
        const alignedById = new Map(aligned.map(a => [a.name.split(/\s+/)[0], a]));
        for (const idx of selectedIndices) {
            const match = alignedById.get('r' + idx);
            if (match) {
                const origCase = originalCaseByHeader.get(match.name);
                const recased = origCase ? _remapCaseOntoAligned(origCase, match.seq) : match.seq;
                state.seqs[idx].seq = recased;
                state.seqs[idx].gaplessPositions = calculateGaplessPositions(recased);
            }
        }

        // Pad all sequences to the same length
        const maxLen = _maxOf(state.seqs, s => s.seq.length);
        for (const s of state.seqs) {
            s.seq = s.seq.padEnd(maxLen, '-');
        }

        state.selectedColumns.clear();
        _dropColumnAnalysesAfterRealign();   // the realigned rows' columns changed
        renderAlignment();
        showMessage(`Realigned ${selectedIndices.length} selected sequences!`, 2000);
    }).catch(err => {
        _dropNoOpUndo(redoBeforeRealign); showMessage("Realign selected error: " + err.message, 4000);
        console.error("Realign selected error:", err);
    });
}
