/**
 * Walk a read's CIGAR and return a comprehensive set of features for
 * Phase 5 visual treatment: match/mismatch bases,
 * deletion positions, insertion positions, and left/right soft-clip lengths.
 */
function _getReadCigarFeatures(read, refSeq) {
    const features = {
        bases: new Map(),
        deletions: new Set(),
        insertions: [],
        softClipLeft: 0,
        softClipRight: 0
    };
    if (!read.seq || !read.cigar) return features;
    let readPos = 0;
    let refPos = read.pos;
    let seenRefConsuming = false;
    for (const op of read.cigar) {
        const len = op.len;
        const type = op.op;
        switch (type) {
            case 'M': case '=': case 'X':
                seenRefConsuming = true;
                for (let i = 0; i < len; i++) {
                    const refb = (refSeq[refPos + i] || 'N').toUpperCase();
                    // '=' in SEQ: the reference base (it was a base of its own, a mismatch everywhere)
                    let rb = (read.seq[readPos + i] || 'N').toUpperCase();
                    if (rb === '=') rb = refb;
                    const isMatch = type === '=' || (type === 'M' && rb === refb);
                    features.bases.set(refPos + i, { base: rb, type: isMatch ? 'match' : 'mismatch' });
                }
                readPos += len;
                refPos += len;
                break;
            case 'D':
                seenRefConsuming = true;
                for (let i = 0; i < len; i++) {
                    features.deletions.add(refPos + i);
                }
                refPos += len;
                break;
            case 'N':
                seenRefConsuming = true;
                refPos += len;
                break;
            case 'I':
                features.insertions.push({ refPos: refPos - 1, length: len, bases: (read.seq.substr(readPos, len) || '').toUpperCase() });
                readPos += len;
                break;
            case 'S':
                if (!seenRefConsuming) {
                    features.softClipLeft = len;
                } else {
                    features.softClipRight = len;
                }
                readPos += len;
                break;
        }
    }
    return features;
}

/**
 * Lazily create the "Show bases" checkbox for Reads mode and insert it
 * next to the quick mode switcher. Returns the toggle label element.
 */
function ensureReadsDisplayToggle() {
    let toggle = document.getElementById('readsDisplayToggle');
    if (toggle) {
        const cb = document.getElementById('readsDisplayBases');
        if (cb) cb.checked = (_readsDisplayMode === 'bases');
        return toggle;
    }
    toggle = document.createElement('label');
    toggle.id = 'readsDisplayToggle';
    toggle.style.cssText = 'display:none;align-items:center;gap:4px;font-size:11px;margin-left:8px;cursor:pointer;user-select:none;';
    const cb = document.createElement('input');
    cb.type = 'checkbox';
    cb.id = 'readsDisplayBases';
    cb.checked = (_readsDisplayMode === 'bases');
    cb.style.margin = '0';
    cb.addEventListener('change', () => {
        _readsDisplayMode = cb.checked ? 'bases' : 'diff';
        renderReadsAlignment();
    });
    const span = document.createElement('span');
    span.textContent = 'Show bases';
    toggle.appendChild(cb);
    toggle.appendChild(span);
    const quickSwitch = document.getElementById('quickModeSwitch');
    if (quickSwitch && quickSwitch.parentNode) {
        quickSwitch.parentNode.insertBefore(toggle, quickSwitch.nextSibling);
    } else {
        document.body.appendChild(toggle);
    }
    return toggle;
}

function addSampleRowFromReads() {
    if (!bamState.reads.length || !bamState.refSeq || typeof SampleCall === 'undefined') return;
    const ri = state.seqs.findIndex(s => String(s.fullHeader || s.header || '').trim().split(/\s+/)[0] === bamState.refName);
    if (ri < 0) { showMessage(`The reference ${bamState.refName} is no longer in the alignment.`, 4000); return; }
    const res = _readsCalls();   // every read, not only the drawn ones
    const refRow = state.seqs[ri].seq;
    // sample row over the alignment's columns; outside the reads it is missing ('-', as a leading or trailing gap)
    const row = [];
    let p = 0;
    const colOfPos = [];
    for (let c = 0; c < refRow.length; c++) {
        const ch = refRow[c];
        if (ch === '-' || ch === '.') { row.push('-'); continue; }
        colOfPos[p] = c;
        row.push(res.bases[p] || '-');
        p++;
    }
    // <reference>_sample; a second click selects the row it added (it added a duplicate row and the insertion
    // columns again); the reads of another file get their own row, <reference>_<file name>
    let name = `${bamState.refName}_sample`;
    const fromFile = bamState.sampleName || '';
    const taken = state.seqs.find(s => s.header === name);
    if (taken && taken._fromReads !== fromFile) name = `${bamState.refName}_${fromFile}`.replace(/\s+/g, '_');
    const existing = state.seqs.findIndex(s => s.header === name && s._fromReads === fromFile);
    if (existing >= 0) {
        state.selectedRows = new Set([existing]);
        state.lastSelectedIndex = existing;
        _leaveReadsForCanvas();
        renderAlignment();
        showMessage(`${name} is already row ${existing + 1} (selected). Another sample: open its reads file, then Sample → row.`, 5000);
        openVariantsPanel();
        return;
    }
    pushUndo('sample-row');
    // insertions most reads carry: columns before the reference base; gap columns already there (another
    // sample's insertion) are used first, new ones (gaps in every other row) only for the rest. Right to left.
    const ins = [...res.insAfter].filter(([q]) => colOfPos[q] !== undefined).sort((a, b) => b[0] - a[0]);
    for (const [q, bases] of ins) {
        const c = colOfPos[q];
        let g = 0;
        while (c - g - 1 >= 0 && (refRow[c - g - 1] === '-' || refRow[c - g - 1] === '.') && (q === 0 || colOfPos[q - 1] < c - g - 1)) g++;
        const add = Math.max(0, bases.length - g);
        if (add) {
            const pad = '-'.repeat(add);
            for (const s of state.seqs) { s.seq = s.seq.slice(0, c) + pad + s.seq.slice(c); s.gaplessPositions = calculateGaplessPositions(s.seq); }
            row.splice(c, 0, ...pad.split(''));
        }
        for (let k = 0; k < bases.length; k++) row[c - g + k] = bases[k];
    }
    const seq = row.join('');
    const used = (bamState.allReads || bamState.reads).length;
    state.seqs.splice(ri + 1, 0, {
        header: name,
        fullHeader: `${name} called from ${used.toLocaleString()} reads (depth >= ${SampleCall.DEFAULTS.minDepth}, second allele >= ${Math.round(SampleCall.DEFAULTS.hetFrac * 100)}% = IUPAC)`,
        seq,
        gaplessPositions: calculateGaplessPositions(seq),
        _fromReads: fromFile
    });
    // to Canvas view: the rows are what codon analysis and MACSE work on (the reads stay loaded for the Reads view)
    _leaveReadsForCanvas(true);
    renderAlignment();
    updateBamButtonVisibility();
    const v = res.variants;
    const fmt = x => `${x.pos} ${x.type === 'snv' ? x.ref + '>' + x.alt : x.type === 'del' ? 'del ' + x.ref : 'ins ' + x.alt} ${x.zyg}`;
    const het = v.filter(x => x.zyg === 'het' && x.type !== 'snv');
    console.log('Sample row: changes against ' + bamState.refName, v);
    showMessage(`Added ${name} under ${bamState.refName}: ${v.length ? v.length + ' change' + (v.length > 1 ? 's' : '') + ' (' + v.slice(0, 8).map(fmt).join(', ') + (v.length > 8 ? ', ...' : '') + ')' : 'no changes'}.`
        + (het.length ? ` Heterozygous indels are not in the row (one row holds one haplotype): ${het.map(fmt).join(', ')}.` : '')
        + ' The Variants window lists them with depth, strands and their effect on each gene; Codon analysis (Display) marks them in the alignment.', 12000);
    openVariantsPanel();
}

/**
 * Render the reads alignment view (BAM mode).
 * Shows reference track at top, reads below in pileup order.
 */
// The Reads view is one canvas that holds the window's size, over a spacer as large as the whole pile: native scrolling
// moves the spacer and each frame draws only the bases and tracks on screen. It was one SVG of the whole reference,
// with a node per read, cap, mismatch and base: 1 Mb at 30x took 90 s and 4.6 million nodes, a frame up to 1.6 s.
const _READS_CELL = 12, _READS_NAME_W = 160, _READS_TRACK_H = 16, _READS_REF_H = 22, _READS_SCALE_H = 14, _READS_STATUS_H = 21;
// Compact: rows of 4 px (the reads as bars and ticks), for a deep place. The gene row is there only with genes on this row.
const _READS_COMPACT_H = 4, _READS_GENE_LANE = 14, _READS_GENE_LANES = 3;
let _readsCompact = false;
const _readsTrackH = () => (_readsCompact ? _READS_COMPACT_H : _READS_TRACK_H);
const _readsGeneH = () => (_readsView ? _readsView.geneH : 0);
const _readsVcfH = () => (_readsView ? _readsView.vcfH : 0);
const _readsHeadH = () => _READS_REF_H + _READS_SCALE_H + _COV_H + _readsGeneH() + _readsVcfH() + _READS_STATUS_H;
// Zoom: px per base, from the whole region to _READS_CELL. Letters need _READS_LETTERS px; past _READS_MAX_DRAWN aligned
// bases on screen only coverage is drawn: about 40 kb at 30x, the whole of a few long reads (igv.js: a fixed 30 kb). The spacer is capped (an element stops growing near 33
// million px, 2.8 Mb at 12 px per base) and its scroll range mapped onto the pile: _readsLeft / _readsSetLeft.
const _READS_LETTERS = 7, _READS_MAX_DRAWN = 1200000, _READS_MAX_SPACER = 16000000, _READS_ZOOM_STEP = 1.5;
const _readsCell = () => (_readsView ? _readsView.cell : _READS_CELL);
function _readsScrollRatio(v) {
    const spacerW = Math.max(Math.min(v.fullW, _READS_MAX_SPACER), v.w);
    return spacerW > v.w ? Math.max(0, v.fullW - v.w) / (spacerW - v.w) : 1;
}
// (a genome view keeps its place itself: at 250 Mb one pixel of this range is 188 pixels of the view)
const _readsLeft = () => (_readsView ? (_genome ? _genome.left : alignmentContainer.scrollLeft * _readsScrollRatio(_readsView)) : alignmentContainer.scrollLeft);
function _readsSetLeft(px) {
    if (!_readsView) return;
    if (_genome) { _genomeSetLeft(px); return; }
    alignmentContainer.scrollLeft = Math.max(0, px) / (_readsScrollRatio(_readsView) || 1);
    // the bar too, at once: it trusts its own place during input (Enter counts) and wrote the old one back a frame later
    window._syncHorizontalScrollbar?.();
}
const _readsCoverageOnly = v => (_genome ? _genomeCoverageOnly(v) : ((v.w - _READS_NAME_W) / v.cell) * (bamState.alignedBases || 0) / Math.max(1, bamState.refEnd - bamState.refStart) > _READS_MAX_DRAWN);
function _readsSizeSpacer(v) {
    v.fullW = _READS_NAME_W + (bamState.refEnd - bamState.refStart) * v.cell;
    v.spacer.style.width = Math.max(Math.min(v.fullW, _READS_MAX_SPACER), v.w) + 'px';
    v.fullH = _readsHeadH() + (bamState.nTracks || 0) * _readsTrackH() + 4;
    v.spacer.style.height = (_readsCoverageOnly(v) ? v.h : Math.max(v.fullH, v.h)) + 'px';
    if (_genome) _genomeSyncBar(v);
}
const _readsMinCell = v => Math.min(_READS_CELL, (v.w - _READS_NAME_W) / Math.max(1, bamState.refEnd - bamState.refStart));
// The base at x on the canvas stays at x
function _readsZoomTo(cell, x) {
    const v = _readsView;
    if (!v) return;
    cell = Math.max(_readsMinCell(v), Math.min(_READS_CELL, cell));
    const base = (x + _readsLeft() - _READS_NAME_W) / v.cell;
    v.cell = cell;
    _readsSizeSpacer(v);
    _readsSetLeft(_READS_NAME_W + base * cell - x);
    _readsDraw();
}
function _readsZoomBy(f, x) {
    const v = _readsView;
    if (v) _readsZoomTo(v.cell * f, x == null ? _READS_NAME_W + (v.w - _READS_NAME_W) / 2 : x);
}
// Position p (0-based) in the middle of the plot
function _readsCentre(p) {
    const v = _readsView;
    if (!v) return;
    _readsSetLeft(_READS_NAME_W + (p - bamState.refStart) * v.cell - (_READS_NAME_W + (v.w - _READS_NAME_W) / 2));
    _readsDraw();
}
// What the jump box takes, as 0-based inclusive { from, to } or { error }: a position or range as on the ruler, the same
// after a name (the reads file's name counts in the file's coordinates, so on a region row chrJ:1,500 is ruler 500), or
// a gene of the annotation (any case; the start of a name when only one gene has it).
function _readsParseJump(text) {
    const t = String(text || '').trim(), { refStart, refEnd } = bamState;
    if (!t) return { error: '' };
    const NUM = /^([\d,\s]+?)(?:\s*(?:-|\u2013|\.\.)\s*([\d,\s]+))?$/;
    const fmt = n => n.toLocaleString();
    const inPlot = (_genome ? 'This chromosome runs from ' : 'The reads cover ') + fmt(refStart + 1) + ' to ' + fmt(refEnd) + ' here.';
    const range = (m, shift) => {
        const n = x => parseInt(String(x).replace(/[,\s]/g, ''), 10);
        let a = n(m[1]) - 1 - shift, b = m[2] != null ? n(m[2]) - 1 - shift : null;
        if (b != null && b < a) [a, b] = [b, a];
        if (a >= refEnd || (b ?? a) < refStart || (b == null && a < refStart)) return { error: `${t} is outside the reads. ${inPlot}` };
        return b == null ? { from: a, to: a, single: true } : { from: Math.max(a, refStart), to: Math.min(b, refEnd - 1) };
    };
    let m = NUM.exec(t);
    if (m) return range(m, 0);
    const colon = t.lastIndexOf(':');
    if (colon > 0 && (m = NUM.exec(t.slice(colon + 1).trim()))) {
        const name = t.slice(0, colon).trim();
        if (name === bamState.refName) return range(m, 0);
        if (name === bamState.bamRefName) return range(m, bamState.offset || 0);
        return { error: `These reads are on ${bamState.bamRefName || bamState.refName}, not ${name}.` };
    }
    const genes = _readsGenes(), lc = t.toLowerCase();
    let hits = genes.filter(g => String(g.name).toLowerCase() === lc);
    if (!hits.length) hits = genes.filter(g => String(g.name).toLowerCase().startsWith(lc));
    if (!hits.length) return { error: `No gene ${t} in the annotation, and it is not a position. Type a position, a range like 1,000-2,000, or a gene.` };
    const names = [...new Set(hits.map(g => g.name))];
    if (names.length > 1) return { error: `Several genes match ${t}: ${names.slice(0, 6).join(', ')}${names.length > 6 ? ', ...' : ''}.` };
    const from = Math.min(...hits.map(g => g.start)), to = Math.max(...hits.map(g => g.end)) - 1;
    if (to < refStart || from >= refEnd) return { error: `${names[0]} is outside the reads. ${inPlot}` };
    return { from: Math.max(from, refStart), to: Math.min(to, refEnd - 1) };
}
// A position goes to the middle of the view, marked, and to letters if they were too small; a range fills the view
function readsJump(text) {
    if (_genome) { _genomeJump(text); return; }
    const v = _readsView, r = _readsParseJump(text);
    if (!v || r.error === '') return;
    if (r.error) { showMessage(r.error, 6000); return; }
    if (r.single) { readsGoTo(r.from, { sort: false }); return; }
    _readsFit(r.from, r.to);
}
// Positions from..to (0-based, inclusive) fill the view; a short stretch stays at full size, in the middle
function _readsFit(from, to) {
    const v = _readsView;
    if (!v) return;
    bamState.focusPos = null;
    const want = (v.w - _READS_NAME_W) / (to - from + 1);
    v.cell = Math.max(_readsMinCell(v), Math.min(_READS_CELL, want));
    _readsSizeSpacer(v);
    if (want > _READS_CELL) _readsCentre((from + to) / 2);
    else { _readsSetLeft((from - bamState.refStart) * v.cell); _readsDraw(); }
}
// Genes of the row in lanes, so overlapping ones do not cover each other; past the last lane they share it
function _readsGeneLanes() {
    const genes = _readsGenes().slice().sort((a, b) => a.start - b.start || b.end - a.end), ends = [];
    for (const g of genes) {
        let l = ends.findIndex(e => e <= g.start);
        if (l < 0) l = ends.length < _READS_GENE_LANES ? ends.length : _READS_GENE_LANES - 1;
        ends[l] = Math.max(ends[l] || 0, g.end);
        g.lane = l;
    }
    return { genes, h: ends.length ? ends.length * _READS_GENE_LANE + 4 : 0 };
}
