// A regex that can take time doubling with the length of the text: a group that holds a repeat and is
// itself repeated, as in (A+)+$ or (A*)*C. Found by folding the innermost groups one at a time.
function _regexMayExplode(src) {
    let t = String(src || '').replace(/\\./g, 'x').replace(/\[[^\]]*\]/g, 'x');
    const rep = /[+*]|\{\d+,\d*\}/;
    for (let guard = 0; guard < 200; guard++) {
        const m = /\((?:\?[:=!]|\?<[=!]|\?<[^>]*>)?([^()]*)\)([+*]|\{\d+,\d*\})?/.exec(t);
        if (!m) return false;
        if (m[2] && rep.test(m[1])) return true;
        t = t.slice(0, m.index) + (rep.test(m[1]) || m[2] ? 'x+' : 'x') + t.slice(m.index + m[0].length);
    }
    return true;
}

function _loadSnapshotPayload(payload) {
    if (!payload || typeof payload !== 'object') {
        throw new Error('Invalid snapshot payload');
    }
    if (!payload.fasta || typeof payload.fasta !== 'string') {
        throw new Error('Snapshot has no alignment data');
    }

    const fastaInputEl = el('fastaInput');
    if (fastaInputEl) _setInputText(payload.fasta);

    state.currentFilename = payload.sourceTitle || payload.title || 'Snapshot';
    state.currentFilePath = '';

    // Parse and render first, THEN apply view settings

    return parseAndRender(true).then(() => {
        if (!state.seqs.length) return;   // the alignment text could not be read: the error is on screen, nothing to apply the view to (audit M6)
        // the trim first: _applySnapshotView renders, and the consensus, the ruler and the Canvas read the trim then
        if (payload.trim && Number.isInteger(payload.trim.leftTrimEnd) && Number.isInteger(payload.trim.rightTrimStart)) {
            _setTrimBounds(payload.trim.leftTrimEnd, payload.trim.rightTrimStart, false);
            state.trimManualLeft = !!payload.trim.manualLeft; state.trimManualRight = !!payload.trim.manualRight;
        }
        _applySnapshotView(payload.view || {});
        // A snapshot gives back exactly what was saved: no colours saved means none shown
        _applySnapshotColourState(payload.colourState || { mappings: [] });
        if (payload.searchHistory) {
            _applySnapshotSearchHistory(payload.searchHistory, payload.view?.maxMismatches);
        }
        // Older snapshots have no selections: still clear what the previous session showed
        _applySnapshotSelections(payload.selections || {});
        // The dot-plot collection of the snapshot joins the user's own (by id)
        if (Array.isArray(payload.dotBookmarks)) {
            if (!Array.isArray(state.dotBookmarks)) state.dotBookmarks = [];
            const have = new Set(state.dotBookmarks.map(b => b.id));
            _dotSanitizeBookmarks(payload.dotBookmarks).forEach(b => { if (!have.has(b.id)) { state.dotBookmarks.push(b); have.add(b.id); } });
            _dotSaveBookmarks(); _dotRenderBookmarks();
        }
        // The annotation track is part of what was saved (none saved: none shown)
        if (payload.annotation && typeof payload.annotation.text === 'string') {
            const cb = el('showAnnotation');
            if (cb) cb.checked = payload.annotation.shown !== false;
            setAnnotation(payload.annotation.text, payload.annotation.sourceName || 'snapshot');
        } else {
            clearAnnotation();
        }
        // a saved pane layout opens its panes again (in the main page only)
        if (payload.panes) _paneRestoreLayout(payload.panes);
    });
}

// Colours read from a snapshot (a shared link or file) end up in CSS rules:
// accept only a plain colour, never text that could close the rule.
function _safeCssColor(c) {
    const v = String(c == null ? '' : c).trim();
    if (/^#[0-9a-f]{3,8}$/i.test(v) || /^[a-z]{3,30}$/i.test(v)
        || /^(rgb|hsl)a?\([\d\s.,%/+-]+\)$/i.test(v)) return v;
    return null;
}

function _applySnapshotColourState(cs) {
    if (!cs || !Array.isArray(cs.mappings)) return;
    // Replace, not merge - like selectedRows/selectedColumns above, loading a snapshot
    // should give back exactly what was saved, not saved-plus-whatever-was-already-there.
    colourState.mappings.clear();
    cs.mappings.forEach(([name, color]) => {
        const safe = _safeCssColor(color);
        if (safe) colourState.mappings.set(name, safe);
    });
    if (typeof applyColourToSeqNames === 'function') {
        applyColourToSeqNames(colourState.mappings);
    }
}

// legacyMismatches: the snapshot's view.maxMismatches, the only record old snapshots kept
function _applySnapshotSearchHistory(sh, legacyMismatches) {
    if (!Array.isArray(sh)) return;
    state.searchHistory.forEach(_removeSearchStyle);
    state.searchHistory = [];
    sh.forEach(entry => {
        const strand = entry.strand || 'fwd';
        const rawMotif = (entry.motif || '').replace(/:(fwd|rev comp|rev)$/i, '');
        const motifKey = entry.motif && entry.motif.includes(':') ? entry.motif : `${rawMotif}:${strand}`;
        const className = 'search-hit-' + Math.random().toString(36).substring(2, 12);
        // Older snapshots have no searchValue; their motif key is "<motif>:<strand>", with
        // a trailing ":rev" on the reverse-complement key (the label is display text only).
        const legacyValue = rawMotif.replace(/:rev$/i, '');
        // Old entries had no useRegex: a motif is plain letters, anything else was a regex
        const legacyIsRegex = !entry.searchValue && /[^A-Za-z]/.test(legacyValue);
        let legacyRegex = null;
        const re = !entry.searchValue && /^restriction:[^:]*:([^:]+):(fwd|rev)/i.exec(motifKey);
        if (re) legacyRegex = restrictionSiteToRegex(re[2] === 'rev' ? reverseComplement(re[1]).replace(/U/g, 'T') : re[1]);
        state.searchHistory.push({
            id: className,
            motif: motifKey,
            color: _safeCssColor(entry.color) || '#ffcc00',
            className,
            label: entry.label || rawMotif,
            strand,
            searchValue: entry.searchValue || legacyRegex || (/rev comp/i.test(strand) ? reverseComplement(legacyValue) : legacyValue),
            useRegex: !!entry.useRegex || !!legacyRegex || legacyIsRegex,
            maxMismatches: Number.isInteger(entry.maxMismatches) ? entry.maxMismatches
                : (legacyRegex || legacyIsRegex ? 0 : (parseInt(legacyMismatches, 10) || 0)),
            enabled: entry.enabled !== false,
            matchCount: entry.matchCount || 0,
            sequencesWithMatches: entry.sequencesWithMatches || 0
        });
    });
    // A saved pattern that can freeze the tab is loaded switched off (audit M7); the user can switch it on
    const risky = state.searchHistory.filter(h => h.useRegex && h.enabled && _regexMayExplode(h.searchValue));
    risky.forEach(h => { h.enabled = false; });
    bumpSearchLayer();
    reapplySearchHighlights();
    updateActiveSearchesPanel();
    if (risky.length) setTimeout(() => showMessage(`A saved search was switched off because its pattern can take very long: ${risky.map(h => h.searchValue).join(', ').slice(0, 80)}`, 7000), 50);
}
