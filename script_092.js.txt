// Direct-DOM highlight removal (fast, no full renderAlignment)
function _repeatInfoFromRow(tr) {
    // the anchored result (sequence + residue ranges) if the search made one: the row and
    // columns in the table are from when it ran
    const ri = parseInt(String(tr.dataset.repeatId || '').split('-').pop(), 10);
    const res = _lastRepeatResults?.[ri];
    if (res?._info?.anchor && String(tr.dataset.repeatId).startsWith((_lastRepeatView?.mode || '') + '-')) {
        return { segs: res._info.segs.map(x => x.slice()), row: res._info.row, color: tr.dataset.color, anchor: res._info.anchor };
    }
    const segs = (tr.dataset.segs || `${tr.dataset.start}-${tr.dataset.end}`).split(';')
        .map(x => x.split('-').map(Number)).filter(a => a.length === 2 && a.every(Number.isFinite));
    const row = tr.dataset.row === '' || tr.dataset.row == null ? null : parseInt(tr.dataset.row, 10);
    return { segs, row: Number.isInteger(row) ? row : null, color: tr.dataset.color };
}

function _removeRepeatHighlight(row) {
    const rid = row.dataset.repeatId;
    const hl = state.repeatHighlights;
    if (hl.has(rid)) {
        const info = hl.get(rid);
        hl.delete(rid);
        _paintRepeatHighlight(info, false);   // resync: survivors sharing those columns stay
    }
    row.dataset.active = '';
    row.style.background = '';
}

function _toggleRepeatHighlight(row) {
    const rid = row.dataset.repeatId;
    const hl = state.repeatHighlights;
    if (hl.has(rid)) {
        _removeRepeatHighlight(row);
        return;
    }
    const info = _repeatInfoFromRow(row);
    hl.set(rid, info);
    _paintRepeatHighlight(info, true);
    if (info.row !== null) { const v = _repeatView(info); if (v.row >= 0) _showRowInAlignment(v.row, v.segs[0]?.[0]); }
    else requestAnimationFrame(() => _scrollToColumn(info.segs[0]?.[0] ?? 0));
    row.dataset.active = '1';
    row.style.background = info.color + '55';
}

// B2: open or close a microsatellite group row: its pairs are listed under it (20 at a time) and its arrays highlighted
function _toggleRepeatGroup(tr) {
    const v = _lastRepeatView;
    const id = parseInt(tr.dataset.group, 10);
    if (!v || !Number.isInteger(id)) return;
    if (!v.expanded) v.expanded = new Map();
    const rid = tr.dataset.repeatId, hl = state.repeatHighlights;
    if (v.expanded.has(id)) {
        v.expanded.delete(id);
        if (hl.has(rid)) { hl.delete(rid); _paintRepeatHighlight(null, false); }
    } else {
        v.expanded.set(id, 20);
        const info = _repeatInfoFromRow(tr);
        _anchorRepeat(info);
        hl.set(rid, info);
        _paintRepeatHighlight(info, true);
        if (info.row !== null) { const w = _repeatView(info); if (w.row >= 0) _showRowInAlignment(w.row, w.segs[0]?.[0]); }
    }
    const el = document.getElementById('repeatResults');
    if (el) _renderRepeatResultsHTML(el, _lastRepeatResults, v.mode, v.name || '', v.totalLen, v);
}

function _applyLineHighlights(dataSpan) {
    const line = dataSpan.parentElement;
    if (!line || line.classList.contains('consensus-line')) return;
    const row = parseInt(line.dataset.seqIndex, 10);
    if (!(row >= 0)) return;
    const cols = _repeatColsForRow(row);
    if (!cols.size) return;
    for (const span of dataSpan.children) {
        const p = span.dataset.pos;
        if (p !== undefined && cols.has(+p)) _setSpanRepeatPaint(span, cols.get(+p));
    }
}

function _scrollToColumn(colIndex) {
    if (isCanvasMode() || _foldDomInfo()) _foldRevealCol(colIndex);
    const container = document.getElementById('alignmentContainer');
    if (!container) return;
    // Block view (Canvas Block): go to the band that holds the column (offsetX is always 0 there)
    if (isBlockLayout() && _canvasBlockReveal(null, colIndex)) return;
    // Canvas mode: pan the canvas instead of scrolling the container
    if (document.getElementById('modeCanvas')?.checked) {
        const m = _canvasState.metrics;
        if (m?.charW) {
            const slotI = _canvasColToSlotNear(colIndex);
            _canvasState.offsetX = Math.max(0, (slotI >= 0 ? slotI : colIndex) * m.charW - 120);
            _canvasState.scheduleDraw?.();
        }
        return;
    }
    // Measure actual character width from a rendered span
    const sampleSpan = container.querySelector('.seq-data > span[data-pos]');
    let charW = 7.8; // fallback for 13px Courier New
    if (sampleSpan) { charW = sampleSpan.getBoundingClientRect().width || charW; }
    container.scrollLeft = Math.max(0, colIndex * charW - 120);
}

// Search highlight colour: hovering the colour box opens a small palette (a native colour
// dialog cannot open without a click); a click on the box still opens the full picker.
const SEARCH_HOVER_COLOURS = ['#FFFF00', '#FFA500', '#FF6B6B', '#FF69B4', '#BB8FCE', '#45B7D1',
    '#4ECDC4', '#7CFC00', '#98D8C8', '#F8B88B', '#A0A0A0', '#00BFFF'];
function _initSearchColourHover() {
    const input = document.getElementById('searchColor');
    if (!input || input.dataset.hoverPalette) return;
    input.dataset.hoverPalette = '1';
    const pop = document.createElement('div');
    pop.id = 'searchColourPop';
    pop.className = 'search-colour-pop';
    pop.hidden = true;
    for (const c of SEARCH_HOVER_COLOURS) {
        const b = document.createElement('button');
        b.type = 'button'; b.className = 'search-colour-chip'; b.style.background = c; b.title = c; b.dataset.colour = c;
        b.addEventListener('click', () => {
            input.value = c;
            input.dispatchEvent(new Event('input', { bubbles: true }));
            input.dispatchEvent(new Event('change', { bubbles: true }));
            hide(0);
        });
        pop.appendChild(b);
    }
    const more = document.createElement('button');
    more.type = 'button'; more.className = 'search-colour-more'; more.textContent = 'More…';
    more.title = 'Open the full colour picker';
    more.addEventListener('click', () => { hide(0); input.click(); });
    pop.appendChild(more);
    document.body.appendChild(pop);
    let timer = null;
    const mark = () => pop.querySelectorAll('.search-colour-chip').forEach(b =>
        b.classList.toggle('current', b.dataset.colour.toLowerCase() === input.value.toLowerCase()));
    const show = () => {
        clearTimeout(timer);
        if (!pop.hidden) return;
        const r = input.getBoundingClientRect();
        pop.hidden = false; mark();
        const w = pop.offsetWidth;
        pop.style.left = Math.max(4, Math.min(window.innerWidth - w - 4, r.left)) + 'px';
        pop.style.top = (r.bottom + 2) + 'px';
    };
    function hide(delay = 250) { clearTimeout(timer); timer = setTimeout(() => { pop.hidden = true; }, delay); }
    input.addEventListener('mouseenter', show);
    input.addEventListener('mouseleave', () => hide());
    pop.addEventListener('mouseenter', () => clearTimeout(timer));
    pop.addEventListener('mouseleave', () => hide());
    input.addEventListener('click', () => hide(0));
    input.addEventListener('focus', show);
    input.addEventListener('blur', () => hide());
    document.addEventListener('keydown', e => { if (e.key === 'Escape' && !pop.hidden) hide(0); });
}
document.addEventListener('DOMContentLoaded', _initSearchColourHover);
