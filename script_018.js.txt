                return;
            }
        }

        // Row selection: Ctrl+click on name column (mirrors handleRowSelectMouseDown)
        if (isCtrlModifier(e) && !isAltModifier(e)) {
            const nameHit = _canvasHitTestName(e.clientX, e.clientY);
            if (nameHit) {
                const index = nameHit.row;
                state.isDragging = true;
                state.dragStartRow = index;
                state.dragMode = 'row';
                state.selectedNucs.clear();
                state.pendingNucStart = null;
                scheduleNucSelectionRefresh();
                toggleRowSelection(index);
                state.dragRowRemove = !state.selectedRows.has(index);   // a drag from a selected row deselects
                hideTooltip();
                scheduleOverlay();
                return;
            }
        }

        // Row range selection: Shift+click on name column
        if (e.shiftKey && state.lastSelectedIndex !== null) {
            const nameHit = _canvasHitTestName(e.clientX, e.clientY);
            if (nameHit) {
                const index = nameHit.row;
                _rowRangeSelect(state.lastSelectedIndex, index, state.selectedRows.has(index));
                state.lastSelectedIndex = index;
                // a residue click before this started a two-click range; it must not complete with that old anchor (as in Full/Block)
                if (state.pendingNucStart) { state.pendingNucStart = null; scheduleNucSelectionRefresh(); }
                updateRowSelections();
                hideTooltip();
                scheduleOverlay();
                return;
            }
        }

        // Show 2D: a typed block's rectangle takes the click (DOM: the SVG rect's click handler)
        if (!isCtrlModifier(e) && !isAltModifier(e) && !e.shiftKey) {
            const mr = _canvasBlockMaskHit(e.clientX, e.clientY);
            if (mr) { e.preventDefault(); hideTooltip(); groupRowsByType(mr.typeRows); return; }
        }

        const hit = _canvasHitTest(e.clientX, e.clientY);
        if (hit) {
            // Selection: mirror handleNucleotideSelectMouseDown's state mutations
            const idx = hit.row, pos = hit.col;
            if (state.pendingNucStart === null) {
                state.selectedNucs.clear();
                state.pendingNucStart = { row: idx, pos };
                state.isDragging = true;
                state.dragStartRow = idx;
                state.dragStartCol = pos;
                state.dragMode = 'nuc';
                state.selectedNucs.set(idx, new Set([pos]));
            } else if (state.pendingNucStart.row === idx) {
                state.selectedNucs.set(idx, buildNucSelectionSet(idx, state.pendingNucStart.pos, pos));
                state.pendingNucStart = null;
            } else if (state.pendingNucStart.row >= 0 && idx >= 0) {
                // the second click on another row: the rectangle between the two cells (a drag from here replaces it)
                _setNucRectangle(state.pendingNucStart.row, state.pendingNucStart.pos, idx, pos);
                state.pendingNucStart = null;
                state.isDragging = true;
                state.dragStartRow = idx;
                state.dragStartCol = pos;
                state.dragMode = 'nuc';
            } else {
                state.selectedNucs.clear();
                state.pendingNucStart = { row: idx, pos };
                state.isDragging = true;
                state.dragStartRow = idx;
                state.dragStartCol = pos;
                state.dragMode = 'nuc';
