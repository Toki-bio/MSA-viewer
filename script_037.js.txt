/** Run pre-parse scan; for crazy alignments prompt before parse. Returns false if cancelled. */
// What is loaded now (name, path, input text, size class): a load that is cancelled or cannot be read
// puts these back, so the header, copies and Recent do not name a file that never loaded.
let _loadedIdentity = null;
function _rememberLoadedIdentity(text) {
    _loadedIdentity = { name: state.currentFilename, path: state.currentFilePath, text, index: state.alignmentIndex };
}
function _restoreLoadedIdentity() {
    const L = _loadedIdentity;
    state.currentFilename = L ? L.name : '';
    state.currentFilePath = L ? L.path : '';
    state.alignmentIndex = L ? L.index : null;
    if (L && typeof fastaInput !== 'undefined' && fastaInput) _setInputText(L.text);
}
function _fileReadFailed(file, err) {
    console.error('Error reading file:', file && file.name, err);
    state._pendingFileHandle = null;
    _restoreLoadedIdentity();
    const why = err && err.message ? ': ' + err.message : '';
    if (!state.seqs.length) alignmentContainer.innerHTML = '<div class="error-message">Error reading file.</div>';
    showMessage(`Could not read ${file ? file.name : 'the file'}${why}`, 6000);
}

async function runAlignmentPreflight(inputText) {
    const stats = scanAlignmentText(inputText);
    if (!stats || stats.nSeqs < 1) {
        state.alignmentIndex = null;
        return true;
    }
    const classification = classifyAlignmentSize(stats);
    state.alignmentIndex = { ...stats, ...classification };

    if (classification.mode === 'crazy') {
        const choice = await showLargeAlignmentDialog(stats, classification);
        if (choice === 'cancel') return false;
    } else if (classification.mode === 'tall') {
        showMessage(
            `Large alignment (${stats.nSeqs.toLocaleString()} sequences) - vertical paging when enabled.`,
            3500
        );
    } else if (classification.mode === 'long') {
        showMessage(
            `Long alignment (${stats.maxLen.toLocaleString()} columns) - horizontal chunking when enabled.`,
            3500
        );
    }
    return true;
}

// CORE FUNCTIONS
// A pasted URL isn't sequence content itself - recognized here so
// parseAndRender can fetch it and continue, instead of falling through
// every format parser and reporting a generic "no valid sequences found"
// for what was actually a location, not data.
function _looksLikeUrl(text) {
    return /^https?:\/\/\S+$/i.test(text);
}
// Same idea for a pasted local filesystem path, resolved via the local
// dev server's /api/local-cat (read-only, same trust boundary as the
// existing BLAST database file management, which already reads/writes
// arbitrary local paths with no additional auth).
// One or more GenBank/RefSeq accessions (e.g. "NC_012920.1 MN908947"),
// separated by spaces, commas or new lines: fetched from NCBI and loaded.
function _parseAccessionList(text) {
    if (text.length > 5000 || text.startsWith('>')) return null;
    const ids = text.split(/[\s,;]+/).filter(Boolean);
    if (!ids.length || ids.length > 200) return null;
    return ids.every(id => /^[A-Za-z]{1,6}_?\d{4,12}(\.\d+)?$/.test(id)) ? ids : null;
}

// NCBI E-utilities allows cross-origin requests, so this works on the static
// deployment too. Nucleotide records first, then protein (GenPept).
async function _fetchGenBankFromNcbi(ids) {
    const base = 'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?rettype=gbwithparts&retmode=text&id='
        + encodeURIComponent(ids.join(','));
    for (const db of ['nuccore', 'protein']) {
        const resp = await fetch(base + '&db=' + db);
        if (!resp.ok) continue;
        const text = await resp.text();
        if (/^LOCUS\s/m.test(text)) return text;
    }
    throw new Error('NCBI returned no GenBank record for ' + ids.join(', '));
}

function _looksLikeLocalPath(text) {
    if (text.includes('\n') || text.length > 500 || text.startsWith('>')) return false;
    return /^[A-Za-z]:[\\/]/.test(text) || /^\//.test(text) || /^~[\\/]/.test(text) || /^\.\.?[\\/]/.test(text);
}

// A load that failed leaves no alignment behind: the view shows the error, so
// Save, Copy and the analyses must not keep working on the previous file's rows.
// After parseAndRender for ?url= / ?data=: did it load an alignment? A load replaces state.seqs with
// the new rows; a failure empties them (_clearFailedLoad) or leaves the old ones (a cancelled load).
function _autoLoadSucceeded(seqsBefore) {
    return state.seqs !== seqsBefore && Array.isArray(state.seqs) && state.seqs.length > 0;
}

function _clearFailedLoad() {
    if (_genome) _genomeClose();   // its message goes where the genome's canvas was: no genome left half open (GLM audit #35)
    state.seqs = [];
    state.selectedRows?.clear();
    state.selectedColumns?.clear();
    state.selectedNucs?.clear();
    state.currentFilename = '';
    state.currentFilePath = '';
    const src = el('sourceInfo');
    if (src) src.innerHTML = 'No file loaded';
}

async function parseAndRender(isFromDrop = false) {

            statusMessage.style.display = 'none';

            return;

        statusMessage.style.display = 'none';
        if (isFromDrop) {
            _clearFailedLoad();
            showMessage('The file is empty.', 4000);
        }
        return;
    }
    // NUL bytes or mostly control characters: an image, archive or other binary file
    const _ctrl = (inputText.slice(0, 4096).match(/[\x00-\x08\x0E-\x1F]/g) || []).length;
    if (inputText.includes('\x00') || _ctrl > 0.1 * Math.min(4096, inputText.length)) {
        _clearFailedLoad();
        alignmentContainer.innerHTML = '<div class="error-message">This is a binary file, not an alignment. Open FASTA, MSF, Clustal, PHYLIP, NEXUS, Stockholm, GenBank, SAM/BAM or AB1.</div>';
        showMessage('Error: not an alignment (binary file).', 5000);
        return;
    }
    if (/^@[^\n]*\n[^\n]*\n\+/.test(inputText)) {
        _clearFailedLoad();
        alignmentContainer.innerHTML = '<div class="error-message">This is a FASTQ file of reads, not an alignment. Convert it to FASTA, or map the reads (SAM/BAM) onto an open reference.</div>';
        showMessage('Error: FASTQ reads are not an alignment.', 5000);
        return;
    }
    // A snapshot .json downloaded by Snapshot > Create
    if (inputText[0] === '{') {
        let payload = null;
        try { payload = JSON.parse(inputText); } catch (_) {}
        if (payload && typeof payload === 'object' && typeof payload.fasta === 'string' && 'schema' in payload) {
            // An empty "fasta" used to throw, leaving the old rows under the snapshot's file name (audit M6)
            if (!payload.fasta.trim()) {
                _clearFailedLoad();
                alignmentContainer.innerHTML = '<div class="error-message">Error: this snapshot file has no alignment data.</div>';
                showMessage('Error: this snapshot file has no alignment data.', 5000);
                return;
            }
            return _loadSnapshotPayload(payload);
        }
    }

    if (_looksLikeUrl(inputText)) {
        showMessage('Fetching from URL...', 0);
        try {
            const resp = await fetch(inputText);
            if (!resp.ok) throw new Error('HTTP ' + resp.status);
            _setInputText(await resp.text());
            state.currentFilename = state.currentFilename || (inputText.split('/').filter(Boolean).pop() || 'url_import');
            return parseAndRender(isFromDrop);
        } catch (err) {
            statusMessage.style.display = 'none';
            showMessage('Could not fetch URL: ' + err.message, 4000);
            return;
        }
    }
    const accessions = _parseAccessionList(inputText);
    if (accessions) {
        showMessage(`Fetching ${accessions.length === 1 ? accessions[0] : accessions.length + ' records'} from NCBI...`, 0);
        try {
            _setInputText(await _fetchGenBankFromNcbi(accessions));
            state.currentFilename = state.currentFilename || (accessions.length === 1 ? accessions[0] : accessions.length + '_accessions');
            return parseAndRender(isFromDrop);
        } catch (err) {
            statusMessage.style.display = 'none';
            showMessage('Could not fetch from NCBI: ' + err.message, 6000);
            return;
        }
    }
    if (_looksLikeLocalPath(inputText)) {
        showMessage('Reading local file...', 0);
        try {
            const resp = await fetch('/api/local-cat?file=' + encodeURIComponent(inputText));
            // On a static deployment (e.g. GitHub Pages) with no server.js
            // running, /api/local-cat doesn't exist - the host serves its
            // normal 404/index HTML instead. Reading that as JSON used to
            // throw a raw, confusing SyntaxError ("Unexpected token '<'").
            // Detect the non-JSON-response case explicitly and say what's
            // actually true: this feature needs the optional local server.
            let data;
            try {
                data = await resp.json();
            } catch (parseErr) {
                throw new Error('this feature requires the optional local server (server.js) - it is not available on the static/public deployment');
            }
            if (!resp.ok) throw new Error(data.error || ('HTTP ' + resp.status));
            _setInputText(data.content);
            state.currentFilename = state.currentFilename || (inputText.split(/[\\/]/).filter(Boolean).pop() || 'local_import');
            state.currentFilePath = inputText;
            return parseAndRender(isFromDrop);
        } catch (err) {
            statusMessage.style.display = 'none';
            showMessage('Could not read local file: ' + err.message, 6000);
            return;
        }
    }

    const proceed = await runAlignmentPreflight(inputText);
    if (!proceed) {
        statusMessage.style.display = 'none';
        _restoreLoadedIdentity();   // the alignment on screen is still the previous one
        showMessage('Load cancelled.', 2500);
        return;
    }

    showMessage("Parsing file...", 0);
    try {
        let parsed;
    // PRIORITIZE CONTENT-BASED DETECTION.
        const isSamContent = _isSamInput(inputText);
        const isMsfContent = (inputText.includes('MSF:') && inputText.includes('Check:')) ||
                             inputText.includes('!!AA_MULTIPLE_ALIGNMENT') ||
                             inputText.includes('!!NA_MULTIPLE_ALIGNMENT');

        if (isSamContent) {
