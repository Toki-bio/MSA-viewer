function updateVarThresholdBounds() {

    const slider = document.getElementById('varSitesThreshold');
    const input = document.getElementById('varSitesThresholdInput');
    const nSeq = state.seqs ? state.seqs.length : 0;
    const max = mode === 'count' ? Math.max(nSeq - 1, 1) : 100;   // at most nSeq - 1 can differ
    [slider, input].forEach(elRef => {
        if (!elRef) return;
        elRef.max = String(max);

    });

function searchMotif(options = {}) {
    if (!isMotifSearchSupported()) {
        if (!options.suppressMessage) {
            showMessage('Motif search works in Full, Block and Canvas: switch out of Reads.', 3500);
        }
        return null;
    }
    const raw = options.motif ?? (el('searchInput').value || '');
    const useRegex = options.useRegex ?? el('searchRegex')?.checked;
    // A regex keeps its case: upper-casing turns \w, \s, \d into their negations.
    // Matching is case-insensitive either way (flag i, and sequences are upper-cased).
    const motif = useRegex ? raw.trim() : raw.trim().replace(/\s+/g, '').toUpperCase();
    if (!motif) {
        if (!options.suppressMessage) showMessage('Type a motif in the Search box first.', 2500);
        return;
    }
    if (useRegex) {
        try { new RegExp(motif, 'gi'); } catch (e) {
            showMessage('Invalid regex: ' + e.message, 4000);
            return null;
        }
    }
    const color = options.color || el('searchColor').value;
    const maxMismatches = Math.max(0, options.maxMismatches ?? (parseInt(el('maxMismatches').value) || 0));   // a negative box value matched nothing, not even exact hits
    const checkboxEl = el('searchBothStrands');
    // A regex is not a sequence: complementing its text letter by letter turns [AT] into
    // N...N. Restriction sites build their own reverse-complement regex instead.
    const bothStrandsAsked = options.bothStrands ?? (checkboxEl && checkboxEl.checked);
    // A protein has no reverse strand: complementing KRQ gave NYM and searched for that.
    const protein = isProteinAlignment();
    const bothStrands = bothStrandsAsked && !useRegex && !protein;
    const regexSingleStrand = bothStrandsAsked && (useRegex || protein);

    debugLog('=== SEARCH MOTIF STARTED ===');
    debugLog('Input motif:', motif);
    debugLog('Checkbox element found:', !!checkboxEl);
    debugLog('Checkbox checked:', bothStrands);
    debugLog('===============================');

    const fwdMotifRaw = motif;
    const fwdMotif = isProteinAlignment() ? fwdMotifRaw : fwdMotifRaw.replace(/U/g, 'T');   // U is selenocysteine in proteins
    const baseLabel = options.label || fwdMotifRaw;
    const baseKey = options.key || fwdMotif;
    const motifsToSearch = [{
        motif: fwdMotif,
        color,
        label: bothStrands ? `${baseLabel} (fwd)` : baseLabel,
        strand: options.strand || 'fwd',
        key: baseKey
    }];
    if (bothStrands) {
        const rcMotifRaw = reverseComplement(fwdMotifRaw);
        const rcMotif = rcMotifRaw.replace(/U/g, 'T');
        debugLog('Both strands enabled. Reverse complement:', { fwdMotif, fwdMotifRaw, rcMotif, rcMotifRaw });
        if (rcMotif && rcMotif !== fwdMotif) {
            debugLog('Adding reverse complement motif to search list');
            motifsToSearch.push({
                motif: rcMotif,
                color: getComplementaryColor(color),
                label: `${rcMotifRaw} (rev comp)`,
                strand: 'rev comp',
                key: `${baseKey}:rev`
            });
        } else {
            debugLog('NOT adding reverse complement (palindromic or empty)');
        }
    } else {
        debugLog('Both strands NOT checked');
    }
    debugLog('Final motifs to search:', motifsToSearch.map(m => ({ motif: m.motif, label: m.label })));

    let totalMatches = 0;
    let fwdMatches = 0;
    let revMatches = 0;
    let fwdSeqs = new Set();
    let revSeqs = new Set();
    const searchResults = [];

    // Degap every row once; all motifs of this search reuse it. Counts come from state,
    // not from the rendered spans, so windowed views count rows that are off screen.
    const degapped = state.seqs.map(sq => _degapForSearch(sq.seq));

    motifsToSearch.forEach(({ motif: searchMotifValue, color: searchColorValue, label, strand, key }) => {
        let motifMatches = 0;
        const motifAlignmentSites = new Set();
        const motifRowSites = new Set();
        const motifSeqsWithMatches = new Set();
        const motifKey = `${key || searchMotifValue}:${strand}`;
        const className = 'search-hit-' + Math.random().toString(36).substring(2, 12);
        const entry = {
            id: className, motif: motifKey, className, color: searchColorValue, label, strand,
            searchValue: searchMotifValue, useRegex: !!useRegex,
            maxMismatches: useRegex ? 0 : maxMismatches, enabled: true,
            ...(options.overlapping ? { overlapping: true } : {}),
            // In a nucleotide motif R, Y, N ... stand for several bases (in protein they are residues).
            // A plain search compared letters exactly, so GANTC found nothing in GAATC/GATTC.
            ...(!useRegex && !protein && /[RYSWKMBDHVN]/.test(searchMotifValue) ? { iupac: true } : {})
        };

        // Re-running an identical search replaces it rather than stacking a copy
        state.searchHistory = state.searchHistory.filter(item => {
            if (item.motif === motifKey) { _removeSearchStyle(item); return false; }
            return true;
        });

        let invalid = false;
        const isRev = String(strand || '').includes('rev comp');
        degapped.forEach(({ cols, text }, index) => {
            if (invalid || !text) return;
            const matches = _searchMatchesInDegapped(text, entry);
            if (matches === null) { invalid = true; return; }
            if (!matches.length) return;
            motifSeqsWithMatches.add(index);
            (isRev ? revSeqs : fwdSeqs).add(index);
            totalMatches += matches.length;
            motifMatches += matches.length;
            if (isRev) revMatches += matches.length; else fwdMatches += matches.length;
            matches.forEach(m => {
                const startCol = cols[m.idx];
                if (startCol !== undefined) { motifAlignmentSites.add(`${startCol}:${m.len}`); motifRowSites.add(`${index}:${startCol}:${m.len}`); }
            });
        });
        if (invalid) return;   // validated above; kept as a guard

        const displayMatches = options.countMode === 'alignmentSites' ? motifAlignmentSites.size : motifMatches;
        Object.assign(entry, { matchCount: displayMatches, rawMatchCount: motifMatches, sequencesWithMatches: motifSeqsWithMatches.size, countMode: options.countMode || null });
        state.searchHistory.push(entry);
        searchResults.push({ label, strand, matchCount: displayMatches, rawMatchCount: motifMatches, sequencesWithMatches: motifSeqsWithMatches.size, sequenceIndices: Array.from(motifSeqsWithMatches), alignmentSites: Array.from(motifAlignmentSites), rowSites: Array.from(motifRowSites) });
    });
    bumpSearchLayer();
    reapplySearchHighlights();

    updateActiveSearchesPanel();
    const coverNote = options.suppressMessage ? '' : overlayCoverageSentence();
    if (!options.suppressMessage) {
        if (bothStrands) {
            // a sequence matching on both strands is one sequence
            const nSeqs = new Set([...fwdSeqs, ...revSeqs]).size;
            showMessage(
                `Found ${totalMatches} match${totalMatches !== 1 ? 'es' : ''} (fwd:${fwdMatches}, rev comp:${revMatches}) in ${nSeqs} sequence${nSeqs !== 1 ? 's' : ''}`
                + (coverNote ? '. ' + coverNote : ''),
                coverNote ? 5000 : 3000
            );
        } else {
            showMessage(`Found ${totalMatches} match${totalMatches !== 1 ? 'es' : ''} in ${fwdSeqs.size} sequence${fwdSeqs.size !== 1 ? 's' : ''}`
                + (regexSingleStrand ? (protein ? ' (protein: forward strand only)' : ' (regex: forward strand only)') : '') + (coverNote ? '. ' + coverNote : ''),
                (regexSingleStrand || coverNote) ? 5000 : 2000);
        }
    }
    return { totalMatches, fwdMatches, revMatches, fwdSeqs: fwdSeqs.size, revSeqs: revSeqs.size, results: searchResults };
}

// The hit counts of the active searches follow edits: they were set once, at search time, so a search
// with no hit kept '(0)' (and no 'go' arrow) after an edit made matches. Recounted a moment after an edit
// (pushUndo / pushUndoRowPatch / Undo / Redo), over the same rows the search covers.
let _searchRecountTimer = null;
function _scheduleSearchRecount() {
    if (!state.searchHistory || !state.searchHistory.length) return;
    clearTimeout(_searchRecountTimer);
    _searchRecountTimer = setTimeout(_recountSearches, 600);
}
function _recountSearches() {
    if (!state.searchHistory || !state.searchHistory.length) return;
    const degapped = state.seqs.map(q => _degapForSearch(q.seq));
    let changed = false;
    for (const entry of state.searchHistory) {
        let raw = 0, seqsWith = 0; const sites = new Set();
        let invalid = false;
        degapped.forEach(({ cols, text }) => {
            if (invalid || !text) return;
            const m = _searchMatchesInDegapped(text, entry);
            if (m === null) { invalid = true; return; }
            if (!m.length) return;
            seqsWith++; raw += m.length;
            m.forEach(x => { if (cols[x.idx] !== undefined) sites.add(cols[x.idx] + ':' + x.len); });
        });
        if (invalid) continue;
        const shown = entry.countMode === 'alignmentSites' ? sites.size : raw;
        if (shown !== entry.matchCount || seqsWith !== entry.sequencesWithMatches) changed = true;
        Object.assign(entry, { matchCount: shown, rawMatchCount: raw, sequencesWithMatches: seqsWith });
    }
    if (changed) updateActiveSearchesPanel();
}

function updateActiveSearchesPanel() {
    const panel = el('activeSearches');
    const list = el('searchList');
    if (panel && list) {
        if (state.searchHistory.length === 0) {
            panel.style.display = 'none';
        } else {
            panel.style.display = 'block';
            list.innerHTML = '';
            state.searchHistory.forEach(search => list.appendChild(_buildSearchListItem(search)));
        }
    }
    if (typeof refreshSelectionsPanel === 'function') refreshSelectionsPanel();
}

function _buildSearchListItem(search) {
    const item = document.createElement('div');
    item.className = 'search-item' + (search.enabled === false ? ' is-off' : '');
    const toggle = document.createElement('input');
    toggle.type = 'checkbox';
    toggle.checked = search.enabled !== false;
    toggle.title = 'Show or hide this search';
    toggle.addEventListener('change', () => setSearchEnabled(search, toggle.checked));
    const swatch = document.createElement('div');
    swatch.className = 'search-swatch';
    swatch.style.backgroundColor = search.color;
    const text = document.createElement('span');
    text.textContent = `${search.label || search.motif} (${search.matchCount || 0})`;
    const remove = document.createElement('button');
    remove.className = 'search-remove';
    remove.textContent = 'x';
    remove.title = 'Remove this search';
    remove.addEventListener('click', () => { _rememberForUndo('search ' + (search.label || search.searchValue || '')); removeSearch(search); scheduleSelectionsPanelRefresh(); });
    item.append(toggle, swatch, text, remove);
    return item;
}

function setSearchEnabled(search, on) {

function removeSearch(search) {
    const i = state.searchHistory.indexOf(search);
    if (i < 0) return;
    state.searchHistory.splice(i, 1);
    bumpSearchLayer();
    reapplySearchHighlights();
    _removeSearchStyle(search);
    updateActiveSearchesPanel();
}

// ---- Named search presets (owner, 2026-10-23): "for complex search tasks ... named search presets should be
// saved/loaded (example: promoter boxes)". A preset is the set of queries behind the active searches: motif text,
// regex flag, mismatches, both strands, colour and a label; loading runs each through searchMotif again (which
// derives IUPAC handling and the strand entries, replaces an entry with the same motif instead of doubling it,
// and refreshes the panels), so a preset works on any alignment. Stored like the colour and clustering presets.
const SEARCH_PRESETS_KEY = 'searchPresets';
function _searchPresets() {
    const p = safeLocalGet(SEARCH_PRESETS_KEY, {});
    // a null-prototype store: a preset named __proto__ or constructor is a key like any other
    const out = Object.create(null);
    if (p && typeof p === 'object' && !Array.isArray(p)) Object.keys(p).forEach(k => { out[k] = p[k]; });
    return out;
}
// The queries behind the active searches: a reverse-complement entry joins its forward one as "both strands"
function _searchQueriesFromActive() {
    const queries = [], byBase = new Map();
    const base = h => String(h.motif || '').replace(/:(fwd|rev comp)$/, '').replace(/:rev$/, '');
    (state.searchHistory || []).forEach(h => {
        const key = base(h);
        if (h.strand === 'rev comp') { const q = byBase.get(key); if (q) { q.bothStrands = true; return; } }
        const useRegex = !!h.useRegex;
        const motif = useRegex ? String(h.searchValue || '') : String(h.searchValue || '').toUpperCase();
        if (!motif) return;
        const shownLabel = String(h.label || '').replace(/ \((fwd|rev comp)\)$/, '');
        const q = { motif, useRegex, maxMismatches: useRegex ? 0 : (parseInt(h.maxMismatches, 10) || 0), bothStrands: false,
            color: _safeCssColor(h.color) || '#ffcc00', label: shownLabel && shownLabel !== motif ? shownLabel : '' };
        if (h.overlapping) q.overlapping = true;                 // restriction sites: every site, counted per column
        if (h.countMode) q.countMode = String(h.countMode);
        queries.push(q);
        if (h.strand !== 'rev comp') byBase.set(key, q);
    });
    return queries;
}

function saveSearchPreset() {
    const inp = el('searchPresetName');
    const name = (inp?.value || '').trim().slice(0, 60);
    if (!name) { showMessage('Type a name for the preset first.', 2500); return; }
    const queries = _searchQueriesFromActive();
    if (!queries.length) { showMessage('No active searches to save: run the searches first, then save them under a name.', 3500); return; }
    const presets = _searchPresets();
    const replaced = !!presets[name];
    presets[name] = { savedAt: new Date().toISOString(), queries };
    if (!safeLocalSet(SEARCH_PRESETS_KEY, JSON.stringify(presets), 'the search preset')) return;
    if (inp) inp.value = '';
    updateSearchPresetList(name);
    showMessage(`Search preset "${name}" ${replaced ? 'replaced' : 'saved'} (${queries.length} ${queries.length === 1 ? 'query' : 'queries'}).`, 2500);
}
function loadSearchPreset() {
    const name = el('searchPresetSelect')?.value;
    const p = name ? _searchPresets()[name] : null;
    if (!p || !Array.isArray(p.queries)) { showMessage('Choose a saved preset first.', 2500); return; }
    if (!state.seqs.length) { showMessage('Load an alignment first.', 2500); return; }
    let ran = 0;
    p.queries.forEach(q => {
        if (!q || typeof q.motif !== 'string' || !q.motif) return;
        const r = searchMotif({ motif: q.motif, useRegex: !!q.useRegex, maxMismatches: Math.max(0, parseInt(q.maxMismatches, 10) || 0),
            bothStrands: !!q.bothStrands, color: _safeCssColor(q.color) || '#ffcc00', label: q.label || undefined, suppressMessage: true,
            ...(q.overlapping ? { overlapping: true } : {}), ...(q.countMode ? { countMode: String(q.countMode) } : {}) });
        if (r !== null) ran++;
    });
    const hits = (state.searchHistory || []).reduce((n, h) => n + (h.matchCount || 0), 0);
    showMessage(`Preset "${name}": ${ran} ${ran === 1 ? 'search' : 'searches'} run; ${hits} ${hits === 1 ? 'hit' : 'hits'} in all active searches.`, 3000);
}
function deleteSearchPreset() {
    const name = el('searchPresetSelect')?.value;
    const presets = _searchPresets();
    if (!name || !presets[name]) { showMessage('Choose a saved preset first.', 2500); return; }
    delete presets[name];
    if (!safeLocalSet(SEARCH_PRESETS_KEY, JSON.stringify(presets), 'the search presets')) return;
    updateSearchPresetList();
    showMessage(`Search preset "${name}" removed.`, 2000);
}
