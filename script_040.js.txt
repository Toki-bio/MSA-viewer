                break;
            case 'f':
                if (e.shiftKey) {
                    clearLastSearch();
                } else if (e.altKey) {
                    clearAllSearches();
                } else {
                    // Ctrl+F opens the Search box (Canvas and Block draw letters as pixels, so the
                    // browser's find bar cannot see them) and runs the search when it holds a motif
                    const inp = el('searchInput');
                    const sec = inp?.closest('.menu-section');
                    if (sec) openMenuSection(sec);
                    if (inp && inp.value.trim()) searchMotif();
                    if (inp) { inp.focus(); inp.select(); }
                }
                e.preventDefault();
                break;
            case 'k':
                if (e.shiftKey) {
                    copyConsensus();
                    e.preventDefault();
                }
                break;
            case 'j':
                if (e.shiftKey) {   // alias of Alt+J (browsers keep Ctrl+Shift+J for DevTools)
                    copySelectedConsensus();
                    e.preventDefault();
                }
                break;
            case 'v':
                if (e.shiftKey) {
                    copySelectedColumns();
                    e.preventDefault();
                }
                break;
            case '+':
            case '=': // Ctrl with the main-row +/= key reports '=' without Shift
                adjustZoom(10);
                e.preventDefault();
                break;
            case '-':
                adjustZoom(-10);
                e.preventDefault();
                break;
            case '0':
                resetZoom();
                e.preventDefault();
                break;
            case 'm':
                // Toggle all menu sections
                toggleAllSections();
                e.preventDefault();
                break;
        }
    }
    if (e.ctrlKey || e.metaKey) {
        if (e.key === 'ArrowUp') {
            moveSelectedUp();
            e.preventDefault();
        } else if (e.key === 'ArrowDown') {
            moveSelectedDown();
            e.preventDefault();
        } else if (e.key === 'Home') {
            moveSelectedToTop();
            e.preventDefault();
        } else if (e.key === 'End') {
            moveSelectedToBottom();
            e.preventDefault();
        }
    }
