/**
 * Compute the total span of a read across the reference from its CIGAR.
 * Sum M, D, N operations (but not I or S).
 */
// At most _READS_PER_WINDOW reads starting in each 100-bp window are drawn (igv.js: sampling windows with a reservoir);
// the rest stay in bamState.allReads. Deterministic (fixed seed). reads must be sorted by pos.
const _READS_PER_WINDOW = 50;
function _downsampleReads(reads, perWindow = _READS_PER_WINDOW, windowBp = 100) {
    if (reads.length <= perWindow) return { shown: reads, dropped: [] };
    let seed = 12345;
    const rnd = () => (seed = (Math.imul(seed, 1103515245) + 12345) & 0x7fffffff) / 0x80000000;
    const shown = [], dropped = [];
    for (let i = 0; i < reads.length;) {
        const w = Math.floor(reads[i].pos / windowBp);
        let j = i;
        while (j < reads.length && Math.floor(reads[j].pos / windowBp) === w) j++;
        if (j - i <= perWindow) { for (let k = i; k < j; k++) shown.push(reads[k]); }
        else {
            const res = reads.slice(i, i + perWindow);
            for (let k = i + perWindow; k < j; k++) { const r = Math.floor(rnd() * (k - i + 1)); if (r < perWindow) res[r] = reads[k]; }
            res.sort((a, b) => a.pos - b.pos || (a.name < b.name ? -1 : a.name > b.name ? 1 : 0));
            shown.push(...res);
            dropped.push({ start: w * windowBp, end: (w + 1) * windowBp, dropped: j - i - perWindow });
        }
        i = j;
    }
    return { shown, dropped };
}

function computeReadSpan(cigar) {
    let span = 0;
    for (const op of cigar) {
        if (op.op === 'M' || op.op === '=' || op.op === 'X' || op.op === 'D' || op.op === 'N') {
            span += op.len;
        }
    }
    return span;
}

/**
 * Parse SAM text into a format compatible with the BAM parser's buffer output.
 * Simplified: creates a synthetic buffer-like object with SAM records.
 * In practice, we parse SAM as text and build record objects directly.
 */
function parseSAMToBuffer(text) {
    // Parse SAM text into BAM-like records for the reads view pipeline.
    // Returns a pseudo-buffer object with header and records compatible with
    // BamParser.parseBAMHeader and BamParser.parseBAMRecords.
    // CRLF line ends (a SAM saved on Windows): split on '\n' alone, the last field kept its '\r', so QUAL '*' was '*\r',
    // read as a quality string of low values: every base failed the quality minimum, a row of N and no variants
    const lines = text.split(/\r?\n/).filter(l => l.trim());
    const headerLines = lines.filter(l => l.startsWith('@'));
    const readLines = lines.filter(l => !l.startsWith('@'));

    // Parse @SQ headers to get reference names and lengths
    const refNames = [];
    const refLengths = [];
    for (const line of headerLines) {
        if (line.startsWith('@SQ')) {
            const fields = line.split('\t');
            let name = '', len = 0;
            for (const f of fields) {
                if (f.startsWith('SN:')) name = f.substring(3);
                else if (f.startsWith('LN:')) len = parseInt(f.substring(3)) || 0;
            }
            if (name) { refNames.push(name); refLengths.push(len); }
        }
    }

    // No @SQ lines (samtools view without -h): the references are the reads' RNAME values, each as long as its
    // reads reach (nothing matched before, so a headerless SAM never loaded)
    if (!refNames.length) {
        const end = new Map();
        for (const line of readLines) {
            const f = line.split('\t');
            if (f.length < 11 || f[2] === '*') continue;
            const e = (parseInt(f[3]) || 1) - 1 + _cigarRefSpan(f[5] || '');
            end.set(f[2], Math.max(end.get(f[2]) || 0, e));
        }
        for (const [n, e] of end) { refNames.push(n); refLengths.push(e); }
    }

    // Parse alignment records; reads that are not evidence (unmapped, secondary, QC-failed, duplicate,
    // supplementary: BamParser.skipReason) are counted and dropped, as for BAM (duplicates were kept)
    const records = [];
    const skipped = {};
    let totalReads = 0;
    for (const line of readLines) {
        const f = line.split('\t');
        if (f.length < 11) continue;
        totalReads++;
        const flag = parseInt(f[1]) || 0;
        const why = BamParser.skipReason(flag, true) || (!f[5] || f[5] === '*' || !f[9] || f[9] === '*' ? 'no CIGAR or sequence' : null);
        if (why) { skipped[why] = (skipped[why] || 0) + 1; continue; }

        const rname = f[2];
        const refID = refNames.indexOf(rname);
        const pos = parseInt(f[3]) - 1; // 0-based
        const mapq = parseInt(f[4]) || 0;
        const cigarStr = f[5];
        const seq = f[9];
        const tlen = parseInt(f[8]) || 0;
        const nextRef = f[6] === '=' ? refID : (f[6] !== '*' ? refNames.indexOf(f[6]) : -1);
        const nextPos = f[7] !== '*' && f[7] !== '=' ? parseInt(f[7]) - 1 : (f[7] === '=' ? pos : -1);

        // Parse CIGAR string into ops array
        const cigarOps = [];
        const cigarMatches = cigarStr.match(/\d+[MIDNSHP=X]/g) || [];
        for (const cm of cigarMatches) {
            cigarOps.push({ len: parseInt(cm), op: cm[cm.length - 1] });
        }

        // Parse optional tags
        const tags = {};
        for (let i = 11; i < f.length; i++) {
            const tagMatch = f[i].match(/^([A-Za-z][A-Za-z0-9]):([AifZHB]):(.*)$/);
            if (tagMatch) {
                const [, tag, type, value] = tagMatch;
                if (type === 'i' || type === 'I') tags[tag] = parseInt(value);
                else if (type === 'f') tags[tag] = parseFloat(value);
                else tags[tag] = value;
            }
        }

        records.push({
            name: f[0],
            refID,
            pos,
            mapq,
            flag,
            seq,
            qual: f[10] || '',
            cigar: cigarOps,
            tlen,
            next_refID: nextRef,
            next_pos: nextPos,
            tags,
        });
    }

    // Return a pseudo-buffer object compatible with the BAM pipeline
    return { _isSAM: true, _refNames: refNames, _refLengths: refLengths, _records: records, _skipped: skipped, _totalReads: totalReads };
}
